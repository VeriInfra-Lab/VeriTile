def MainClaim (actual reference : StorageOps (BitVec 32) (BitVec 16))
    (castA castR : HostCast (BitVec 32) (BitVec 16)) : Prop :=
  ∀ {σ : Type} {s : Shape} (abi : LocalEventABI (BitVec 32) (BitVec 16) σ actual)
    (p : Parameters (BitVec 32)) (b : Buffers) (domain : HostDomain s p b)
    (binding : CompilerBinding (BitVec 32) (BitVec 16) s p b) (state : σ)
    (caller : Memory (BitVec 32) (BitVec 16))
    (prep : Preparation p.K castA domain.sourceDType domain.originalRms b caller (abi.observe state)),
    (∀ x : Cell s,abi.observe ((hostCode s binding).native abi state).2 (outputAddress b x)=
      hostTarget reference castR domain.sourceDType domain.originalRms p b caller x) ∧
    (∀ q,(∀ x : Cell s,q≠outputAddress b x) →
      abi.observe ((hostCode s binding).native abi state).2 q=abi.observe state q) ∧
    (∀ q,q.1≠b.output.storage → (domain.sourceDType=weightDType b ∨ q.1≠b.rms.storage) →
      abi.observe ((hostCode s binding).native abi state).2 q=caller q)

theorem main_claim (actual reference : StorageOps (BitVec 32) (BitVec 16))
    (leaves : NumericLeaves actual reference)
    (castA castR : HostCast (BitVec 32) (BitVec 16)) (castLeaf : HostCastLeaf castA castR) :
    MainClaim actual reference castA castR := by
  intro σ s abi p b domain binding state caller prep
  exact whole_host actual reference leaves castA castR castLeaf abi p b domain binding state caller prep

theorem reference_consistent (a : StorageOps (BitVec 32) (BitVec 16)) : NumericLeaves a a := by
  constructor
  · constructor <;> intros <;> rfl
  all_goals intros; rfl

def WordClaim (a ref : Primitives) : Prop := ∀ {Store : Type}, ∀ (api : Memory.API Word Store (BitVec 8))
    (r : Request) (input output : RegionName) (s : State Tagged)
    (call : Native.Preparation api leaf r input s)
    (abi : Native.IndexABI api leaf r input s call)
    (codec : ∀ w k, api.encode w k=wordByte w k),
    (∀ b m c, b<r.batches → m<r.rows → c<r.columns → ∀ k : Fin 4,
      api.byte (Native.entry api leaf Tagged.word (operations a) r input output s call).memory
        (Native.logicalAddress call.output b m c+k.val) =
          wordByte (result ref r (fun o => ((s.mem input o).read .fp32).word) b m c) k) ∧
    (∀ b, call.caller b → api.byte (Native.entry api leaf Tagged.word (operations a) r input output s call).memory b=api.byte call.before b)
theorem word_claim (a ref : Primitives) (numeric : Rotary.Correspondence a ref) : WordClaim a ref := by
  intro Store api r input output s call abi codec
  exact Rotary.whole_reference api a ref r input output s call abi codec numeric
end RawClaims

def MainBytesClaim (actual reference : StorageOps (BitVec 32) (BitVec 16))
    (castA castR : HostCast (BitVec 32) (BitVec 16)) : Prop :=
  ∀ {σ : Type} {s : Shape} (abi : LocalEventABI (BitVec 32) (BitVec 16) σ actual)
    (raw : RawByteABI abi) (p : Parameters (BitVec 32)) (b : Buffers) (domain : HostDomain s p b)
    (binding : CompilerBinding (BitVec 32) (BitVec 16) s p b) (state : σ)
    (caller : Memory (BitVec 32) (BitVec 16))
    (prep : Preparation p.K castA domain.sourceDType domain.originalRms b caller (abi.observe state)),
    (∀ (x : Cell s) (i : Fin (raw.width (outputAddress b x).1)),
      raw.bytes ((hostCode s binding).native abi state).2
        ((outputAddress b x).1,raw.width (outputAddress b x).1*(outputAddress b x).2+i.val)=
      cellByte (hostTarget reference castR domain.sourceDType domain.originalRms p b caller x) i.val) ∧
    (∀ storage offset,(∀ x : Cell s,(storage,offset/raw.width storage)≠outputAddress b x) →
      raw.bytes ((hostCode s binding).native abi state).2 (storage,offset)=raw.bytes state (storage,offset))

theorem main_bytes_claim (actual reference : StorageOps (BitVec 32) (BitVec 16))
    (leaves : NumericLeaves actual reference)
    (castA castR : HostCast (BitVec 32) (BitVec 16)) (castLeaf : HostCastLeaf castA castR) :
    MainBytesClaim actual reference castA castR := by
  intro σ s abi raw p b domain binding state caller prep
  constructor
  · intro x i
    rw [native_output_bytes actual reference leaves abi raw p b domain binding state x i,
      prepared_target actual reference castA castR castLeaf domain.sourceDType domain.originalRms p b caller (abi.observe state) prep x]
  · exact native_full_byte_frame actual abi raw binding state

def MainMetadataClaim : Prop :=
  (∀ {s : Shape} {p : Parameters (BitVec 32)} {b : Buffers} (domain : HostDomain s p b),
    (returned s b domain.device domain.extents).storage=b.output.storage ∧
    (returned s b domain.device domain.extents).shape=[s.B,s.M,s.heads,s.headDim] ∧
    (returned s b domain.device domain.extents).dtype=weightDType b ∧
    (returned s b domain.device domain.extents).shape.prod=s.B*s.M*s.N) ∧
  (∀ {σ : Type} device (body : Nat × σ → Nat × σ) before,(withDevice device body before).1=before.1)
theorem main_metadata_claim : MainMetadataClaim := ⟨return_view,device_restored⟩

def WholeNative (a ref : StorageOps (BitVec 32) (BitVec 16))
    (castA castR : HostCast (BitVec 32) (BitVec 16))
    (rawA rawR : VeriTile.Bench.Repairs.RbeTritonTransform.Numerics.Primitives) : Prop :=
  MainClaim a ref castA castR ∧ MainBytesClaim a ref castA castR ∧ MainMetadataClaim ∧ RawClaims.WordClaim rawA rawR

theorem whole_native (a : StorageOps (BitVec 32) (BitVec 16))
    (cast : HostCast (BitVec 32) (BitVec 16))
    (raw : VeriTile.Bench.Repairs.RbeTritonTransform.Numerics.Primitives) :
    WholeNative a a cast cast raw raw := by
  exact ⟨main_claim a a (reference_consistent a) cast cast ⟨fun _ _ _ => rfl⟩,
    main_bytes_claim a a (reference_consistent a) cast cast ⟨fun _ _ _ => rfl⟩,
    main_metadata_claim, RawClaims.word_claim raw raw (Rotary.reference_consistent raw)⟩

theorem whole_fp (actual reference : StorageOps (BitVec 32) (BitVec 16))
    (leaves : NumericLeaves actual reference)
    (castA castR : HostCast (BitVec 32) (BitVec 16)) (castLeaf : HostCastLeaf castA castR)
    (rawA rawR : VeriTile.Bench.Repairs.RbeTritonTransform.Numerics.Primitives)
    (rawLeaves : Rotary.Correspondence rawA rawR) :
    WholeNative actual reference castA castR rawA rawR :=
  ⟨main_claim actual reference leaves castA castR castLeaf,
    main_bytes_claim actual reference leaves castA castR castLeaf,main_metadata_claim,
    RawClaims.word_claim rawA rawR rawLeaves⟩
