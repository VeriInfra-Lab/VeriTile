import bench.tritonbench_g.relu_triton_kernel.Proofs.AccessMemory

/-! Ordered byte loads for every active lane, including nonzero program ids.
All reads use Except bind before a program may inspect its pid-zero store gate.
Unmasked inactive lanes perform no byte access in this selected model. -/
namespace VeriTile.Bench.TritonBenchG.ReluTritonKernel.AccessMemory
open VeriTile.Triton FP.Structural IntegerWords FiniteAddress ByteMemory

def loadInput (encoding : Encoding) (mem : Memory) (base offset : Nat) : Except Fault (Word encoding) := do
  let word ← AccessMemory.loadLE (elementBytes encoding) mem (base + offset * elementBytes encoding)
  pure (word.cast (element_bytes_width encoding))

theorem load_input_success (encoding : Encoding) (mem : Memory) (base offset : Nat)
    (available : Readable mem (base + offset * elementBytes encoding) (elementBytes encoding)) :
    loadInput encoding mem base offset = .ok (ByteMemory.loadInput encoding mem.bytes base offset) := by
  unfold loadInput
  rw [load_success _ _ _ available]
  rfl

theorem load_input_first_unavailable (encoding : Encoding) (mem : Memory) (base offset : Nat)
    (unavailable : ¬ (mem.allocated (base + offset * elementBytes encoding) = Bool.true ∧
      mem.readable (base + offset * elementBytes encoding) = Bool.true)) :
    loadInput encoding mem base offset = .error (.loadUnavailable (base + offset * elementBytes encoding)) := by
  have positive := element_bytes_positive encoding
  unfold loadInput
  have failure (count : Nat) (nonempty : 0 < count) :
      AccessMemory.loadLE count mem (base + offset * elementBytes encoding) =
      .error (.loadUnavailable (base + offset * elementBytes encoding)) := by
    cases count with
    | zero => omega
    | succ count => exact load_first_unavailable _ _ _ unavailable
  rw [failure (elementBytes encoding) positive]
  rfl

def loadWords (encoding : Encoding) (mem : Memory) (base start : Nat) :
    Nat → Except Fault (Nat → Word encoding)
  | 0 => .ok (fun _ => 0)
  | count + 1 => do
    let previous ← loadWords encoding mem base start count
    let word ← loadInput encoding mem base (start + count)
    pure (fun lane => if lane = count then word else previous lane)

def loadedPrefix (encoding : Encoding) (mem : Memory) (base start count : Nat) : Nat → Word encoding :=
  fun lane => if lane < count then ByteMemory.loadInput encoding mem.bytes base (start + lane) else 0

theorem load_words_success (encoding : Encoding) (mem : Memory) (base start count : Nat)
    (available : ∀ lane, lane < count →
      Readable mem (base + (start + lane) * elementBytes encoding) (elementBytes encoding)) :
    loadWords encoding mem base start count = .ok (loadedPrefix encoding mem base start count) := by
  induction count with
  | zero =>
    have zero : loadedPrefix encoding mem base start 0 = fun _ => 0 := by
      funext lane; simp [loadedPrefix]
    rw [zero]
    rfl
  | succ count ih =>
    have earlier := ih (fun lane within => available lane (by omega))
    rw [loadWords, earlier]
    rw [load_input_success _ _ _ _ (available count (by omega))]
    change Except.ok (fun lane => if lane = count then _ else loadedPrefix encoding mem base start count lane) = _
    congr 1
    funext lane
    by_cases last : lane = count
    · subst lane
      simp [loadedPrefix]
    · have same : lane < count ↔ lane < count + 1 := by omega
      simp only [last, if_false, loadedPrefix, same]

theorem load_words_first_unavailable (encoding : Encoding) (mem : Memory) (base start count : Nat)
    (unavailable : ¬ (mem.allocated (base + start * elementBytes encoding) = Bool.true ∧
      mem.readable (base + start * elementBytes encoding) = Bool.true)) :
    loadWords encoding mem base start (count + 1) =
      .error (.loadUnavailable (base + start * elementBytes encoding)) := by
  induction count with
  | zero =>
    simp only [loadWords, Nat.add_zero]
    rw [load_input_first_unavailable _ _ _ _ unavailable]
    rfl
  | succ count ih =>
    rw [loadWords, ih]
    rfl

theorem readable_element (encoding : Encoding) (mem : Memory) (base N index : Nat)
    (available : Readable mem base (N * elementBytes encoding)) (active : index < N) :
    Readable mem (base + index * elementBytes encoding) (elementBytes encoding) := by
  intro byte within
  have bound := relative_byte_bound (elementBytes encoding) N index byte active within
  simpa only [Nat.add_assoc] using available (index * elementBytes encoding + byte) bound

def activeCount (N pid : Nat) : Nat := min blockSize (N - pid * blockSize)

theorem active_index (N pid lane : Nat) (active : lane < activeCount N pid) :
    pid * blockSize + lane < N := by
  unfold activeCount at active
  omega

theorem active_count_zero (N : Nat) : activeCount N 0 = min N blockSize := by
  simp [activeCount, Nat.min_comm]

def loadInputAt64 (encoding : Encoding) (mem : Memory) (address : BitVec 64) : Except Fault (Word encoding) := do
  let word ← AccessMemory.loadLE (elementBytes encoding) mem address.toNat
  pure (word.cast (element_bytes_width encoding))

theorem finite_input_access (encoding : Encoding) (mem : Memory) (base N pid lane : Nat)
    (indexDomain : IndexDomain N) (domain : ByteDomain base (elementBytes encoding) N)
    (launched : pid < GatedGrid.gridCount N blockSize) (inBlock : lane < blockSize)
    (active : offset pid lane < N) :
    loadInputAt64 encoding mem (pointer64 base (elementBytes encoding) pid lane) =
      loadInput encoding mem base (offset pid lane) := by
  unfold loadInputAt64 loadInput
  rw [pointer_exact base (elementBytes encoding) N pid lane indexDomain domain
    (element_bytes_positive encoding) launched inBlock active]

#specNonCircular loadedPrefix avoiding [loadWords, AccessMemory.loadInput, AccessMemory.loadLE, readByte]
#axiomsClean load_input_success
#axiomsClean load_input_first_unavailable
#axiomsClean load_words_success
#axiomsClean load_words_first_unavailable
#axiomsClean readable_element
#axiomsClean active_index
#axiomsClean active_count_zero
#axiomsClean finite_input_access
#guard_msgs (drop info) in
#auditModuleAxioms
end VeriTile.Bench.TritonBenchG.ReluTritonKernel.AccessMemory
