import bench.tritonbench_g.relu_triton_kernel.Proofs.AccessLoads

/-! Execution with partial accesses, including every active load of every pid.
Whole programs are sequential here; faults are mathematical access failures.
The existing numerical callbacks are total and have no additional side effects.
No native source/compiler, poison, trap or instruction-scheduling claim follows. -/
namespace VeriTile.Bench.TritonBenchG.ReluTritonKernel.AccessMemory
open VeriTile.Triton FP.Structural IntegerWords FiniteAddress ByteMemory GatedGrid

def program (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N pid : Nat) (mem : Memory) : Execution :=
  match loadWords encoding mem inputBase (pid * blockSize) (activeCount N pid) with
  | .error fault => .error (fault, mem)
  | .ok loaded =>
    if pid = 0 then
      storeWords (fun lane => wordValue M encoding (loaded lane)) outputBase mem (activeCount N pid)
    else .ok mem

theorem store_prefix_congr (left right : Nat → BitVec 32) (base : Nat)
    (mem : ByteMemory.Memory) (count : Nat) (same : ∀ lane, lane < count → left lane = right lane) :
    ByteMemory.storePrefix left base mem count = ByteMemory.storePrefix right base mem count := by
  rw [ByteMemory.prefix_correct, ByteMemory.prefix_correct]
  funext address
  unfold ByteMemory.prefixSpec
  by_cases within : base ≤ address ∧ address < base + count * 4
  · rw [if_pos within, if_pos within, same ((address - base) / 4) (by omega)]
  · rw [if_neg within, if_neg within]

theorem program_success (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N pid : Nat) (mem : Memory)
    (inputReadable : Readable mem inputBase (N * elementBytes encoding))
    (outputWritable : Writable mem outputBase (min N blockSize * 4)) :
    program M encoding inputBase outputBase N pid mem =
      .ok (withBytes mem (byteProgram M encoding inputBase outputBase N pid mem.bytes)) := by
  have loads := load_words_success encoding mem inputBase (pid * blockSize) (activeCount N pid)
    (fun lane active => readable_element encoding mem inputBase N (pid * blockSize + lane)
      inputReadable (active_index N pid lane active))
  rw [program, loads]
  by_cases first : pid = 0
  · subst pid
    simp only [if_true, active_count_zero]
    rw [store_words_success _ _ _ _ outputWritable]
    have same : ByteMemory.storePrefix
        (fun lane => wordValue M encoding (loadedPrefix encoding mem inputBase (0 * blockSize)
          (activeCount N 0) lane)) outputBase mem.bytes (min N blockSize) =
        ByteMemory.storePrefix (ByteMemory.values M encoding inputBase mem.bytes)
          outputBase mem.bytes (min N blockSize) := by
      apply store_prefix_congr
      intro lane active
      simp only [loadedPrefix, active_count_zero, active, if_true, Nat.zero_mul, Nat.zero_add, ByteMemory.values]
    rw [same]
    rfl
  · simp only [first, if_false, byteProgram]
    rfl

/-- The old memory projection cannot detect an unavailable nonzero-pid input.
The access execution fails on that first active byte before the store gate. -/
theorem nonzero_program_unavailable (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N pid : Nat) (mem : Memory) (nonzero : pid ≠ 0)
    (active : pid * blockSize < N)
    (unavailable : ¬ (mem.allocated (inputBase + pid * blockSize * elementBytes encoding) = Bool.true ∧
      mem.readable (inputBase + pid * blockSize * elementBytes encoding) = Bool.true)) :
    byteProgram M encoding inputBase outputBase N pid mem.bytes = mem.bytes ∧
    program M encoding inputBase outputBase N pid mem =
      .error (.loadUnavailable (inputBase + pid * blockSize * elementBytes encoding), mem) := by
  refine ⟨by simp only [byteProgram, nonzero, if_false], ?_⟩
  have positive : 0 < activeCount N pid := by
    unfold activeCount blockSize
    unfold blockSize at active
    omega
  have failure : loadWords encoding mem inputBase (pid * blockSize) (activeCount N pid) =
      .error (.loadUnavailable (inputBase + pid * blockSize * elementBytes encoding)) := by
    have h := load_words_first_unavailable encoding mem inputBase (pid * blockSize)
      (activeCount N pid - 1) unavailable
    simpa only [show activeCount N pid - 1 + 1 = activeCount N pid by omega] using h
  rw [program, failure]

def runPrefix (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N : Nat) (initial : Memory) : Nat → Execution
  | 0 => .ok initial
  | count + 1 => do
    let current ← runPrefix M encoding inputBase outputBase N initial count
    program M encoding inputBase outputBase N count current

def runGrid (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N : Nat) (initial : Memory) : Execution :=
  runPrefix M encoding inputBase outputBase N initial (gridCount N blockSize)

theorem prefix_success (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N count : Nat) (initial : Memory)
    (inputReadable : Readable initial inputBase (N * elementBytes encoding))
    (outputWritable : Writable initial outputBase (min N blockSize * 4)) :
    runPrefix M encoding inputBase outputBase N initial count =
      .ok (withBytes initial (bytePrefix M encoding inputBase outputBase N initial.bytes count)) := by
  induction count with
  | zero => rfl
  | succ count ih =>
    rw [runPrefix, ih]
    change program M encoding inputBase outputBase N count
      (withBytes initial (bytePrefix M encoding inputBase outputBase N initial.bytes count)) = _
    rw [program_success _ _ _ _ _ _ _ inputReadable outputWritable]
    rfl

theorem grid_success (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N : Nat) (initial : Memory)
    (inputReadable : Readable initial inputBase (N * elementBytes encoding))
    (outputWritable : Writable initial outputBase (min N blockSize * 4)) :
    runGrid M encoding inputBase outputBase N initial =
      .ok (withBytes initial (byteGrid M encoding inputBase outputBase N initial.bytes)) :=
  prefix_success M encoding inputBase outputBase N _ initial inputReadable outputWritable

theorem grid_partial_frame (M : Algebra (BitVec 32)) (encoding : Encoding)
    (inputBase outputBase N address : Nat) (initial : Memory)
    (outside : address < outputBase ∨ outputBase + min N blockSize * 4 ≤ address) :
    partialContents (withBytes initial (byteGrid M encoding inputBase outputBase N initial.bytes)) address =
      partialContents initial address := by
  unfold partialContents withBytes
  rw [byte_grid_frame M encoding inputBase outputBase N address initial.bytes outside]

#axiomsClean program_success

#axiomsClean prefix_success
#axiomsClean grid_success
#axiomsClean grid_partial_frame
#guard_msgs (drop info) in
#auditModuleAxioms
end VeriTile.Bench.TritonBenchG.ReluTritonKernel.AccessMemory
