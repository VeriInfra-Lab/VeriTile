import bench.tritonbench_g.vector_addition.Proofs.DTypeAdmission
open VeriTile.Bench.TritonBenchG.VectorAddition DTypeScope DTypeAdmission

def main : IO Unit := do
  for pair in pairs do
    let p := match promote pair.1 pair.2 with
      | none => "none"
      | some d => backendName d
    let outcome := match admission pair.1 pair.2 with
      | .signatureRejected arg d => "signature|" ++ reprStr arg ++ "|" ++ backendName d
      | .additionRejected => "addition||"
      | .passesDTypeGates d s => "admitted|" ++ backendName d ++ "|" ++ backendName s
    IO.println (backendName pair.1 ++ "|" ++ backendName pair.2 ++ "|" ++ p ++ "|" ++ outcome)
