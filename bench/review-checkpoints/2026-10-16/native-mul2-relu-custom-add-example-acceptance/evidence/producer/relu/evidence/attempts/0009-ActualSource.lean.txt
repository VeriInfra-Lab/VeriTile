inductive IndexExpr where
  | literal (n : Nat)
  | program (ax : Nat)
  | register (name : RegName)
  | lane (width : Nat)
  | add (a b : IndexExpr)
  | mul (a b : IndexExpr)
  deriving DecidableEq

def lowerIndex {shape : TileShape} : Op .nat shape → Option IndexExpr
  | .constNat n => some (.literal n)
  | .programId ax => some (.program ax)
  | .ref _ _ name => some (.register name)
  | .arange width => some (.lane width)
  | .add _ _ a b => return .add (←lowerIndex a) (←lowerIndex b)
  | .mul _ _ a b => return .mul (←lowerIndex a) (←lowerIndex b)
  | _ => none

inductive BoolExpr where
  | register (name : RegName)
  | lessConstant (a : IndexExpr) (n : Nat)
  | equal (a b : IndexExpr)
  deriving DecidableEq

def lowerBool {shape : TileShape} : Op .bool shape → Option BoolExpr
  | .ref _ _ name => some (.register name)
  | .lt .nat _ a (.constNat n) => return .lessConstant (←lowerIndex a) n
  | .eq .nat _ a b => return .equal (←lowerIndex a) (←lowerIndex b)
  | _ => none

def lowerMask {shape : TileShape} : MaskOpt .real shape → Option BoolExpr
  | .mask e => lowerBool e
  | _ => none

inductive Instruction where
  | indexAssign (name : RegName) (value : IndexExpr)
  | boolAssign (name : RegName) (value : BoolExpr)
  | load (name : RegName) (region : RegionName) (offset : IndexExpr) (mask : BoolExpr)
  | reluAssign (name input : RegName)
  | gatedStore (gate : BoolExpr) (region : RegionName) (offset : IndexExpr) (value : RegName) (mask : BoolExpr)
  deriving DecidableEq

def lowerStatement : Stmt → Option Instruction
  | .assign .nat _ name value => return .indexAssign name (←lowerIndex value)
  | .assign .bool _ name value => return .boolAssign name (←lowerBool value)
  | .assign .real _ name (.load .real (.region region offset) mask _) =>
      return .load name region (←lowerIndex offset) (←lowerMask mask)
  | .assign .real _ name (.where (.ge .real _ (.ref _ _ input) (.const zero))
      (.ref _ _ yes) (.broadcast (.const no) _)) =>
      if input=yes ∧ zero=0 ∧ no=0 then some (.reluAssign name input) else none
  | .ifThen gate [.store .real _ (.region region offset) (.ref _ _ value) mask] =>
      return .gatedStore (←lowerBool gate) region (←lowerIndex offset) value (←lowerMask mask)
  | _ => none

def lowerBody : List Stmt → Option (List Instruction)
  | [] => some []
  | statement::rest => return (←lowerStatement statement)::(←lowerBody rest)

def sourceCode (X O : RegionName) (n : Nat) : List Instruction :=
  [.indexAssign "pid" (.program 0),
   .indexAssign "block_start" (.mul (.register "pid") (.literal 1024)),
   .indexAssign "offsets" (.add (.register "block_start") (.lane 1024)),
   .boolAssign "mask" (.lessConstant (.register "offsets") n),
   .load "x" X (.register "offsets") (.register "mask"),
   .reluAssign "result" "x",
   .gatedStore (.equal (.register "pid") (.literal 0)) O (.register "offsets") "result" (.register "mask")]

theorem original_ast_lowering (X O : RegionName) (n : Nat) :
    lowerBody (relu_kernel X O n 1024).body=some (sourceCode X O n) := by
  simp [lowerBody,lowerStatement,lowerIndex,lowerBool,lowerMask,relu_kernel,ComputeKernel.body,
    ComputeKernel.toAlgKernel,ComputeKernel.toAlgorithm?,ComputeStmt.listToAlgorithm?,ComputeStmt.toAlgorithm?,
    sourceCode,bind,pure,Except.bind,Option.bind]
  all_goals norm_num

structure Registers (d : Storage) where
  indices : RegName → Nat → BitVec 32
  bools : RegName → Nat → Bool
  loaded : RegName → Nat → Word d
  results : RegName → Nat → Stage d
  memory : ByteMemory.Memory

def evalIndex (pid lane : Nat) (regs : RegName → Nat → BitVec 32) : IndexExpr → BitVec 32
  | .literal n => BitVec.ofNat 32 n
  | .program ax => if ax=0 then BitVec.ofNat 32 pid else 0
  | .register name => regs name lane
  | .lane _ => BitVec.ofNat 32 lane
  | .add a b => evalIndex pid lane regs a+evalIndex pid lane regs b
  | .mul a b => evalIndex pid lane regs a*evalIndex pid lane regs b

def evalBool (pid lane : Nat) (indices : RegName → Nat → BitVec 32)
    (bools : RegName → Nat → Bool) : BoolExpr → Bool
  | .register name => bools name lane
  | .lessConstant a n => mask n (evalIndex pid lane indices a)
  | .equal a b => evalIndex pid lane indices a == evalIndex pid lane indices b

structure Write where
  address : Nat
  value : BitVec 32
  deriving DecidableEq, Repr

def applyWrites : List Write → ByteMemory.Memory → ByteMemory.Memory
  | [],memory => memory
  | w::rest,memory => applyWrites rest (ByteMemory.storeLE 4 memory w.address w.value)

/-- Every loaded input word and result register is materialized before the
pid-zero gate. Only the gated store converts/stores destination fp32 words. -/
def step (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (pid : Nat) (s : Registers d) : Instruction → Registers d
  | .indexAssign name value => {s with indices := fun key lane ↦ if key=name then evalIndex pid lane s.indices value else s.indices key lane}
  | .boolAssign name value => {s with bools := fun key lane ↦ if key=name then evalBool pid lane s.indices s.bools value else s.bools key lane}
  | .load name region offset condition => {s with loaded := fun key lane ↦ if key=name then if evalBool pid lane s.indices s.bools condition then readWord d s.memory (address (pointers region) d.bytes (evalIndex pid lane s.indices offset)) else undef else s.loaded key lane}
  | .reluAssign name input => {s with results := fun key lane ↦ if key=name then selectSource b d (s.loaded input lane) else s.results key lane}
  | .gatedStore gate region offset value condition =>
      if evalBool pid 0 s.indices s.bools gate then
        {s with memory := (List.range 1024).foldl (fun memory lane ↦ if evalBool pid lane s.indices s.bools condition then ByteMemory.storeLE 4 memory (address (pointers region) 4 (evalIndex pid lane s.indices offset)) (castStore b d (s.results value lane)) else memory) s.memory}
      else s

def executeBody (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (pid : Nat) : List Instruction → Registers d → Registers d
  | [],s => s
  | instruction::rest,s => executeBody b d pointers undef pid rest (step b d pointers undef pid s instruction)

def runProgram (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (X O : RegionName) (n pid : Nat) (memory : ByteMemory.Memory) : Option ByteMemory.Memory := do
  let code ← lowerBody (relu_kernel X O n 1024).body
  let initial : Registers d := ⟨fun _ _ => 0,fun _ _ => Bool.false,fun _ _ => undef,
    fun _ _ => selectSource b d undef,memory⟩
  pure (executeBody b d pointers undef pid code initial).memory

/-- Initial-memory formula for every actual source request. -/
def programWrites (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n pid : Nat) (memory : ByteMemory.Memory) : List Write :=
  if stores32 pid then (lanes n pid).map fun lane =>
    ⟨laneAddress output 4 pid lane,scalarFormula b d (readWord d memory (laneAddress x d.bytes pid lane))⟩
  else []

theorem fold_stores (addresses : Nat → Nat) (values : Nat → BitVec 32) (condition : Nat → Bool)
    (jobs : List Nat) (memory : ByteMemory.Memory) :
    jobs.foldl (fun state lane => if condition lane then ByteMemory.storeLE 4 state (addresses lane) (values lane) else state) memory=
    applyWrites ((jobs.filter condition).map fun lane => ⟨addresses lane,values lane⟩) memory := by
  induction jobs generalizing memory with
  | nil => rfl
  | cons lane rest ih =>
    cases h : condition lane <;> simp [List.foldl_cons,List.filter_cons,h,applyWrites,ih]

theorem source_program_correspondence (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (X O : RegionName) (n pid : Nat) (memory : ByteMemory.Memory) :
    runProgram b d pointers undef X O n pid memory=
      some (applyWrites (programWrites b d (pointers X) (pointers O) n pid memory) memory) := by
  simp only [runProgram,original_ast_lowering,Option.bind_some,bind,pure,Option.bind,sourceCode,
    executeBody,step,evalIndex,evalBool]
  try simp only [if_pos rfl,if_true,String.reduceEq,if_false]
  by_cases hp : stores32 pid=Bool.true
  · have hp' : (BitVec.ofNat 32 pid == BitVec.ofNat 32 0)=Bool.true := hp
    simp only [hp',if_true]
    congr 1
    unfold programWrites
    rw [if_pos hp]
    unfold lanes
    rw [←fold_stores]
    apply congrArg (fun f => List.foldl f memory (List.range 1024))
    funext state lane
    change (if active n pid lane then _ else _)=_
    by_cases ha : active n pid lane=Bool.true
    all_goals simp [laneAddress,offset32,start32,blockSize,active] at ha ⊢
    all_goals simp only [ha,Bool.false_eq_true,if_true,if_false]
    all_goals try rw [show ∀ v : Word d, castStore b d (selectSource b d v)=scalarFormula b d v from scalar_correspondence b d]
  · have hp' : (BitVec.ofNat 32 pid == BitVec.ofNat 32 0)≠Bool.true := hp
    simp [hp',programWrites,hp,applyWrites]

#specNonCircular programWrites avoiding [runProgram,executeBody,step,applyWrites,scalarSource]
#axiomsClean original_ast_lowering
#axiomsClean fold_stores
#axiomsClean source_program_correspondence
#guard_msgs (drop info) in
#auditModuleAxioms
end
