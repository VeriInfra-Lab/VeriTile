/-- These are addresses in the current launch device's address space. The
ABI pointer may be translated by extractPointer and need not equal data_ptr. -/
structure PhysicalView where
  pointer : BitVec 64
  base : Nat
  capacity : Nat

structure AccessContext where
  currentDevice : Nat
  readableByte : Nat → Nat → Prop
  writableByte : Nat → Nat → Prop

def PhysicalView.Inside (v : PhysicalView) (width address : Nat) : Prop :=
  v.base≤address ∧ address+width≤v.base+v.capacity ∧ address+width≤2^64

def PhysicalView.Disjoint (a b : PhysicalView) : Prop :=
  a.base+a.capacity≤b.base ∨ b.base+b.capacity≤a.base

def ReadOK (ctx : AccessContext) (v : PhysicalView) (width address : Nat) : Prop :=
  v.Inside width address ∧ address%width=0 ∧
    ∀ k, k<width → ctx.readableByte ctx.currentDevice (address+k)

def WriteOK (ctx : AccessContext) (v : PhysicalView) (address : Nat) : Prop :=
  v.Inside 4 address ∧ address%4=0 ∧
    ∀ k, k<4 → ctx.writableByte ctx.currentDevice (address+k)

/-- This is an applicability observation, not a guard executed by ReLU,
not a native trap model, and not a promise of rollback on invalid access. -/
def ProgramSafe (ctx : AccessContext) (d : Storage) (input output : PhysicalView)
    (n pid : Nat) : Prop :=
  (∀ lane∈lanes n pid, ReadOK ctx input d.bytes (laneAddress input.pointer d.bytes pid lane)) ∧
  (FiniteAddress.stores32 pid=Bool.true →
    ∀ lane∈lanes n pid, WriteOK ctx output (laneAddress output.pointer 4 pid lane))

def Preserved (output : PhysicalView) (current initial : ByteMemory.Memory) : Prop :=
  ∀ address, address<output.base ∨ output.base+output.capacity≤address → current address=initial address

theorem preserved_refl (output : PhysicalView) (s : ByteMemory.Memory) : Preserved output s s := fun _ _ => rfl

theorem preserved_trans (output : PhysicalView) (a b c : ByteMemory.Memory)
    (h : Preserved output a b) (k : Preserved output b c) : Preserved output a c :=
  fun address outside => (h address outside).trans (k address outside)

theorem write_preserved (output : PhysicalView) (s : ByteMemory.Memory) (w : Write)
    (inside : output.Inside 4 w.address) :
    Preserved output (ByteMemory.storeLE 4 s w.address w.value) s := by
  intro address outside
  apply ByteMemory.store_frame
  unfold PhysicalView.Inside at inside
  rcases outside with h|h <;> omega

theorem writes_preserved (output : PhysicalView) (writes : List Write) (s : ByteMemory.Memory)
    (inside : ∀ w∈writes,output.Inside 4 w.address) :
    Preserved output (applyWrites writes s) s := by
  induction writes generalizing s with
  | nil => exact preserved_refl output s
  | cons w rest ih =>
    exact preserved_trans output _ _ _ (ih _ (fun w hw => inside w (by simp [hw])))
      (write_preserved output s w (inside w (by simp)))

theorem input_preserved (d : Storage) (input output : PhysicalView) (current initial : ByteMemory.Memory)
    (same : Preserved output current initial) (sep : input.Disjoint output)
    (address : Nat) (inside : input.Inside d.bytes address) :
    readWord d current address=readWord d initial address := by
  apply ByteMemory.load_congr
  intro byte hb
  apply same
  unfold PhysicalView.Disjoint at sep
  unfold PhysicalView.Inside at inside
  rcases sep with h|h <;> omega

theorem program_inside (b : Numerics) (d : Storage) (input output : PhysicalView)
    (ctx : AccessContext) (n pid : Nat) (memory : ByteMemory.Memory)
    (safe : ProgramSafe ctx d input output n pid) :
    ∀ w∈programWrites b d input.pointer output.pointer n pid memory,output.Inside 4 w.address := by
  intro w hw
  unfold programWrites at hw
  split at hw
  next h =>
    obtain ⟨lane,hl,rfl⟩ := List.mem_map.mp hw
    exact (safe.2 h lane hl).1
  next h => simp at hw

theorem program_reads_initial (b : Numerics) (d : Storage) (input output : PhysicalView)
    (ctx : AccessContext) (n pid : Nat) (current initial : ByteMemory.Memory)
    (same : Preserved output current initial) (sep : input.Disjoint output)
    (safe : ProgramSafe ctx d input output n pid) :
    programWrites b d input.pointer output.pointer n pid current=
      programWrites b d input.pointer output.pointer n pid initial := by
  unfold programWrites
  split
  · apply List.map_congr_left
    intro lane hl
    rw [input_preserved d input output current initial same sep _ (safe.1 lane hl).1]
  · rfl

/-- Independent full-byte specification, including partially overlapping
stores and every untouched allocator byte. -/
def byteFormula (writes : List Write) (initial : ByteMemory.Memory) (address : Nat) : ByteMemory.Byte :=
  match writes.reverse.find? (fun w => decide (w.address≤address ∧ address<w.address+4)) with
  | none => initial address
  | some w => ByteMemory.splitByte w.value (address-w.address)

theorem writes_append (left right : List Write) (s : ByteMemory.Memory) :
    applyWrites (left++right) s=applyWrites right (applyWrites left s) := by
  induction left generalizing s with
  | nil => rfl
  | cons w rest ih => exact ih _

theorem writes_pointwise (writes : List Write) (initial : ByteMemory.Memory) (address : Nat) :
    applyWrites writes initial address=byteFormula writes initial address := by
  induction writes using List.reverseRecOn with
  | nil => rfl
  | append_singleton rest w ih =>
    rw [writes_append]
    simp only [applyWrites]
    unfold byteFormula
    simp only [List.reverse_append,List.reverse_cons,List.reverse_nil,List.nil_append,List.cons_append,List.find?_cons]
    by_cases h : w.address≤address ∧ address<w.address+4
    · simp [ByteMemory.storeLE,h]
    · simp [ByteMemory.storeLE,h,ih,byteFormula]

theorem writes_byte_frame (writes : List Write) (initial : ByteMemory.Memory) (address : Nat)
    (outside : ∀ w∈writes,address<w.address ∨ w.address+4≤address) :
    applyWrites writes initial address=initial address := by
  induction writes generalizing initial with
  | nil => rfl
  | cons w rest ih =>
    rw [applyWrites,ih _ (fun v hv => outside v (by simp [hv]))]
    exact ByteMemory.store_frame 4 initial w.address w.value address (outside w (by simp))
