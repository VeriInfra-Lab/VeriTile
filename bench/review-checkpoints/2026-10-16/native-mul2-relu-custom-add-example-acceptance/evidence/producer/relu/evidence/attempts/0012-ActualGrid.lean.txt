def GridSafe (ctx : AccessContext) (d : Storage) (input output : PhysicalView)
    (n : Nat) (jobs : List Nat) : Prop :=
  ∀ pid∈jobs,ProgramSafe ctx d input output n pid

/-- Sequential schedule of the actual lowered source body, including each
program's pre-gate load. Output freshness makes the proven result schedule-independent. -/
def runGrid (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (X O : RegionName) (n : Nat) : List Nat → ByteMemory.Memory → Option ByteMemory.Memory
  | [],memory => some memory
  | pid::rest,memory => do
      let next ← runProgram b d pointers undef X O n pid memory
      runGrid b d pointers undef X O n rest next

/-- The reference samples all inputs from the allocated initial snapshot. -/
def gridWrites (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory) : List Write :=
  jobs.flatMap (fun pid => programWrites b d x output n pid initial)

def gridFormula (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory) : ByteMemory.Memory :=
  byteFormula (gridWrites b d x output n jobs initial) initial

theorem grid_inside (b : Numerics) (d : Storage) (input output : PhysicalView)
    (ctx : AccessContext) (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory)
    (safe : GridSafe ctx d input output n jobs) :
    ∀ w∈gridWrites b d input.pointer output.pointer n jobs initial,output.Inside 4 w.address := by
  intro w hw
  obtain ⟨pid,hp,hw⟩ := List.mem_flatMap.mp hw
  exact program_inside b d input output ctx n pid initial (safe pid hp) w hw

theorem source_grid_correspondence (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (X O : RegionName) (input output : PhysicalView) (ctx : AccessContext)
    (n : Nat) (jobs : List Nat) (current initial : ByteMemory.Memory)
    (hx : pointers X=input.pointer) (ho : pointers O=output.pointer)
    (same : Preserved output current initial) (sep : input.Disjoint output)
    (safe : GridSafe ctx d input output n jobs) :
    runGrid b d pointers undef X O n jobs current=
      some (applyWrites (gridWrites b d input.pointer output.pointer n jobs initial) current) := by
  induction jobs generalizing current with
  | nil => rfl
  | cons pid rest ih =>
    have hp : ProgramSafe ctx d input output n pid := safe pid (by simp)
    have hr : GridSafe ctx d input output n rest := fun p hm => safe p (by simp [hm])
    rw [runGrid,source_program_correspondence,hx,ho,
      program_reads_initial b d input output ctx n pid current initial same sep hp]
    simp only [Option.bind_some]
    rw [ih _ (preserved_trans output _ _ _
      (writes_preserved output _ current (program_inside b d input output ctx n pid initial hp)) same) hr]
    simp only [gridWrites,List.flatMap_cons,writes_append]

theorem source_grid_formula (b : Numerics) (d : Storage) (pointers : RegionName → BitVec 64)
    (undef : Word d) (X O : RegionName) (input output : PhysicalView) (ctx : AccessContext)
    (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory)
    (hx : pointers X=input.pointer) (ho : pointers O=output.pointer)
    (sep : input.Disjoint output) (safe : GridSafe ctx d input output n jobs) :
    runGrid b d pointers undef X O n jobs initial=
      some (gridFormula b d input.pointer output.pointer n jobs initial) := by
  rw [source_grid_correspondence b d pointers undef X O input output ctx n jobs initial initial hx ho
    (preserved_refl output initial) sep safe]
  congr 1
  funext address
  exact writes_pointwise _ _ _

theorem grid_frame (b : Numerics) (d : Storage) (input output : PhysicalView) (ctx : AccessContext)
    (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory)
    (safe : GridSafe ctx d input output n jobs) :
    Preserved output (gridFormula b d input.pointer output.pointer n jobs initial) initial := by
  have eq : gridFormula b d input.pointer output.pointer n jobs initial=
      applyWrites (gridWrites b d input.pointer output.pointer n jobs initial) initial := by
    funext address
    exact (writes_pointwise _ _ _).symm
  rw [eq]
  exact writes_preserved output _ initial (grid_inside b d input output ctx n jobs initial safe)

theorem stores32_exact (pid : Nat) (hp : pid<2^32) : stores32 pid=decide (pid=0) := by
  unfold stores32
  apply Bool.eq_iff_iff.mpr
  simp only [beq_iff_eq,←BitVec.toNat_inj,BitVec.toNat_ofNat,Nat.mod_eq_of_lt hp,decide_eq_true_eq]
  norm_num

theorem nonzero_program_writes (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n pid : Nat) (initial : ByteMemory.Memory) (hp : pid<2^32) (hn : pid≠0) :
    programWrites b d x output n pid initial=[] := by
  simp [programWrites,stores32_exact pid hp,hn]

theorem nonzero_jobs_empty (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n : Nat) (jobs : List Nat) (initial : ByteMemory.Memory)
    (valid : ∀ pid∈jobs,pid<2^32 ∧ pid≠0) : gridWrites b d x output n jobs initial=[] := by
  induction jobs with
  | nil => rfl
  | cons pid rest ih =>
    have hp := valid pid (by simp)
    simp only [gridWrites,List.flatMap_cons,nonzero_program_writes b d x output n pid initial hp.1 hp.2,List.nil_append]
    exact ih (fun p hm => valid p (by simp [hm]))

/-- The native CUDA grid-x limit implies this bound; it is independent of
constexpr N and never replaces any nonzero program's read obligation. -/
theorem launched_stores_only_first (b : Numerics) (d : Storage) (x output : BitVec 64)
    (n count : Nat) (initial : ByteMemory.Memory) (bound : count≤2^32) :
    gridWrites b d x output n (List.range count) initial=
      if count=0 then [] else programWrites b d x output n 0 initial := by
  cases count with
  | zero => rfl
  | succ count =>
    rw [List.range_succ_eq_map]
    simp only [gridWrites,List.flatMap_cons,show count+1≠0 by omega,if_false]
    have empty := nonzero_jobs_empty b d x output n ((List.range count).map Nat.succ) initial (by
      intro pid hp
      obtain ⟨p,hp,rfl⟩ := List.mem_map.mp hp
      have := List.mem_range.mp hp
      constructor <;> omega)
    change _ ++ gridWrites b d x output n ((List.range count).map Nat.succ) initial=_
    rw [empty,List.append_nil]

theorem empty_grid (b : Numerics) (d : Storage) (x output : BitVec 64) (initial : ByteMemory.Memory) :
    gridFormula b d x output 0 (List.range (grid 0)) initial=initial := rfl
