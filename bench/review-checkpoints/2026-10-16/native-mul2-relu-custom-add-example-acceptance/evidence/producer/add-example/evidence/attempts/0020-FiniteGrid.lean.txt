@[simp] theorem bind_ok (value : α) (f : α → Except ε β) :
    (Except.ok value >>= f)=f value := rfl
@[simp] theorem bind_error (issue : ε) (f : α → Except ε β) :
    (Except.error issue >>= f)=Except.error issue := rfl
@[simp] theorem pure_ok (value : α) : (pure value : Except ε α)=.ok value := rfl

theorem program_plan_frame {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n pid : Nat) (left right : M) (seen : List Nat)
    (same : Preserved ops output left right) (freshX : x.Disjoint output) (freshY : y.Disjoint output) :
    programPlan b r ops x y output n pid left seen=programPlan b r ops x y output n pid right seen := by
  by_cases hx : ∀ lane∈programLanes n pid,ReadOK ops x (x.nativeLane pid lane)
  · by_cases hy : ∀ lane∈programLanes n pid,ReadOK ops y (y.nativeLane pid lane)
    · have ex : ∀ lane∈programLanes n pid,
          ops.read left x.dtype (x.lane pid lane)=ops.read right x.dtype (x.lane pid lane) :=
        fun lane hl => input_preserved ops x output left right same freshX _ (hx lane hl).1
      have ey : ∀ lane∈programLanes n pid,
          ops.read left y.dtype (y.lane pid lane)=ops.read right y.dtype (y.lane pid lane) :=
        fun lane hl => input_preserved ops y output left right same freshY _ (hy lane hl).1
      have es : ScalarChecks b r ops x y n pid left=ScalarChecks b r ops x y n pid right := by
        unfold ScalarChecks
        apply propext
        constructor <;> intro h lane hl
        · simpa only [ex lane hl,ey lane hl] using h lane hl
        · simpa only [ex lane hl,ey lane hl] using h lane hl
      have ew : expectedWrites b r ops x y output n pid left=expectedWrites b r ops x y output n pid right := by
        apply List.map_congr_left
        intro lane hl
        simp only [ex lane hl,ey lane hl]
      simp only [programPlan,inspectProgram,if_pos hx,if_pos hy,es,ew]
    · simp [programPlan,inspectProgram,hx,hy]
  · simp [programPlan,inspectProgram,hx]

def gridPlan {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n : Nat) (snapshot : M) :
    List Nat → List Nat → Except Observation (List (Write (Carrier r)) × List Nat)
  | [],seen => .ok ([],seen)
  | pid::rest,seen => do
      let (head,owners) ← programPlan b r ops x y output n pid snapshot seen
      let (tail,finalOwners) ← gridPlan b r ops x y output n snapshot rest owners
      pure (head++tail,finalOwners)

def observedGrid {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName) (n : Nat) :
    List Nat → M × List Nat → Except Observation (M × List Nat)
  | [],state => .ok state
  | pid::rest,state => do
      let next ← observedProgram b r ops views undef X Y O n pid state.1 state.2
      observedGrid b r ops views undef X Y O n rest next

theorem grid_correspondence {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName)
    (n : Nat) (snapshot : M) (pids : List Nat) (state : M × List Nat)
    (same : Preserved ops (views O) state.1 snapshot)
    (freshX : (views X).Disjoint (views O)) (freshY : (views Y).Disjoint (views O)) :
    observedGrid b r ops views undef X Y O n pids state=
      (gridPlan b r ops (views X) (views Y) (views O) n snapshot pids state.2).map
        (fun plan => (applyWrites ops (views O).dtype plan.1 state.1,plan.2)) := by
  induction pids generalizing state with
  | nil => rfl
  | cons pid rest ih =>
    simp only [observedGrid,gridPlan,program_correspondence]
    rw [program_plan_frame b r ops (views X) (views Y) (views O) n pid state.1 snapshot state.2 same freshX freshY]
    cases hh : programPlan b r ops (views X) (views Y) (views O) n pid snapshot state.2 with
    | error issue => rfl
    | ok head =>
      rcases head with ⟨head,owners⟩
      simp only [Except.map,bind_ok]
      have hb := program_plan_bounds b r ops (views X) (views Y) (views O) n pid snapshot state.2 owners head hh
      have hs : Preserved ops (views O) (applyWrites ops (views O).dtype head state.1) snapshot :=
        preserved_trans ops (views O) _ _ _ (writes_preserved ops (views O) head state.1 hb) same
      rw [ih (applyWrites ops (views O).dtype head state.1,owners) hs]
      cases gridPlan b r ops (views X) (views Y) (views O) n snapshot rest owners with
      | error issue => rfl
      | ok tail =>
        rcases tail with ⟨tail,finalOwners⟩
        simp only [Except.map,bind_ok,pure_ok,writes_append]

theorem grid_plan_bounds {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n : Nat) (snapshot : M) (pids : List Nat) (seen owners : List Nat)
    (writes : List (Write (Carrier r)))
    (h : gridPlan b r ops x y output n snapshot pids seen=.ok (writes,owners)) :
    ∀ w∈writes,output.Inside w.address := by
  induction pids generalizing seen writes owners with
  | nil => cases h; simp
  | cons pid rest ih =>
    simp only [gridPlan] at h
    cases hh : programPlan b r ops x y output n pid snapshot seen with
    | error issue => simp [hh] at h
    | ok head =>
      rcases head with ⟨head,nextOwners⟩
      cases ht : gridPlan b r ops x y output n snapshot rest nextOwners with
      | error issue => simp [hh,ht] at h
      | ok tail =>
        rcases tail with ⟨tail,finalOwners⟩
        simp [hh,ht] at h
        rcases h with ⟨rfl,rfl⟩
        intro w hw
        rcases List.mem_append.1 hw with hw|hw
        · exact program_plan_bounds b r ops x y output n pid snapshot seen nextOwners head hh w hw
        · exact ih _ _ _ ht w hw

def launchPlan {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n : Nat) (snapshot : M) : Except Observation (List (Write (Carrier r))) :=
  (gridPlan b r ops x y output n snapshot (List.range (gridCount n)) []).map Prod.fst

def observedLaunch {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName) (n : Nat) (s : M) :
    Except Observation M :=
  (observedGrid b r ops views undef X Y O n (List.range (gridCount n)) (s,[])).map Prod.fst

theorem launch_correspondence {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName) (n : Nat) (s : M)
    (freshX : (views X).Disjoint (views O)) (freshY : (views Y).Disjoint (views O)) :
    observedLaunch b r ops views undef X Y O n s=
      (launchPlan b r ops (views X) (views Y) (views O) n s).map
        (fun writes => applyWrites ops (views O).dtype writes s) := by
  unfold observedLaunch launchPlan
  rw [grid_correspondence b r ops views undef X Y O n s _ _ (preserved_refl ops (views O) s) freshX freshY]
  cases gridPlan b r ops (views X) (views Y) (views O) n s (List.range (gridCount n)) [] <;> rfl

theorem launch_plan_bounds {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n : Nat) (s : M) (writes : List (Write (Carrier r)))
    (h : launchPlan b r ops x y output n s=.ok writes) : ∀ w∈writes,output.Inside w.address := by
  unfold launchPlan at h
  cases hg : gridPlan b r ops x y output n s (List.range (gridCount n)) [] with
  | error issue => simp [hg,Except.map] at h
  | ok plan =>
    simp only [hg,Except.map,Except.ok.injEq] at h
    subst writes
    exact grid_plan_bounds b r ops x y output n s _ _ _ _ hg

/-- All requests remain in the outcome statement. An applicability issue
claims no native error, return, rollback or race winner. Successful execution
has a complete independent byte formula, initial-snapshot values and frame. -/
theorem launch_all_outcomes {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName) (n : Nat) (s : M)
    (freshX : (views X).Disjoint (views O)) (freshY : (views Y).Disjoint (views O)) :
    (∃ issue, launchPlan b r ops (views X) (views Y) (views O) n s=.error issue ∧
      observedLaunch b r ops views undef X Y O n s=.error issue) ∨
    (∃ writes final, launchPlan b r ops (views X) (views Y) (views O) n s=.ok writes ∧
      observedLaunch b r ops views undef X Y O n s=.ok final ∧
      Preserved ops (views O) final s ∧
      ∀ address,ops.observeByte final address=byteFormula ops (views O).dtype writes s address) := by
  have hc := launch_correspondence b r ops views undef X Y O n s freshX freshY
  cases hp : launchPlan b r ops (views X) (views Y) (views O) n s with
  | error issue => exact Or.inl ⟨issue,rfl,by simpa [hp,Except.map] using hc⟩
  | ok writes => exact Or.inr ⟨writes,applyWrites ops (views O).dtype writes s,rfl,
      by simpa [hp,Except.map] using hc,
      writes_preserved ops (views O) writes s (launch_plan_bounds b r ops (views X) (views Y) (views O) n s writes hp),
      writes_pointwise ops (views O).dtype writes s⟩

theorem all_masked_grid_plan {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : PhysicalView) (n : Nat) (s : M) (pids seen : List Nat)
    (hn : extent32 n=BitVec.ofNat 32 (2^31)) :
    gridPlan b r ops x y output n s pids seen=.ok ([],seen) := by
  induction pids generalizing seen with
  | nil => rfl
  | cons pid rest ih => simp only [gridPlan,all_masked_program_plan b r ops x y output n pid s seen hn,
      bind_ok,ih,pure_ok,List.nil_append]

/-- At the signed-i32 minimum every mask is false. No memory, numeric,
ownership or freshness premise is needed to leave the allocated memory intact. -/
theorem all_masked_launch {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (views : RegionName → PhysicalView) (undef : Carrier r) (X Y O : RegionName) (n : Nat) (s : M)
    (hn : extent32 n=BitVec.ofNat 32 (2^31)) : observedLaunch b r ops views undef X Y O n s=.ok s := by
  have programs : ∀ pid state, observedProgram b r ops views undef X Y O n pid state.1 state.2=.ok state := by
    intro pid state
    rw [program_correspondence,all_masked_program_plan b r ops (views X) (views Y) (views O) n pid state.1 state.2 hn]
    rfl
  have grids : ∀ pids state,observedGrid b r ops views undef X Y O n pids state=.ok state := by
    intro pids
    induction pids with
    | nil => intro state; rfl
    | cons pid rest ih => intro state; simp only [observedGrid,programs,bind_ok,ih]
  rw [observedLaunch,grids]
  rfl
