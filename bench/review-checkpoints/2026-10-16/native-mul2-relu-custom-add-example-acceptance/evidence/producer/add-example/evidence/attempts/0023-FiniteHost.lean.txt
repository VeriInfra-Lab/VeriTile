/-- Actual Torch metadata, including scalar/empty tensors, raw data_ptr,
preserve-format strides, storage offset, backing span and owner device. -/
structure Tensor where
  region : RegionName
  dtype : DenseDType
  device : String
  shape : List Nat
  strides : List Int
  storageOffset : Nat
  dataPtr : BitVec 64
  storageBase : Nat
  storageBytes : Nat
  deriving DecidableEq

def Tensor.numel (t : Tensor) : Nat := t.shape.prod

structure Metadata where
  view : Tensor → PhysicalView
  extractedPointer : Tensor → BitVec 64
  currentDevice : Nat

/-- Successful pointer extraction may translate raw data_ptr. Translation is
required only for nonnull tensors; no owner=current-device equality is added.
All later access permissions are checked in the actual current launch device. -/
structure TensorBinding (metadata : Metadata) (t : Tensor) (d : DType) : Prop where
  dtype : (metadata.view t).dtype=d
  backing : (metadata.view t).base=t.storageBase
  capacity : (metadata.view t).capacity=t.storageBytes
  actualPointer : (metadata.view t).pointer=metadata.extractedPointer t
  translated : t.dataPtr≠0 → ∀ byte,byte<t.storageBytes →
    (metadata.view t).toBacking ((metadata.view t).deviceBase+byte)=t.storageBase+byte
  pointerOffset : t.numel>0 → (metadata.view t).pointer.toNat=
    (metadata.view t).deviceBase+t.storageOffset*bytes d
  null : t.dataPtr=0 → (metadata.view t).pointer=0

/-- Fresh zero allocation. Neither the output strides nor the raw pointers
are reconstructed. Freshness applies to both live inputs regardless of aliasing
between x and y, their devices, shapes, lengths, or layouts. -/
structure AllocationBinding {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (metadata : Metadata) (x y output : Tensor) (before allocated : M) : Prop where
  dtype : output.dtype=x.dtype
  shape : output.shape=x.shape
  device : output.device=x.device
  offset : output.storageOffset=0
  freshX : (metadata.view x).Disjoint (metadata.view output)
  freshY : (metadata.view y).Disjoint (metadata.view output)
  bindings : ∀ r,admission x.dtype y.dtype=.ok r →
    TensorBinding metadata x r.signature.1 ∧ TensorBinding metadata y r.signature.2 ∧
      TensorBinding metadata output r.signature.1 ∧ (ops r).launchDevice=metadata.currentDevice
  frame : ∀ r,Preserved (ops r) (metadata.view output) allocated before
  zero : ∀ r address,output.storageBase≤address → address<output.storageBase+output.storageBytes →
    (ops r).observeByte allocated address=0

inductive HostEvent where
  | zerosLike (input : Tensor)
  | blockSize (value : Nat)
  | numel (input : Tensor) (value : Nat)
  | computedGrid (n grid : Nat)
  | launch (x y output : Tensor) (n grid block : Nat)
  | returned (output : Tensor)
  deriving DecidableEq
inductive NativeStage where
  | currentDeviceAndStream | compilation | launcherContext | pointerExtraction | enqueue
  deriving DecidableEq
abbrev NativeEnvironment := NativeStage → Option String
abbrev Allocator (M : Type) := Tensor → M → Except String (Tensor × M)
inductive HostFailure where
  | allocation (detail : String)
  | primitive (stage : NativeStage) (detail : String)
  | dtypeSetup (reason : SetupReject)
  | gridParse (requested : Nat)
  deriving DecidableEq
inductive HostOutcome (M : Type) where
  | nativeFailure (trace : List HostEvent) (reason : HostFailure)
  | outsideModel (trace : List HostEvent) (issue : Observation)
  | returned (trace : List HostEvent) (output : Tensor) (memory : M)

def setupTrace (x y output : Tensor) : List HostEvent :=
  [.zerosLike x,.blockSize 4,.numel x x.numel,.computedGrid x.numel (gridCount x.numel),
   .launch x y output x.numel (gridCount x.numel) 4]
def successTrace (x y output : Tensor) : List HostEvent := setupTrace x y output++[.returned output]
def sourceViews (metadata : Metadata) (x y output : Tensor) : RegionName → PhysicalView :=
  fun name => if name="x" then metadata.view x else if name="y" then metadata.view y else metadata.view output

def unused (r : Route) : Carrier r := match r with
  | .wide _ _ => none
  | _ => .undef

def gridSupported (n : Nat) : Prop := gridCount n≤2^31-1

/-- Source order after allocation: numel/grid, JIT device/stream, dtype binding
and compilation, native context, signed-int grid parsing, pointer extraction,
then enqueue. No CUDA, rank, shape, stride, length or same-device guard. -/
def sourceAllocated {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (env : NativeEnvironment) (metadata : Metadata) (x y output : Tensor) (allocated : M) : HostOutcome M :=
  match env .currentDeviceAndStream with
  | some detail => .nativeFailure (setupTrace x y output) (.primitive .currentDeviceAndStream detail)
  | none => match admission x.dtype y.dtype with
    | .error reason => .nativeFailure (setupTrace x y output) (.dtypeSetup reason)
    | .ok r => match env .compilation with
      | some detail => .nativeFailure (setupTrace x y output) (.primitive .compilation detail)
      | none => match env .launcherContext with
        | some detail => .nativeFailure (setupTrace x y output) (.primitive .launcherContext detail)
        | none => if ¬gridSupported x.numel then .nativeFailure (setupTrace x y output) (.gridParse (gridCount x.numel))
          else match env .pointerExtraction with
            | some detail => .nativeFailure (setupTrace x y output) (.primitive .pointerExtraction detail)
            | none => match env .enqueue with
              | some detail => .nativeFailure (setupTrace x y output) (.primitive .enqueue detail)
              | none => match observedLaunch b r (ops r) (sourceViews metadata x y output) (unused r) "x" "y" "out" x.numel allocated with
                | .error issue => .outsideModel (setupTrace x y output) issue
                | .ok final => .returned (successTrace x y output) output final

def sourceHost {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : NativeEnvironment) (metadata : Metadata) (x y : Tensor) (before : M) : HostOutcome M :=
  match alloc x before with
  | .error detail => .nativeFailure [.zerosLike x] (.allocation detail)
  | .ok (output,allocated) => sourceAllocated ops b env metadata x y output allocated

/-- The formula has no interpreter or source execution dependency. -/
inductive ByteOutcome where
  | nativeFailure (trace : List HostEvent) (reason : HostFailure)
  | outsideModel (trace : List HostEvent) (issue : Observation)
  | returned (trace : List HostEvent) (output : Tensor) (bytes : Nat → Fin 256)

def byteObservation {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (x y : Tensor) : HostOutcome M → ByteOutcome
  | .nativeFailure trace reason => .nativeFailure trace reason
  | .outsideModel trace issue => .outsideModel trace issue
  | .returned trace output memory =>
    match admission x.dtype y.dtype with
    | .ok r => .returned trace output ((ops r).observeByte memory)
    | .error reason => .nativeFailure trace (.dtypeSetup reason)

def allocatedFormula {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (env : NativeEnvironment) (metadata : Metadata) (x y output : Tensor) (allocated : M) : ByteOutcome :=
  match env .currentDeviceAndStream with
  | some detail => .nativeFailure (setupTrace x y output) (.primitive .currentDeviceAndStream detail)
  | none => match admission x.dtype y.dtype with
    | .error reason => .nativeFailure (setupTrace x y output) (.dtypeSetup reason)
    | .ok r => match env .compilation with
      | some detail => .nativeFailure (setupTrace x y output) (.primitive .compilation detail)
      | none => match env .launcherContext with
        | some detail => .nativeFailure (setupTrace x y output) (.primitive .launcherContext detail)
        | none => if ¬gridSupported x.numel then .nativeFailure (setupTrace x y output) (.gridParse (gridCount x.numel))
          else match env .pointerExtraction with
            | some detail => .nativeFailure (setupTrace x y output) (.primitive .pointerExtraction detail)
            | none => match env .enqueue with
              | some detail => .nativeFailure (setupTrace x y output) (.primitive .enqueue detail)
              | none => match launchPlan b r (ops r) (metadata.view x) (metadata.view y) (metadata.view output) x.numel allocated with
                | .error issue => .outsideModel (setupTrace x y output) issue
                | .ok writes => .returned (successTrace x y output) output
                    (byteFormula (ops r) (metadata.view output).dtype writes allocated)

def hostFormula {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : NativeEnvironment) (metadata : Metadata) (x y : Tensor) (before : M) : ByteOutcome :=
  match alloc x before with
  | .error detail => .nativeFailure [.zerosLike x] (.allocation detail)
  | .ok (output,allocated) => allocatedFormula ops b env metadata x y output allocated

theorem allocated_correspondence {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (env : NativeEnvironment) (metadata : Metadata) (x y output : Tensor) (before allocated : M)
    (binding : AllocationBinding ops metadata x y output before allocated) :
    byteObservation ops x y (sourceAllocated ops b env metadata x y output allocated)=
      allocatedFormula ops b env metadata x y output allocated := by
  unfold sourceAllocated allocatedFormula
  cases he : env .currentDeviceAndStream with
  | some detail => rfl
  | none =>
    cases ha : admission x.dtype y.dtype with
    | error reason => rfl
    | ok r =>
      cases hc : env .compilation with
      | some detail => rfl
      | none =>
        cases hl : env .launcherContext with
        | some detail => rfl
        | none =>
          by_cases hg : gridSupported x.numel
          · simp only [hg,not_true_eq_false,↓reduceIte]
            cases hp : env .pointerExtraction with
            | some detail => rfl
            | none =>
              cases hq : env .enqueue with
              | some detail => rfl
              | none =>
                have corr := launch_correspondence b r (ops r) (sourceViews metadata x y output) (unused r) "x" "y" "out" x.numel allocated
                  (by simpa [sourceViews] using binding.freshX) (by simpa [sourceViews] using binding.freshY)
                simp only [sourceViews,↓reduceIte] at corr
                rw [corr]
                cases ht : launchPlan b r (ops r) (metadata.view x) (metadata.view y) (metadata.view output) x.numel allocated with
                | error issue => rfl
                | ok writes =>
                  simp only [Except.map,byteObservation,ha]
                  congr 1
                  funext address
                  exact writes_pointwise (ops r) (metadata.view output).dtype writes allocated address
          · simp only [hg,not_false_eq_true,↓reduceIte]; rfl

theorem host_correspondence {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : NativeEnvironment) (metadata : Metadata) (x y : Tensor) (before : M)
    (contract : ∀ output allocated,alloc x before=.ok (output,allocated) →
      AllocationBinding ops metadata x y output before allocated) :
    byteObservation ops x y (sourceHost ops b alloc env metadata x y before)=hostFormula ops b alloc env metadata x y before := by
  cases ha : alloc x before with
  | error detail => rfl
  | ok result =>
    rcases result with ⟨output,allocated⟩
    simp only [sourceHost,hostFormula,ha]
    exact allocated_correspondence ops b env metadata x y output before allocated (contract output allocated ha)

/-- Allocation really is first, even for unsupported dtypes and invalid native
launch setup; failure preserves its actual observed allocation detail. -/
theorem allocation_first {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : NativeEnvironment) (metadata : Metadata) (x y : Tensor) (before : M)
    (detail : String) (h : alloc x before=.error detail) :
    sourceHost ops b alloc env metadata x y before=.nativeFailure [.zerosLike x] (.allocation detail) := by
  simp [sourceHost,h]

theorem grid_parse_before_pointers {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (env : NativeEnvironment) (metadata : Metadata) (x y output : Tensor) (allocated : M) (r : Route)
    (he : env .currentDeviceAndStream=none) (ha : admission x.dtype y.dtype=.ok r)
    (hc : env .compilation=none) (hl : env .launcherContext=none) (hg : ¬gridSupported x.numel) :
    sourceAllocated ops b env metadata x y output allocated=
      .nativeFailure (setupTrace x y output) (.gridParse (gridCount x.numel)) := by
  simp [sourceAllocated,he,ha,hc,hl,hg]

theorem n2p31_grid_supported : gridSupported (2^31) := by decide +kernel

theorem n2p31_returns_allocation {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (metadata : Metadata) (x y output : Tensor) (before allocated : M) (r : Route)
    (ha : alloc x before=.ok (output,allocated)) (hr : admission x.dtype y.dtype=.ok r) (hn : x.numel=2^31) :
    sourceHost ops b alloc (fun _ => none) metadata x y before=.returned (successTrace x y output) output allocated := by
  simp only [sourceHost,ha,sourceAllocated,hr,hn,n2p31_grid_supported,not_true_eq_false,↓reduceIte,
    all_masked_launch b r (ops r) (sourceViews metadata x y output) (unused r) "x" "y" "out" (2^31) allocated rfl]

theorem rank_zero_numel (x : Tensor) (h : x.shape=[]) : x.numel=1 := by simp [Tensor.numel,h]
theorem empty_grid (x : Tensor) (h : 0∈x.shape) : gridCount x.numel=0 := by
  have hz : x.shape.prod=0 := List.prod_eq_zero h
  simp [Tensor.numel,hz,gridCount]

end
