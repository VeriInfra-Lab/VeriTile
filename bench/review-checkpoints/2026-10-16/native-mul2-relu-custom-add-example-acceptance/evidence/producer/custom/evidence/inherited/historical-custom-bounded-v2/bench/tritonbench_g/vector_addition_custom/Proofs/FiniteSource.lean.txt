/-- Finite lowering of the ordinary index-expression subset used by the
unchanged source. Unsupported AST constructors fail rather than disappearing. -/
inductive IndexExpr where
  | literal (n : Nat)
  | program (ax : Nat)
  | register (name : RegName)
  | lane (width : Nat)
  | add (a b : IndexExpr)
  | mul (a b : IndexExpr)
  deriving DecidableEq

def lowerIndex {shape : TileShape} : Op .nat shape → Option IndexExpr
  | .constNat n => some (.literal n)
  | .programId ax => some (.program ax)
  | .ref _ _ name => some (.register name)
  | .arange width => some (.lane width)
  | .add _ _ a b => return .add (← lowerIndex a) (← lowerIndex b)
  | .mul _ _ a b => return .mul (← lowerIndex a) (← lowerIndex b)
  | _ => none

inductive MaskExpr where
  | less (a b : IndexExpr)
  deriving DecidableEq

def lowerMask {shape : TileShape} : MaskOpt .real shape → Option MaskExpr
  | .mask (.lt .nat _ a b) => return .less (← lowerIndex a) (← lowerIndex b)
  | _ => none

/-- The value lowering retains both raw loaded operands. `storeAdd` executes
exactly the selected finite load-normalization/promotion/add/store-cast source
path; this is concrete `scalarSource`, not a supplied output-value callback. -/
inductive Instruction where
  | indexAssign (name : RegName) (value : IndexExpr)
  | load (name : RegName) (region : RegionName) (offset : IndexExpr) (mask : MaskExpr)
  | storeAdd (region : RegionName) (offset : IndexExpr) (a b : RegName) (mask : MaskExpr)
  deriving DecidableEq

def lowerStatement : Stmt → Option Instruction
  | .assign .nat _ name value => return .indexAssign name (← lowerIndex value)
  | .assign .real _ name (.load .real (.region region offset) mask _) =>
      return .load name region (← lowerIndex offset) (← lowerMask mask)
  | .store .real _ (.region region offset) (.add .real _ (.ref _ _ a) (.ref _ _ b)) mask =>
      return .storeAdd region (← lowerIndex offset) a b (← lowerMask mask)
  | _ => none

def lowerBody : List Stmt → Option (List Instruction)
  | [] => some []
  | stmt::rest => return (← lowerStatement stmt)::(← lowerBody rest)

def sourceCode (A B C : RegionName) (n : Nat) : List Instruction :=
  [.indexAssign "prog_id" (.program 0),
   .indexAssign "offs" (.add (.mul (.register "prog_id") (.literal 16)) (.lane 16)),
   .load "a" A (.register "offs") (.less (.register "offs") (.literal n)),
   .load "b" B (.register "offs") (.less (.register "offs") (.literal n)),
   .storeAdd C (.register "offs") "a" "b" (.less (.register "offs") (.literal n))]

/-- Checked extraction from the existing original-source DSL AST. The
addition is inside the fifth statement, and the original source has B16. -/
theorem original_ast_lowering (A B C : RegionName) (n : Nat) :
    lowerBody (_add_kernel A B C n 16).body=some (sourceCode A B C n) := by
  simp [lowerBody,lowerStatement,lowerIndex,lowerMask,_add_kernel,ComputeKernel.body,
    ComputeKernel.toAlgKernel,ComputeKernel.toAlgorithm?,sourceCode,bind,pure,Except.bind,Option.bind]

structure Registers (M α : Type) where
  indices : RegName → Nat → Nat
  values : RegName → Nat → α
  memory : M

def evalIndex (pid lane : Nat) (registers : RegName → Nat → Nat) : IndexExpr → Nat
  | .literal n => n
  | .program ax => if ax=0 then pid else 0
  | .register name => registers name lane
  | .lane _ => lane
  | .add a b => evalIndex pid lane registers a + evalIndex pid lane registers b
  | .mul a b => evalIndex pid lane registers a * evalIndex pid lane registers b

def evalMask (pid lane : Nat) (registers : RegName → Nat → Nat) : MaskExpr → Bool
  | .less a b => decide (evalIndex pid lane registers a < evalIndex pid lane registers b)

/-- Every load captures its own full active tile from the current memory.
Masked lanes receive arbitrary `undef`; no masked load accesses storage. -/
def executeInstruction {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (pid : Nat) (s : Registers M α) : Instruction → Registers M α
  | .indexAssign name value =>
      {s with indices := fun key lane ↦ if key=name then evalIndex pid lane s.indices value else s.indices key lane}
  | .load name region offset mask =>
      {s with values := fun key lane ↦ if key=name then
        if evalMask pid lane s.indices mask then
          readView ops s.memory (views region) (evalIndex pid lane s.indices offset) else undef
        else s.values key lane}
  | .storeAdd region offset a b mask =>
      {s with memory := (List.range 16).foldl (fun memory lane ↦
        if evalMask pid lane s.indices mask then
          writeView ops memory (views region) (evalIndex pid lane s.indices offset)
            (compute (s.values a lane) (s.values b lane))
        else memory) s.memory}

def executeBody {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (pid : Nat) (s : Registers M α) : List Instruction → Registers M α
  | [] => s
  | stmt::rest => executeBody ops compute views undef pid
      (executeInstruction ops compute views undef pid s stmt) rest

def runProgram {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (A B C : RegionName) (n pid : Nat) (memory : M) : Option M := do
  let code ← lowerBody (_add_kernel A B C n 16).body
  let start : Registers M α := ⟨fun _ _ ↦ 0,fun _ _ ↦ undef,memory⟩
  pure (executeBody ops compute views undef pid start code).memory

def programLanes (n pid : Nat) : List Nat :=
  ((List.range 16).map (fun lane ↦ pid*16+lane)).filter (fun i ↦ decide (i<n))

theorem fold_store {M α : Type} (ops : MemoryPrimitives M α) (output : View)
    (values : Nat → α) (n pid : Nat) (lanes : List Nat) (s : M) :
    lanes.foldl (fun memory lane ↦ if pid*16+lane<n then
      writeView ops memory output (pid*16+lane) (values (pid*16+lane)) else memory) s=
    runLoaded ops values output ((lanes.map (fun lane ↦ pid*16+lane)).filter (fun i ↦ decide (i<n))) s := by
  induction lanes generalizing s with
  | nil => rfl
  | cons lane rest ih =>
    by_cases h : pid*16+lane<n <;> simp [List.foldl_cons,List.filter_cons,h,runLoaded,ih]

/-- Operational reduction of all five actual source statements. Loads precede
all stores, masked values never reach a store, and no final memory is assumed. -/
theorem runProgram_eq_loaded {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (A B C : RegionName) (n pid : Nat) (s : M) :
    runProgram ops compute views undef A B C n pid s=
      some (runLoaded ops (fun i ↦ compute (readView ops s (views A) i) (readView ops s (views B) i))
        (views C) (programLanes n pid) s) := by
  simp only [runProgram,original_ast_lowering,Option.bind_some,bind,pure,Option.bind,sourceCode,executeBody,
    executeInstruction,evalIndex,evalMask]
  try simp only [if_pos rfl,if_true,String.reduceEq,if_false]
  congr 1
  have hfold := fold_store ops (views C)
    (fun i ↦ compute (readView ops s (views A) i) (readView ops s (views B) i)) n pid (List.range 16) s
  change _=runLoaded _ _ _ ((List.range 16).map (fun lane ↦ pid*16+lane) |>.filter (fun i ↦ decide (i<n))) s
  rw [←hfold]
  apply congrArg (fun f ↦ List.foldl f s (List.range 16))
  funext state lane
  by_cases h : pid*16+lane<n <;> simp [h]

def runPrograms {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (A B C : RegionName) (n : Nat) : List Nat → M → Option M
  | [],s => some s
  | pid::rest,s => do
      let t ← runProgram ops compute views undef A B C n pid s
      runPrograms ops compute views undef A B C n rest t

def runGrid {M α : Type} (ops : MemoryPrimitives M α) (compute : α → α → α)
    (views : RegionName → View) (undef : α) (A B C : RegionName) (n : Nat) (s : M) : Option M :=
  runPrograms ops compute views undef A B C n (List.range ((n+15)/16)) s

end
