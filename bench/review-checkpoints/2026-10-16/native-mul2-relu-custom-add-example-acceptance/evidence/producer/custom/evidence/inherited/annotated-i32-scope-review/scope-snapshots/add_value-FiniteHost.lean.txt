inductive InputType where
  | admitted (storage : Storage)
  | fnuzE4 | fnuzE5 | complex32 | complex64 | complex128 | e8m0 | packedFP4
  deriving DecidableEq

def InputType.name : InputType → String
  | .admitted d => TypedHost.dtypeName (TypedHost.wideDType d)
  | .fnuzE4 => "float8_e4m3fnuz" | .fnuzE5 => "float8_e5m2fnuz"
  | .complex32 => "complex32" | .complex64 => "complex64" | .complex128 => "complex128"
  | .e8m0 => "float8_e8m0fnu" | .packedFP4 => "float4_e2m1fn_x2"

def inputTypes : List InputType :=
  [.admitted .boolean,.admitted (.signed .w8),.admitted (.unsigned .w8),
   .admitted (.signed .w16),.admitted (.unsigned .w16),.admitted (.signed .w32),.admitted (.unsigned .w32),
   .admitted (.signed .w64),.admitted (.unsigned .w64),.admitted .fp16,.admitted .bf16,.admitted .fp32,
   .admitted .fp64,.admitted .f8e4fn,.admitted .f8e5,.fnuzE4,.fnuzE5,
   .complex32,.complex64,.complex128,.e8m0,.packedFP4]

def parseType (name : String) : Option InputType := inputTypes.find? (fun k ↦ k.name==name)

theorem parse_name (k : InputType) : parseType k.name=some k := by
  cases k with
  | admitted d => cases d <;> (try cases ‹FP64TypedCore.Width›) <;> decide +kernel
  | _ => decide +kernel

inductive Failure where
  | native (message : String)
  | cudaAssertion
  | signatureRejected
  | architectureRejected
  | unsupportedDType
  | undefinedDiagnostic (offset : Nat)

inductive Result (M : Type) where
  | ok (output : Tensor) (memory : M)
  | error (reason : Failure) (memory : M)

inductive Event where
  | emptyLike (input : RegionName)
  | cudaCheck (input output : Bool)
  | numel (output : RegionName) (n : Nat)
  | launch (n block grid : Nat) (valueBits : BitVec 32)
  | returnTensor (output : RegionName)

structure Run (M : Type) where
  trace : List Event
  result : Result M

structure Context where
  currentDevice : Nat

/-- Ordinary fallible allocation, current-device/stream/target preparation,
and admitted native lowering/argument/launch setup. They do not supply values
or the post-kernel physical state. Binder dtype failures are separately modeled. -/
structure Runtime (M : Type) where
  emptyLike : Tensor → M → Result M
  prepare : Tensor → Tensor → Nat → Except String Context
  nativeReady : Context → Storage → Tensor → Tensor → Nat → Except String Unit

def cudaGuard (x output : Tensor) : Bool := x.isCUDA && output.isCUDA

def launchTrace (x output : Tensor) : List Event :=
  [.emptyLike x.region,.cudaCheck x.isCUDA output.isCUDA,.numel output.region output.numel,
   .launch output.numel 1024 ((output.numel+1023)/1024) argument.bits]

def successTrace (x output : Tensor) : List Event := launchTrace x output ++ [.returnTensor output.region]

def dispatch {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (context : Context) (kind : InputType) (x output : Tensor) (allocated : M) : Result M :=
  match kind with
  | .fnuzE4 | .fnuzE5 => .error .architectureRejected allocated
  | .complex32 | .complex64 | .complex128 | .e8m0 | .packedFP4 => .error .signatureRejected allocated
  | .admitted d => match rt.nativeReady context d x output output.numel with
    | .error reason => .error (.native reason) allocated
    | .ok () => match sourceGrid b d (ops d) views (undef d) x.region output.region output.numel allocated with
      | .error offset => .error (.undefinedDiagnostic offset) allocated
      | .ok final => .ok output final

def launch {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (x output : Tensor) (allocated : M) : Result M :=
  match rt.prepare x output output.numel with
  | .error reason => .error (.native reason) allocated
  | .ok context => match parseType x.dtype with
    | none => .error .unsupportedDType allocated
    | some kind => dispatch b rt ops undef views context kind x output allocated

/-- Exact original wrapper; the assertion precedes numel and launch. Runtime
value10 is represented by its proved exact fp32 ABI word, never constexpr. -/
def puzzle1 {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (x : Tensor) (initial : M) : Run M :=
  match rt.emptyLike x initial with
  | .error reason failed => ⟨[.emptyLike x.region],.error reason failed⟩
  | .ok output allocated =>
    if cudaGuard x output then match launch b rt ops undef views x output allocated with
      | .error reason failed => ⟨launchTrace x output,.error reason failed⟩
      | .ok returned final => ⟨successTrace x output,.ok returned final⟩
    else ⟨[.emptyLike x.region,.cudaCheck x.isCUDA output.isCUDA],.error .cudaAssertion allocated⟩

/-- Torch reports a null data pointer for empty tensors even when their backing
storage and storage_offset are nonzero. Preserve that actual metadata without
identifying the unused affine address with the null pointer. -/
structure InputViewBinding (v : View) (metadata : TensorMetadata) (x : Tensor) : Prop where
  storageBase : v.storageBase=metadata.storageBase x
  storageBytes : v.storageBytes=metadata.storageBytes x
  emptyPointer : x.numel=0 → metadata.dataPointer x=0
  activePointer : 0<x.numel → v.address 0=metadata.dataPointer x

/-- Metadata and actual view bindings are unconditional, also for zero grids.
Fresh empty_like output has zero storage offset and empty storage. The input
retains actual empty pointer metadata independently of its affine address.
No pointer validity, alignment, value or permission is needed for an empty grid. -/
structure AllocationContract {M : Type} (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (x output : Tensor) (views : RegionName → View) (metadata : TensorMetadata) (initial allocated : M) : Prop where
  returned : TypedHost.ReturnedMetadata x output
  shapeProduct : x.numel=x.shape.prod
  strides : output.strides=metadata.emptyLikeStrides x
  inputBinding : InputViewBinding (views x.region) metadata x
  outputBinding : (views output.region).Binds metadata output
  typedViews : ∀ d, x.dtype=TypedHost.dtypeName (TypedHost.wideDType d) →
    (views x.region).Matches x (TypedHost.wideDType d) ∧ (views output.region).Matches output (TypedHost.wideDType d)
  frame : ∀ d address,
    (address<(views output.region).storageBase ∨ (views output.region).storageBase+(views output.region).storageBytes≤address) →
    (ops d).observeByte allocated address=(ops d).observeByte initial address

structure SafeAPI (x output : Tensor) (views : RegionName → View) : Prop where
  indexBound : output.numel<2^31
  devices : (∀ d, x.deviceIndex=some d → (views x.region).device=d) ∧
    (∀ d, output.deviceIndex=some d → (views output.region).device=d)
  inputAligned : (views x.region).Aligned
  outputAligned : (views output.region).Aligned
  inputValid : (views x.region).Valid output.numel (rounded output.numel)
  outputValid : (views output.region).Valid output.numel (rounded output.numel)
  separate : StorageDisjoint (views output.region) (views x.region)

structure MemoryApplicable {M α : Type} (ops : MemoryPrimitives M α) (x output : View) (n : Nat) : Prop where
  readX : ∀ i byte, i<n → byte<bytes x.dtype → ops.readableByte (x.address i+byte)
  readOutput : ∀ i byte, i<n → byte<bytes output.dtype → ops.readableByte (output.address i+byte)
  writeOutput : ∀ i byte, i<n → byte<bytes output.dtype → ops.writableByte (output.address i+byte)

structure LaunchSafety (x output : View) (n : Nat) : Prop where
  alignment : ∀ v∈[x,output], ∀ i, v.address i % bytes v.dtype=0
  indices : ∀ pid lane, pid<(n+1023)/1024 → lane<1024 →
    (start32 pid).toInt=(pid*1024 : Nat) ∧ (offset32 pid lane).toInt=(pid*1024+lane : Nat) ∧
    mask32 n pid lane=decide (pid*1024+lane<n)
  pointers : ∀ v∈[x,output], ∀ pid lane, pid<(n+1023)/1024 → lane<1024 →
    (pointer64 v pid lane).toNat=v.address (pid*1024+lane)
  accesses : ∀ v∈[x,output], ∀ i byte, i<n → byte<bytes v.dtype →
    v.storageBase≤v.address i+byte ∧ v.address i+byte<v.storageBase+v.storageBytes ∧ v.address i+byte<2^64
  ownership : ∀ i, i<n → i/1024<(n+1023)/1024 ∧ i%1024<1024 ∧ (i/1024)*1024+i%1024=i

theorem launch_safety (x output : Tensor) (views : RegionName → View) (safe : SafeAPI x output views) :
    LaunchSafety (views x.region) (views output.region) output.numel := by
  have valid : ∀ v∈[views x.region,views output.region], v.Valid output.numel (rounded output.numel) := by
    intro v hv; simp only [List.mem_cons,List.not_mem_nil,or_false] at hv
    rcases hv with rfl|rfl
    · exact safe.inputValid
    · exact safe.outputValid
  have aligned : ∀ v∈[views x.region,views output.region], v.Aligned := by
    intro v hv; simp only [List.mem_cons,List.not_mem_nil,or_false] at hv
    rcases hv with rfl|rfl
    · exact safe.inputAligned
    · exact safe.outputAligned
  exact ⟨fun v hv i ↦ address_aligned v (aligned v hv) i,
    fun pid lane hp hl ↦ source_index_arithmetic output.numel pid lane safe.indexBound hp hl,
    fun v hv pid lane hp hl ↦ source_pointer_exact v output.numel pid lane safe.indexBound (valid v hv) hp hl,
    fun v hv i byte hi hb ↦ active_byte_access v output.numel i byte (valid v hv) hi hb,
    fun i hi ↦ active_owner output.numel i hi⟩

#axiomsClean parse_name
#axiomsClean launch_safety
#guard_msgs (drop info) in
#auditModuleAxioms
end
