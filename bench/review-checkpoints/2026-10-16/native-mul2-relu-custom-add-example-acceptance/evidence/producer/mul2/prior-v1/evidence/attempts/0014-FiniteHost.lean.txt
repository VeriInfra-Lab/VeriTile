inductive InputType where
  | admitted (d : Storage)
  | fp8e4 | fp8e5 | fnuzE4 | fnuzE5 | complex32 | complex64 | complex128 | e8m0 | packedFP4
  deriving DecidableEq, Repr
def Width.name : Width → String | .w8 => "8" | .w16 => "16" | .w32 => "32" | .w64 => "64"
def FloatFormat.name : FloatFormat → String
  | .fp16 => "float16" | .bf16 => "bfloat16" | .fp32 => "float32" | .fp64 => "float64"
def Storage.name : Storage → String
  | .boolean => "bool" | .signed w => "int"++w.name | .unsigned w => "uint"++w.name | .floating f => f.name
def InputType.name : InputType → String
  | .admitted d => d.name
  | .fp8e4 => "float8_e4m3fn" | .fp8e5 => "float8_e5m2"
  | .fnuzE4 => "float8_e4m3fnuz" | .fnuzE5 => "float8_e5m2fnuz"
  | .complex32 => "complex32" | .complex64 => "complex64" | .complex128 => "complex128"
  | .e8m0 => "float8_e8m0fnu" | .packedFP4 => "float4_e2m1fn_x2"
def InputType.bytes : InputType → Nat
  | .admitted d => d.bytes | .complex32 => 4 | .complex64 => 8 | .complex128 => 16 | _ => 1
def inputTypes : List InputType :=
  [.admitted .boolean,.admitted (.signed .w8),.admitted (.unsigned .w8),
   .admitted (.signed .w16),.admitted (.unsigned .w16),.admitted (.signed .w32),.admitted (.unsigned .w32),
   .admitted (.signed .w64),.admitted (.unsigned .w64),.admitted (.floating .fp16),.admitted (.floating .bf16),
   .admitted (.floating .fp32),.admitted (.floating .fp64),.fp8e4,.fp8e5,.fnuzE4,.fnuzE5,
   .complex32,.complex64,.complex128,.e8m0,.packedFP4]
def parseType (name : String) : Option InputType := inputTypes.find? (fun k => k.name==name)
theorem parse_name (k : InputType) : parseType k.name=some k := by
  cases k with
  | admitted d => cases d <;> (try cases ‹Width›) <;> (try cases ‹FloatFormat›) <;> decide +kernel
  | _ => decide +kernel

structure Tensor where
  identity : Nat
  region : RegionName
  shape : List Nat
  strides : List Int
  numel : Nat
  dtype : InputType
  storageBase : Nat
  storageBytes : Nat
  storageOffset : Nat
  dataPointer : BitVec 64
  device : Option Nat
  readable : Bool
  writable : Bool
  deriving DecidableEq
def Tensor.byteRegion (x : Tensor) : ByteRegion :=
  ⟨x.storageBase,x.storageBytes,x.readable,x.writable,x.device⟩
def Tensor.Valid (x : Tensor) : Prop :=
  x.numel=x.shape.prod ∧ x.shape.length=x.strides.length ∧ x.numel<2^63 ∧
  (x.numel=0 → x.dataPointer=0) ∧
  (0<x.numel → x.dataPointer.toNat=x.storageBase+x.storageOffset*x.dtype.bytes)
def layout (x output : Tensor) (device : Nat) : Layout :=
  ⟨x.byteRegion,output.byteRegion,x.dataPointer,output.dataPointer,device⟩
structure Allocation where
  output : Tensor
  memory : ByteMemory

/-- Exactly Torch's ordinary metadata/allocation boundary. Preserve-format
strides are observed independently; no logical stride enters kernel indexing.
For an empty output both storage size and data pointer are zero. -/
structure ZerosLike (expectedStrides : List Int) (x : Tensor) (initial : ByteMemory)
    (allocation : Allocation) : Prop where
  identity : allocation.output.identity≠x.identity
  metadata : allocation.output.shape=x.shape ∧ allocation.output.numel=x.numel ∧
    allocation.output.dtype=x.dtype ∧ allocation.output.device=x.device
  strides : allocation.output.strides=expectedStrides
  offset : allocation.output.storageOffset=0
  size : allocation.output.storageBytes=x.numel*x.dtype.bytes
  pointer : allocation.output.dataPointer=(if x.numel=0 then 0 else BitVec.ofNat 64 allocation.output.storageBase)
  extent : allocation.output.storageBase+allocation.output.storageBytes≤2^64
  separate : allocation.output.storageBase+allocation.output.storageBytes≤x.storageBase ∨
    x.storageBase+x.storageBytes≤allocation.output.storageBase
  zero : ∀ address, allocation.output.byteRegion.Contains address → allocation.memory address=some 0
  frame : ∀ address, ¬allocation.output.byteRegion.Contains address → allocation.memory address=initial address

inductive Phase where | allocation | prepare | backend | pointers | launch deriving DecidableEq, Repr
inductive Failure where
  | external (phase : Phase) (message : String)
  | divisionByZero | signatureRejected | architectureRejected | blockRejected | fp8BackendRejected
  | gridRejected (requested : Int)
  | observation (issue : Issue)
  deriving DecidableEq, Repr
inductive Event where
  | zerosLike | numel (n : Nat) | cdiv (n : Nat) (block : Int) | prepare
  | signature | frontend | backend | gridParse (g : Int) | pointers
  | launch (g : Nat) | returnTensor (identity : Nat)
  deriving DecidableEq, Repr
structure Returned where
  tensor : Tensor
  memory : ByteMemory
structure Run where
  trace : List Event
  outcome : Except Failure Returned
def prepend (events : List Event) (run : Run) : Run := ⟨events++run.trace,run.outcome⟩
def failed (events : List Event) (issue : Failure) : Run := ⟨events,.error issue⟩

/-- Only ordinary allocation, device/stream preparation, resource-dependent
compilation/context setup, pointer extraction and launch initiation are external. No callback supplies
kernel values or final memory. Pointer setup is performed even for grid zero. -/
structure Runtime where
  zerosLike : Tensor → ByteMemory → Except String Allocation
  preserveFormatStrides : Tensor → List Int
  prepare : Tensor → Tensor → Except String Nat
  backend : Storage → Nat → Except String Unit
  pointers : Tensor → Tensor → Nat → Except String Unit
  nativeLaunch : Nat → Except String Unit
  schedule : Nat → List Nat
structure RuntimeContract (rt : Runtime) : Prop where
  allocations : ∀ x initial a, rt.zerosLike x initial=.ok a →
    ZerosLike (rt.preserveFormatStrides x) x initial a
  schedules : ∀ g, (rt.schedule g).Perm (List.range g)

def validBlock (block : Int) : Prop := ∃ k : Nat, k≤20 ∧ block=(2^k : Nat)
def hostGrid (n : Nat) (block : Int) : Int := ((n : Int)+block-1).fdiv block
def gridFits (g : Int) : Prop := -(2^31)≤g ∧ g<2^31
abbrev Launch := Storage → Layout → Mode → RegionName → RegionName → Nat → Nat →
  List Nat → ByteMemory → Except Issue ByteMemory

/-- Ordered native front-end domain: binder/architecture errors precede the
body block check; the fp8 LLVM arithmetic error follows valid frontend lowering. -/
def signature (dtype : InputType) : Except Failure (Option Storage) :=
  match dtype with
  | .complex32 | .complex64 | .complex128 | .e8m0 | .packedFP4 => .error .signatureRejected
  | .fnuzE4 | .fnuzE5 => .error .architectureRejected
  | .fp8e4 | .fp8e5 => .ok none
  | .admitted d => .ok (some d)

def launchWith (launch : Launch) (rt : Runtime) (mode : Mode) (x output : Tensor)
    (block : Int) (allocated : ByteMemory) : Run :=
  let g := hostGrid x.numel block
  match rt.prepare x output with
  | .error why => failed [.prepare] (.external .prepare why)
  | .ok device => prepend [.prepare,.signature] <|
    match signature x.dtype with
    | .error why => failed [] why
    | .ok dtype => prepend [.frontend] <|
      if ¬validBlock block then failed [] .blockRejected
      else match dtype with
        | none => failed [.backend] .fp8BackendRejected
        | some d => prepend [.backend] <|
          match rt.backend d block.toNat with
          | .error why => failed [] (.external .backend why)
          | .ok () => prepend [.gridParse g] <|
            if ¬gridFits g then failed [] (.gridRejected g)
            else prepend [.pointers] <|
              match rt.pointers x output device with
              | .error why => failed [] (.external .pointers why)
              | .ok () => prepend [.launch g.toNat] <|
                match (if g.toNat=0 then .ok () else rt.nativeLaunch g.toNat) with
                | .error why => failed [] (.external .launch why)
                | .ok () =>
                  match launch d (layout x output device) mode x.region output.region x.numel block.toNat
                      (rt.schedule g.toNat) allocated with
                  | .error issue => failed [] (.observation issue)
                  | .ok final => ⟨[.returnTensor output.identity],.ok ⟨output,final⟩⟩

def afterAllocation (launch : Launch) (rt : Runtime) (mode : Mode) (x : Tensor)
    (block : Int) (a : Allocation) : Run :=
  prepend [.numel x.numel,.cdiv x.numel block] <|
    if block=0 then failed [] .divisionByZero else launchWith launch rt mode x a.output block a.memory
def hostWith (launch : Launch) (rt : Runtime) (mode : Mode) (x : Tensor)
    (block : Int) (initial : ByteMemory) : Run :=
  match mode with
  | .inplace => afterAllocation launch rt mode x block ⟨x,initial⟩
  | .outplace => prepend [.zerosLike] <|
    match rt.zerosLike x initial with
    | .error why => failed [] (.external .allocation why)
    | .ok a => afterAllocation launch rt mode x block a

def executeHost (b : Numerics) : Runtime → Mode → Tensor → Int → ByteMemory → Run :=
  hostWith (sourceLaunch b)
def referenceLaunch (b : Numerics) : Launch := fun d p _ _ _ n block pids initial =>
  (launchPlan b d p n block pids initial).map (fun writes => byteResult writes initial)
def referenceHost (b : Numerics) : Runtime → Mode → Tensor → Int → ByteMemory → Run :=
  hostWith (referenceLaunch b)

theorem host_correspondence (b : Numerics) (rt : Runtime) (mode : Mode) (x : Tensor)
    (block : Int) (initial : ByteMemory) :
    executeHost b rt mode x block initial=referenceHost b rt mode x block initial := by
  have h : sourceLaunch b=referenceLaunch b := by
    funext d p mode input output n block pids initial
    exact launch_correspondence b d p mode input output n block pids initial
  unfold executeHost referenceHost
  rw [h]

theorem division_order (launch : Launch) (rt : Runtime) (x : Tensor) (initial : ByteMemory) :
    hostWith launch rt .inplace x 0 initial=
      ⟨[.numel x.numel,.cdiv x.numel 0],.error .divisionByZero⟩ ∧
    (∀ a, rt.zerosLike x initial=.ok a → hostWith launch rt .outplace x 0 initial=
      ⟨[.zerosLike,.numel x.numel,.cdiv x.numel 0],.error .divisionByZero⟩) := by
  constructor
  · simp [hostWith,afterAllocation,prepend,failed]
  · intro a h; simp [hostWith,h,afterAllocation,prepend,failed]

theorem launch_returned_metadata (launch : Launch) (rt : Runtime) (mode : Mode)
    (x output : Tensor) (block : Int) (allocated : ByteMemory) (ret : Returned)
    (h : (launchWith launch rt mode x output block allocated).outcome=.ok ret) : ret.tensor=output := by
  unfold launchWith at h
  split at h
  · cases h
  · simp only [prepend] at h
    split at h
    · cases h
    · simp only [prepend] at h
      split at h
      · cases h
      · split at h
        · cases h
        · simp only [prepend] at h
          split at h
          · cases h
          · simp only [prepend] at h
            split at h
            · cases h
            · simp only [prepend] at h
              split at h
              · cases h
              · simp only [prepend] at h
                split at h
                · cases h
                · split at h
                  · cases h
                  · cases h; rfl

theorem after_returned_metadata (launch : Launch) (rt : Runtime) (mode : Mode)
    (x : Tensor) (block : Int) (a : Allocation) (ret : Returned)
    (h : (afterAllocation launch rt mode x block a).outcome=.ok ret) : ret.tensor=a.output := by
  unfold afterAllocation prepend at h
  split at h
  · cases h
  · exact launch_returned_metadata launch rt mode x a.output block a.memory ret h

def ReturnedMetadata (rt : Runtime) (mode : Mode) (x : Tensor) (initial : ByteMemory) (ret : Returned) : Prop :=
  match mode with
  | .inplace => ret.tensor=x
  | .outplace => ∃ a, rt.zerosLike x initial=.ok a ∧ ret.tensor=a.output ∧
      ZerosLike (rt.preserveFormatStrides x) x initial a
theorem host_returned_metadata (b : Numerics) (rt : Runtime) (contract : RuntimeContract rt)
    (mode : Mode) (x : Tensor) (block : Int) (initial : ByteMemory) (ret : Returned)
    (h : (executeHost b rt mode x block initial).outcome=.ok ret) :
    ReturnedMetadata rt mode x initial ret := by
  cases mode with
  | inplace => exact after_returned_metadata _ rt .inplace x block ⟨x,initial⟩ ret h
  | outplace =>
    unfold executeHost hostWith prepend at h
    cases ha : rt.zerosLike x initial with
    | error why => simp [ha,failed] at h
    | ok a =>
      simp only [ha] at h
      exact ⟨a,ha,after_returned_metadata _ rt .outplace x block a ret h,contract.allocations x initial a ha⟩
