/-- A block which cannot store must still have its pre-gate load checked. -/
theorem nonzero_program_requires_its_load (ctx : AccessContext) (d : Storage)
    (input output : PhysicalView)
    (denied : ¬ ReadOK ctx input d.bytes (laneAddress input.pointer d.bytes 1 0)) :
    ¬ ProgramSafe ctx d input output 1025 1 := by
  intro safe
  apply denied
  exact safe.1 0 (by decide +kernel)

theorem same_program_has_no_writes (ctx : AccessContext) (b : Numerics) (d : Storage)
    (input output : PhysicalView) (memory : ByteMemory.Memory) :
    programWrites ctx.toBacking b d input.pointer output.pointer 1025 1 memory=[] := by
  simp [programWrites,FiniteAddress.stores32]

/-- The source mask changes width at exact constexpr boundaries. -/
theorem all_promotion_boundaries :
    active (2^31-1) (2^21) 0=Bool.true ∧
    active (2^31) (2^21) 0=Bool.false ∧
    active (2^31+1) (2^21) 0=Bool.true ∧
    active (2^32) (2^21) 0=Bool.true ∧
    active (2^63) (2^21) 0=Bool.false ∧
    active (2^64-1) (2^22-1) 1023=Bool.false := by decide +kernel

/-- Actual device address 8 resolves to canonical byte 108 on current device 7. -/
def ctx : AccessContext := ⟨7,fun a => a+100,fun device a => device=7 ∧ 8≤a ∧ a<16,fun _ _ => False⟩
def input : PhysicalView := ⟨8,108,8,8⟩

theorem translated_load_access : ReadOK ctx input 4 8 := by
  refine ⟨?_,?_,?_,?_,?_,?_⟩
  · norm_num [PhysicalView.Inside,input,ctx]
  · norm_num
  · norm_num [input]
  · norm_num [input]
  · norm_num
  · intro k hk
    dsimp [ctx]
    omega

theorem translated_load_not_raw (memory : ByteMemory.Memory) :
    ByteMemory.loadLE 4 (fun a => memory (ctx.toBacking a)) 8=readWord .fp32 memory 108 := by
  exact translated_load ctx input .fp32 memory 8 translated_load_access

/-- A nonzero allocator byte beyond the first tile is retained, not zero-filled. -/
theorem untouched_allocator_contents (b : Numerics) (d : Storage) (memory : ByteMemory.Memory)
    (address : Nat) (h : ∀ w∈programWrites id b d 0 8192 2048 0 memory,
      address<w.address ∨ w.address+4≤address) :
    byteFormula (programWrites id b d 0 8192 2048 0 memory) memory address=memory address := by
  rw [←writes_pointwise]
  exact writes_byte_frame _ _ _ h

theorem parser_grid_boundary :
    grid ((2^31-1)*1024)=2^31-1 ∧
    grid ((2^31-1)*1024+1)=2^31 ∧
    constexprAccepted ((2^31-1)*1024+1)=Bool.true := by decide +kernel
