theorem exact_grid_threshold (n : Nat) : gridSupported n ↔ n≤(2^31-1)*16 := by
  unfold gridSupported grid
  omega

theorem integer_minimum_discontinuity :
    mask (2^31-1) 0 0=Bool.true ∧
    mask (2^31) 0 0=Bool.false ∧
    mask (2^31+1) (2^27) 0=Bool.true ∧
    pointer (2^33+32) 4 (2^27) 0=32 := by decide +kernel

theorem repeated_native_request (data width : Nat) :
    pointer data width (2^28) 0=pointer data width 0 0 ∧
    mask (2^32+1) (2^28) 0=Bool.true ∧ mask (2^32+1) 0 0=Bool.true := by
  have h0 : offset 0 0=0 := by decide +kernel
  have hw : offset (2^28) 0=0 := by decide +kernel
  constructor
  · simp only [pointer,h0,hw]
  · decide +kernel

theorem repeated_output_is_observed (data width : Nat) (positive : 0<width) :
    Conflict (pointer data width (2^28) 0) width (span (pointer data width 0 0) width) := by
  rw [(repeated_native_request data width).1]
  refine ⟨0,positive,?_⟩
  simp [span,positive]

theorem owner_does_not_grant_access (base capacity address width : Nat) :
    ¬ (Buffer.mk base capacity base (some 0) (fun _ => Bool.false) (fun _ => Bool.false)).Readable 0 address width := by
  simp [Buffer.Readable]

/-- The original first dimension can be huge while numel is zero. This host
return does not require any load/access/scalar premise or a CUDA guard. -/
theorem zero_numel_huge_first_dimension_return {M : Type}
    (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (metadata : Metadata)
    (x y output : HostWrapper.Tensor) (initial allocated : M)
    (allocation : alloc x initial=.ok (output,allocated))
    (shape : output.shape=[2^31,0]) :
    output.shape.prod=0 ∧
    executeHost ops b alloc (fun _ => none) metadata (.canonical .fp32) (.canonical .fp32)
      x y initial=.returned (successTrace x output (2^31)) output allocated := by
  constructor
  · simp [shape]
  · exact vector_addition_custom_two31_return ops b alloc metadata
      (.canonical .fp32) (.canonical .fp32) (.floating .fp32 .fp32)
      x y output [0] initial allocated allocation fp32_admitted shape
