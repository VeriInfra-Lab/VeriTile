/-! Finite source acceptance contract for the original `add` wrapper and its
unchanged 1024-lane kernel. Trusted Torch metadata/allocation and ordinary typed
memory primitives are explicit. Strides are metadata: source pointers advance
linearly from data_ptr and storage_offset, including for noncontiguous inputs.
The safe/defined domain is applicability, never an invented wrapper check. -/

local instance : Inhabited TypedHost.Word := ⟨FP.Value.zero .fp32⟩

structure LaunchSafety (x y output : View) (n : Nat) : Prop where
  indices : ∀ pid lane, pid<(n+1023)/1024 → lane<1024 →
    (start32 pid).toInt=(pid*1024 : Nat) ∧
    (offset32 pid lane).toInt=(pid*1024+lane : Nat) ∧
    mask32 n pid lane=decide (pid*1024+lane<n)
  pointers : ∀ v∈[x,y,output], ∀ pid lane, pid<(n+1023)/1024 → lane<1024 →
    (pointer64 v pid lane).toNat=v.address (pid*1024+lane)
  accesses : ∀ v∈[x,y,output], ∀ i byte, i<n → byte<bytes v.dtype →
    v.storageBase≤v.address i+byte ∧ v.address i+byte<v.storageBase+v.storageBytes ∧ v.address i+byte<2^64
  ownership : ∀ i, i<n → i/1024<(n+1023)/1024 ∧ i%1024<1024 ∧ (i/1024)*1024+i%1024=i
  outputDistinct : ∀ i j, i≠j → SpanDisjoint (output.address i) (output.finish i) (output.address j) (output.finish j)

theorem launch_safety (x y output : View) (n : Nat) (hn : n<2^31)
    (hx : x.Valid n (rounded n)) (hy : y.Valid n (rounded n)) (ho : output.Valid n (rounded n)) :
    LaunchSafety x y output n := by
  have valid : ∀ v∈[x,y,output], v.Valid n (rounded n) := by
    intro v hv
    simp only [List.mem_cons,List.not_mem_nil,or_false] at hv
    rcases hv with rfl|rfl|rfl
    · exact hx
    · exact hy
    · exact ho
  exact ⟨fun pid lane hp hl ↦ source_index_arithmetic n pid lane hn hp hl,
    fun v hv pid lane hp hl ↦ source_pointer_exact v n pid lane hn (valid v hv) hp hl,
    fun v hv i byte hi hb ↦ active_byte_access v n i byte (valid v hv) hi hb,
    fun i hi ↦ active_owner n i hi,fun i j hij ↦ distinct_cells output i j hij⟩

def physicalRun {M : Type} (b : Numerics) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : View) (n : Nat) (physical : M) : M :=
  runBatches ops (scalarSource b r) x y output (sourceBatches n) physical

structure PhysicalCorrect {M σ : Type} (b : Numerics) (r : Route)
    (ops : MemoryPrimitives M (Carrier r)) (p : ValueProofs.Parameters)
    (allocated final : World σ) (physical : M) (x y output : View) : Prop where
  values : ∀ i, i<p.n → readView ops (physicalRun b r ops x y output p.n physical) output i=
    sourceCell b r p allocated i
  logicalResult : ∀ i, i<p.n → readView ops (physicalRun b r ops x y output p.n physical) output i=
    logicalRead r final p.output i
  inputs : InputsPreserved ops x y p.n physical (physicalRun b r ops x y output p.n physical)
  frame : ∀ address, ¬ output.active p.n address →
    ops.observeByte (physicalRun b r ops x y output p.n physical) address=ops.observeByte physical address

theorem physical_correct {M σ : Type} (b : Numerics) (r : Route)
    (ops : MemoryPrimitives M (Carrier r)) (p : ValueProofs.Parameters)
    (allocated final : World σ) (physical : M) (x y output : View)
    (hx : x.Valid p.n (rounded p.n)) (hy : y.Valid p.n (rounded p.n)) (ho : output.Valid p.n (rounded p.n))
    (dx : StorageDisjoint output x) (dy : StorageDisjoint output y)
    (inputs : InputsRelation r ops p allocated physical x y)
    (formula : project r final=expected b r p allocated) :
    PhysicalCorrect b r ops p allocated final physical x y output := by
  have hc : physicalRun b r ops x y output p.n physical=
      runLive ops (scalarSource b r) x y output (activeLanes p.n) physical :=
    source_batches_correspondence ops (scalarSource b r) x y output p.n physical hx hy ho dx dy
  have result := physical_output_matches b r ops p allocated final physical x y output hx hy ho dx dy inputs formula
  refine ⟨?_,?_,?_,?_⟩
  · intro i hi
    rw [hc,result i hi]
    exact output_cell_of_project b r p allocated final formula i hi
  · intro i hi
    rw [hc]
    exact result i hi
  · rw [hc]
    exact live_preserves_inputs ops (scalarSource b r) x y output p.n (activeLanes p.n) physical physical
      (fun i hi ↦ (mem_activeLanes p.n i).1 hi) hx hy ho dx dy (fun _ _ ↦ ⟨rfl,rfl⟩)
  · intro address outside
    rw [hc]
    exact live_byte_frame ops (scalarSource b r) x y output p.n (activeLanes p.n) physical
      (fun i hi ↦ (mem_activeLanes p.n i).1 hi) address outside

/-- Complete successful branch: actual source-order host and actual typed grid,
independent defined initial-memory result, return metadata, both logical frames,
source load-before-store physical replay, byte frame and finite-width safety. -/
structure FullSuccess {M σ : Type} (b : Numerics) (rt : Runtime σ) (r : Route)
    (ops : MemoryPrimitives M (Carrier r)) (x y output : Tensor)
    (initial allocated final : World σ) (physical : M) (vx vy vo : View) : Prop where
  host : add (withLaunch b rt) x y initial=⟨successTrace x y output,.ok output final⟩
  kernel : execute b r (parameters x y output) allocated=.ok final
  metadata : ReturnedMetadata x output
  formula : project r final=definedExpected b r (parameters x y output) allocated
  logicalFrame : Frame (parameters x y output) allocated final
  unselected : UnselectedUnchanged r allocated final
  physical : PhysicalCorrect b r ops (parameters x y output) allocated final physical vx vy vo
  safety : LaunchSafety vx vy vo output.numel

specification vector_addition_finite_success {M σ : Type} (b : Numerics) (rt : Runtime σ)
    (xD yD : DType) (r : Route) (ops : MemoryPrimitives M (Carrier r))
    (x y output : Tensor) (initial allocated : World σ) (physical : M) (vx vy vo : View)
    (selected : route xD yD=some r)
    (ha : rt.emptyLike x initial=.ok output allocated)
    (hc : cudaGuard x y output=Bool.true)
    (allocation : EmptyLikeContract x y output initial allocated)
    (safe : SafeAPI x y output xD yD vx vy vo)
    (values : ValueDomain b r (parameters x y output) allocated)
    (inputs : InputsRelation r ops (parameters x y output) allocated physical vx vy) :
    ∃ final, FullSuccess b rt r ops x y output initial allocated final physical vx vy vo := by
  obtain ⟨final,executed,formula,frame,unchanged⟩ :=
    execute_correct b r (parameters x y output) allocated rfl values.1
  refine ⟨final,?_,executed,allocation.metadata,?_,frame,unchanged,?_,?_⟩
  · apply add_success (withLaunch b rt) x y output initial allocated final ha hc
    exact launch_selected b xD yD r x y output (parameters x y output) allocated final
      safe.xView.2.2 safe.yView.2.2 selected executed
  · exact formula.trans (expected_policy_independent b r (parameters x y output) allocated values.2.2)
  · exact physical_correct b r ops (parameters x y output) allocated final physical vx vy vo
      safe.xValid safe.yValid safe.outputValid safe.outputSeparateX safe.outputSeparateY inputs formula
  · apply launch_safety vx vy vo output.numel _ safe.xValid safe.yValid safe.outputValid
    rw [allocation.metadata.numel]
    exact safe.sourceDomain.1

/-- Exact canonical frontend partition. Safe/defined applicability is required
only for an admitted route; compiler rejections retain their first source stage.
No complementary native rollback or poison result is invented for unsafe input. -/
specification vector_addition_finite_outcome {M σ : Type} (b : Numerics) (rt : Runtime σ)
    (xD yD : DType) (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (x y output : Tensor) (initial allocated : World σ) (physical : M) (vx vy vo : View)
    (hx : x.dtype=dtypeName xD) (hy : y.dtype=dtypeName yD)
    (ha : rt.emptyLike x initial=.ok output allocated)
    (hc : cudaGuard x y output=Bool.true)
    (allocation : EmptyLikeContract x y output initial allocated)
    (applicable : ∀ r, route xD yD=some r → SafeAPI x y output xD yD vx vy vo ∧
      ValueDomain b r (parameters x y output) allocated ∧
      InputsRelation r (ops r) (parameters x y output) allocated physical vx vy) :
    (∃ arg bad, DTypeAdmission.admission xD yD=.signatureRejected arg bad ∧
      add (withLaunch b rt) x y initial=
        ⟨TypedHostCorrect.rejectedTrace x y output,.error (.native "frontend: signature rejected") allocated⟩) ∨
    (DTypeAdmission.admission xD yD=.additionRejected ∧
      add (withLaunch b rt) x y initial=
        ⟨TypedHostCorrect.rejectedTrace x y output,.error (.native "frontend: addition rejected") allocated⟩) ∨
    (∃ r final, route xD yD=some r ∧ FullSuccess b rt r (ops r) x y output initial allocated final physical vx vy vo) := by
  cases h : DTypeAdmission.admission xD yD with
  | signatureRejected arg bad =>
    exact Or.inl ⟨arg,bad,rfl,signature_rejection b rt xD yD arg bad x y output initial allocated hx hy ha hc h⟩
  | additionRejected =>
    exact Or.inr (Or.inl ⟨rfl,addition_rejection b rt xD yD x y output initial allocated hx hy ha hc h⟩)
  | passesDTypeGates computation stored =>
    have existsRoute := route_matches_admission xD yD
    simp only [h,DTypeAdmission.Outcome.stage,decide_true] at existsRoute
    cases hr : route xD yD with
    | none => simp [hr] at existsRoute
    | some r =>
      obtain ⟨safe,values,inputs⟩ := applicable r hr
      obtain ⟨final,success⟩ := vector_addition_finite_success b rt xD yD r (ops r) x y output initial allocated
        physical vx vy vo hr ha hc allocation safe values inputs
      exact Or.inr (Or.inr ⟨r,final,rfl,success⟩)

specification vector_addition_finite_allocation_error {σ : Type} (b : Numerics) (rt : Runtime σ)
    (x y : Tensor) (initial failed : World σ) (reason : Failure)
    (ha : rt.emptyLike x initial=.error reason failed) :
    add (withLaunch b rt) x y initial=⟨[.emptyLike x.region],.error reason failed⟩ :=
  TypedHostCorrect.vector_addition_typed_host_allocation_error b rt x y initial failed reason ha

specification vector_addition_finite_cuda_error {σ : Type} (b : Numerics) (rt : Runtime σ)
    (x y output : Tensor) (initial allocated : World σ)
    (ha : rt.emptyLike x initial=.ok output allocated) (hc : cudaGuard x y output=Bool.false) :
    add (withLaunch b rt) x y initial=
      ⟨[.emptyLike x.region,.cudaCheck x.isCUDA y.isCUDA output.isCUDA],.error .cudaAssertion allocated⟩ :=
  TypedHostCorrect.vector_addition_typed_host_cuda_error b rt x y output initial allocated ha hc
