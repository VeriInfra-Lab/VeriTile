namespace VeriTile.Bench.TritonBenchG.VectorAddition.Finite
set_option maxHeartbeats 6000000

/-- One source program loads both active input tiles before storing its output
tile. Masked lanes have no external memory operation; their arithmetic is
retained in the original typed source interpreter. -/
def runBatches {M A : Type} (ops : MemoryPrimitives M A) (compute : A → A → A)
    (x y output : View) : List (List Nat) → M → M
  | [],s => s
  | jobs::rest,s =>
      let values := fun i ↦ compute (readView ops s x i) (readView ops s y i)
      runBatches ops compute x y output rest (runLoaded ops values output jobs s)

theorem live_append {M A : Type} (ops : MemoryPrimitives M A) (compute : A → A → A)
    (x y output : View) (as bs : List Nat) (s : M) :
    runLive ops compute x y output (as++bs) s=
      runLive ops compute x y output bs (runLive ops compute x y output as s) := by
  induction as generalizing s with
  | nil => rfl
  | cons i as ih => exact ih _

theorem batches_eq_live {M A : Type} (ops : MemoryPrimitives M A) (compute : A → A → A)
    (x y output : View) (n : Nat) (batches : List (List Nat)) (initial : M)
    (active : ∀ i∈batches.flatten, i<n)
    (hx : x.Valid n (rounded n)) (hy : y.Valid n (rounded n)) (ho : output.Valid n (rounded n))
    (dx : StorageDisjoint output x) (dy : StorageDisjoint output y) :
    runBatches ops compute x y output batches initial=
      runLive ops compute x y output batches.flatten initial := by
  induction batches generalizing initial with
  | nil => rfl
  | cons jobs rest ih =>
    have ha : ∀ i∈jobs, i<n := fun i hi ↦ active i (by simp [hi])
    have hr : ∀ i∈rest.flatten, i<n := fun i hi ↦ active i (by simp [hi])
    rw [runBatches,ih _ hr,List.flatten_cons,live_append,
      source_load_all_store_correspondence ops compute x y output n initial jobs ha hx hy ho dx dy]

def programLanes (n pid : Nat) : List Nat :=
  ((List.range 1024).map (fun lane ↦ pid*1024+lane)).filter (fun i ↦ decide (i<n))

def sourceBatches (n : Nat) : List (List Nat) :=
  (List.range ((n+1023)/1024)).map (programLanes n)

theorem source_batches_flatten (n : Nat) : (sourceBatches n).flatten=activeLanes n := by
  simp only [sourceBatches,activeLanes,List.flatMap_def]
  rfl

theorem source_batches_correspondence {M A : Type} (ops : MemoryPrimitives M A)
    (compute : A → A → A) (x y output : View) (n : Nat) (initial : M)
    (hx : x.Valid n (rounded n)) (hy : y.Valid n (rounded n)) (ho : output.Valid n (rounded n))
    (dx : StorageDisjoint output x) (dy : StorageDisjoint output y) :
    runBatches ops compute x y output (sourceBatches n) initial=
      runLive ops compute x y output (activeLanes n) initial := by
  rw [batches_eq_live ops compute x y output n (sourceBatches n) initial ?_ hx hy ho dx dy,
    source_batches_flatten]
  intro i hi
  exact (mem_activeLanes n i).1 (by simpa [source_batches_flatten] using hi)

theorem live_preserves_inputs {M A : Type} (ops : MemoryPrimitives M A) (compute : A → A → A)
    (x y output : View) (n : Nat) (jobs : List Nat) (base current : M)
    (active : ∀ i∈jobs, i<n)
    (hx : x.Valid n (rounded n)) (hy : y.Valid n (rounded n)) (ho : output.Valid n (rounded n))
    (dx : StorageDisjoint output x) (dy : StorageDisjoint output y)
    (preserved : InputsPreserved ops x y n base current) :
    InputsPreserved ops x y n base (runLive ops compute x y output jobs current) := by
  induction jobs generalizing current with
  | nil => exact preserved
  | cons i jobs ih =>
    apply ih _ (fun j hj ↦ active j (by simp [hj]))
    apply write_preserves_inputs ops x y output n i base current _ preserved
    · intro j hj
      exact separated_accesses output x n n (rounded n) (rounded n) i j ho hx (active i (by simp)) hj dx
    · intro j hj
      exact separated_accesses output y n n (rounded n) (rounded n) i j ho hy (active i (by simp)) hj dy

theorem outside_span (output : View) (n i address : Nat) (hi : i<n)
    (outside : ¬ output.active n address) : address<output.address i ∨ output.finish i≤address := by
  have hlo : output.address 0≤output.address i := by
    unfold View.address
    exact Nat.add_le_add_left (Nat.mul_le_mul_right _ (by omega)) _
  have hhi : output.finish i≤output.address n := by
    rw [address_step]
    unfold View.address
    exact Nat.add_le_add_left (Nat.mul_le_mul_right _ (by omega)) _
  unfold View.active at outside
  omega

theorem live_byte_frame {M A : Type} (ops : MemoryPrimitives M A) (compute : A → A → A)
    (x y output : View) (n : Nat) (jobs : List Nat) (initial : M)
    (active : ∀ i∈jobs, i<n) (address : Nat) (outside : ¬ output.active n address) :
    ops.observeByte (runLive ops compute x y output jobs initial) address=ops.observeByte initial address := by
  induction jobs generalizing initial with
  | nil => rfl
  | cons i jobs ih =>
    rw [runLive,ih _ (fun j hj ↦ active j (by simp [hj]))]
    exact ops.byte_frame _ _ _ _ _ (outside_span output n i address (active i (by simp)) outside)
