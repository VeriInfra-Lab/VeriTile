namespace QuantTransposeFinite
set_option maxRecDepth 24000
set_option maxHeartbeats 8000000

/-- Complete source finite-value and physical-memory statement. No Safe
premise hides undefined active conversions. Every successful execution has an
actual host scalar-write and kernel byte-store replay, with independent output
and byte-frame properties. Rejections assert no unused intermediate numerical
value or post-failure memory result. -/
specification quant_transpose_finite_acceptance
    (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations) (m : Memory)
    (x : String) (logical : List (Word d)) (xs : Nat → Word d) (p : PhysicalLayout)
    (D : Domain d a alloc m x logical xs) (P : PhysicalBinding d a alloc m x p)
    (nonempty : 0<a.numel) (rank : a.shape.length=2) (stride : a.sam=1 ∨ a.san=1)
    (cuda : a.isCuda=Bool.true) (trials : List Config) (last : Config) :
    (∃ maximum, produced b d logical=some maximum ∧
      ((∃ i, executeAPI b (fun _ => none) (dtype d) a alloc m x trials last=
          .attempted d (.error (.interpreter (.conversionUndefined i)))) ∨
       (∃ ret physical, executeAPI b (fun _ => none) (dtype d) a alloc m x trials last=.attempted d (.ok ret) ∧
          Result b d a alloc m xs maximum ret ∧
          PhysicalStores a alloc p (allocated d a alloc m maximum (b.reciprocal d maximum))
            (hostPhysical d p maximum (b.reciprocal d maximum)) ret.memory physical ∧
          PhysicalResult b d a alloc p xs maximum ret physical))) ∧
    profiles.map (fun dt => admission (fun _ => none) .cuda dt 3 2 Bool.true)=observedCuda ∧
    (∀ env dt input why, admission env (if input.isCuda then .cuda else .cpu) dt input.numel input.shape.length
      (decide (input.sam=1 ∨ input.san=1))=.error why →
      executeAPI b env dt input alloc m x trials last=.rejected why) := by
  refine ⟨?_,quant_transpose_dtype_admission_inventory,?_⟩
  · obtain ⟨maximum,hm,hout⟩ := quant_transpose_all_finite_value_outcomes b d a alloc m x logical xs
      D nonempty rank stride cuda trials last
    refine ⟨maximum,hm,?_⟩
    rcases hout with hf | hs
    · exact Or.inl hf
    · obtain ⟨ret,hr,R⟩ := hs
      right
      have hroute := admitted_route d a.numel nonempty
      have he : execute b (fun _ => none) d a alloc m x trials last=.ok ret := by
        simpa [executeAPI,cuda,rank,stride,hroute] using hr
      have hp := prepare_success b d a alloc m x logical maximum D.logical hm rank stride
      have trace := execute_physical b d a alloc m x p D.matrixCount D.fresh cuda
        ⟨maximum,b.reciprocal d maximum,allocated d a alloc m maximum (b.reciprocal d maximum)⟩ hp trials last ret he
      change PhysicalStores a alloc p (allocated d a alloc m maximum (b.reciprocal d maximum))
        (externalMemory d a alloc p (allocated d a alloc m maximum (b.reciprocal d maximum)))
        ret.memory (externalBytes d a alloc p ret) at trace
      rw [host_physical_preparation d a alloc m x p D.fresh P maximum (b.reciprocal d maximum)] at trace
      exact ⟨ret,externalBytes d a alloc p ret,hr,R,trace,
        quant_transpose_external_result b d a alloc m x p xs maximum ret D.matrixCount P R⟩
  · intro env dt input why h
    exact quant_transpose_rejected_source_outcome b env dt input alloc m x trials last why h

#axiomsClean quant_transpose_finite_acceptance
#guard_msgs (drop info) in
#auditModuleAxioms
end QuantTransposeFinite
