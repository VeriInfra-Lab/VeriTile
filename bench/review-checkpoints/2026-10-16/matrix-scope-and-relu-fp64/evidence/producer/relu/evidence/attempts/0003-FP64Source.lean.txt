import bench.tritonbench_g.relu_triton_kernel.Proofs.HostWrapper
import bench.tritonbench_g.relu_triton_kernel.Proofs.FiniteAddress

/-! Additive binary64 source route. The memory consists of total typed views of
disjoint input/output regions. Active loads require the host input view to be
valid; inactive loads use an arbitrary supplied f64 word, never implicit zero.
All seven Python statements execute, including the pre-gate statements at
nonzero pids. The f64 comparison/select precedes the gated f32 narrowing/store.
Native numerical correspondence is separate from these execution proofs. -/
namespace VeriTile.Bench.TritonBenchG.ReluTritonKernel.FP64Source
open VeriTile.Triton
noncomputable section
set_option maxHeartbeats 3000000
set_option maxRecDepth 24000
set_option linter.unusedSimpArgs Bool.false

abbrev F64 := BitVec 64
abbrev F32 := BitVec 32

@[ext] structure Memory where
  f64 : RegionName → Nat → F64
  f32 : RegionName → Nat → F32

/-- Exactly one unverified numerical port: TTIR arith.truncf f64→f32.
It is a total word operation, including zero signs, subnormals and specials. -/
structure Numerics where
  narrow64to32 : F64 → F32

/-- IEEE ordered >= +0 on binary64 words, retaining -0 and rejecting NaNs.
Its correspondence to the native ordered comparator is explicitly registered. -/
def orderedGeZero64 (v : F64) : Bool :=
  let exp := (v.extractLsb' 52 11).toNat
  let frac := (v.extractLsb' 0 52).toNat
  !(exp == 2047 && frac != 0) && (!v.msb || (exp == 0 && frac == 0))

def select64 (condition : Bool) (yes no : F64) : F64 := if condition then yes else no

/-- Independent scalar reference; neither the statement interpreter nor its
registers appear in this definition. -/
def value (num : Numerics) (v : F64) : F32 :=
  num.narrow64to32 (if orderedGeZero64 v then v else 0#64)

structure Registers (B : Nat) where
  pid : Nat
  blockStart : Nat
  offsets : Fin B → Nat
  mask : Fin B → Bool
  loaded : Fin B → F64
  result : Fin B → F64

structure State (B : Nat) where
  mem : Memory
  regs : Registers B
  undef : Fin B → F64

def initial (B : Nat) (mem : Memory) (undef : Fin B → F64) : State B :=
  ⟨mem, ⟨0, 0, fun _ => 0, fun _ => Bool.false, undef, undef⟩, undef⟩

/-- Stores use the actual offset/mask/result registers. The source's affine
offsets are injective, as proved below, so the witness never resolves a race. -/
def scatter32 (B : Nat) (mem : Memory) (y : RegionName)
    (offsets : Fin B → Nat) (mask : Fin B → Bool) (values : Fin B → F32) : Memory :=
  { mem with f32 := fun r o =>
      if r = y then
        if h : ∃ i, mask i = Bool.true ∧ offsets i = o then values (Classical.choose h)
        else mem.f32 r o
      else mem.f32 r o }

theorem scatter32_readback (B : Nat) (mem : Memory) (y : RegionName)
    (offsets : Fin B → Nat) (mask : Fin B → Bool) (values : Fin B → F32)
    (injective : Function.Injective offsets) (i : Fin B) (active : mask i = Bool.true) :
    (scatter32 B mem y offsets mask values).f32 y (offsets i) = values i := by
  simp only [scatter32, ↓reduceIte]
  have h : ∃ j, mask j = Bool.true ∧ offsets j = offsets i := ⟨i, active, rfl⟩
  rw [dif_pos h]
  rw [injective (Classical.choose_spec h).2]

theorem scatter32_frame (B : Nat) (mem : Memory) (y r : RegionName) (o : Nat)
    (offsets : Fin B → Nat) (mask : Fin B → Bool) (values : Fin B → F32)
    (outside : r ≠ y ∨ ∀ i, mask i = Bool.true → offsets i ≠ o) :
    (scatter32 B mem y offsets mask values).f32 r o = mem.f32 r o := by
  rcases outside with h | h
  · simp [scatter32, h]
  · by_cases hr : r = y
    · simp only [scatter32, hr, ↓reduceIte]
      rw [dif_neg]
      rintro ⟨i, hi, ho⟩
      exact h i hi ho
    · simp [scatter32, hr]

inductive Statement where
  | programId | blockStart | offsets | mask | load64 | whereGeZero64 | gatedStore32
  deriving DecidableEq, Repr

/-- Source order matches original Python lines 8–21 and frozen fp64 TTIR. -/
def source : List Statement :=
  [.programId, .blockStart, .offsets, .mask, .load64, .whereGeZero64, .gatedStore32]

def step (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (stmt : Statement) (s : State B) : State B :=
  match stmt with
  | .programId => { s with regs.pid := pid }
  | .blockStart => { s with regs.blockStart := s.regs.pid * B }
  | .offsets => { s with regs.offsets := fun i => s.regs.blockStart + i.val }
  | .mask => { s with regs.mask := fun i => decide (s.regs.offsets i < N) }
  | .load64 => { s with regs.loaded := fun i =>
      if (s.regs.mask i) then s.mem.f64 x (s.regs.offsets i) else s.undef i }
  | .whereGeZero64 => { s with regs.result := fun i =>
      select64 (orderedGeZero64 (s.regs.loaded i)) (s.regs.loaded i) 0#64 }
  | .gatedStore32 => if s.regs.pid = 0 then
      { s with mem := scatter32 B s.mem y s.regs.offsets s.regs.mask
          (fun i => num.narrow64to32 (s.regs.result i)) }
    else s

def exec (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (stmts : List Statement) (s : State B) : State B :=
  stmts.foldl (fun t stmt => step num x y N B pid stmt t) s

def run (num : Numerics) (x y : RegionName) (N B pid : Nat) (s : State B) : State B :=
  exec num x y N B pid source s

def loadedValue (x : RegionName) (N B pid : Nat) (s : State B) (i : Fin B) : F64 :=
  if pid * B + i.val < N then s.mem.f64 x (pid * B + i.val) else s.undef i

def computedRegisters (x : RegionName) (N B pid : Nat) (s : State B) : Registers B :=
  ⟨pid, pid * B, fun i => pid * B + i.val,
   fun i => decide (pid * B + i.val < N),
   loadedValue x N B pid s,
   fun i => if orderedGeZero64 (loadedValue x N B pid s i)
     then loadedValue x N B pid s i else 0#64⟩

/-- The output formula is defined directly from the initial memory. -/
def reference (num : Numerics) (x y : RegionName) (N B : Nat) (mem : Memory) : Memory :=
  { mem with f32 := fun r o =>
      if r = y ∧ o < B ∧ o < N then value num (mem.f64 x o) else mem.f32 r o }

theorem pre_gate_registers (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (s : State B) :
    exec num x y N B pid (source.take 6) s =
      { s with regs := computedRegisters x N B pid s } := by
  simp [exec, source, step, computedRegisters, loadedValue, select64]

theorem source_registers (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (s : State B) :
    (run num x y N B pid s).regs = computedRegisters x N B pid s ∧
    (run num x y N B pid s).undef = s.undef := by
  simp only [run, exec, source, List.foldl_cons, List.foldl_nil, step]
  split <;> simp [computedRegisters, loadedValue, select64]

/-- Every pid has already loaded and computed its f64 result before this
gate. Only pid zero narrows/stores. -/
theorem source_memory (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (s : State B) :
    (run num x y N B pid s).mem =
      if pid = 0 then reference num x y N B s.mem else s.mem := by
  by_cases hp : pid = 0
  · subst pid
    simp only [run, exec, source, List.foldl_cons, List.foldl_nil, step,
      Nat.zero_mul, Nat.zero_add, ↓reduceIte]
    apply Memory.ext
    · rfl
    · funext r o
      by_cases h : r = y ∧ o < B ∧ o < N
      · rcases h with ⟨rfl, hb, hn⟩
        let i : Fin B := ⟨o, hb⟩
        have hi : (fun j : Fin B => decide (j.val < N)) i = Bool.true := by simp [i, hn]
        rw [scatter32_readback B s.mem y (fun j : Fin B => j.val)
          (fun j : Fin B => decide (j.val < N)) _ (fun _ _ h => Fin.ext h) i hi]
        simp [reference, value, select64, i, hb, hn]
      · rw [scatter32_frame]
        · simp [reference, h]
        · by_cases hr : r = y
          · right
            intro i hi heq
            apply h
            exact ⟨hr, heq ▸ i.isLt, heq ▸ of_decide_eq_true hi⟩
          · exact Or.inl hr
  · simp [run, exec, source, step, hp]

theorem source_f64_frame (num : Numerics) (x y : RegionName) (N B pid : Nat)
    (s : State B) : (run num x y N B pid s).mem.f64 = s.mem.f64 := by
  rw [source_memory]
  split <;> rfl

theorem reference_frame (num : Numerics) (x y r : RegionName) (N B o : Nat)
    (mem : Memory) (outside : r ≠ y ∨ B ≤ o ∨ N ≤ o) :
    (reference num x y N B mem).f32 r o = mem.f32 r o := by
  simp only [reference]
  rw [if_neg]
  intro h
  rcases outside with h' | h' | h'
  · exact h' h.1
  · omega
  · omega

theorem reference_idempotent (num : Numerics) (x y : RegionName) (N B : Nat)
    (mem : Memory) : reference num x y N B (reference num x y N B mem) =
      reference num x y N B mem := by
  apply Memory.ext
  · rfl
  · funext r o
    simp only [reference]
    split <;> rfl

/-- Check specials without substituting real numbers or truncating f64 input. -/
theorem comparison_boundaries :
    orderedGeZero64 (0#64) = Bool.true ∧
    orderedGeZero64 (0x8000000000000000#64) = Bool.true ∧
    orderedGeZero64 (0x0000000000000001#64) = Bool.true ∧
    orderedGeZero64 (0x8000000000000001#64) = Bool.false ∧
    orderedGeZero64 (0x7ff0000000000000#64) = Bool.true ∧
    orderedGeZero64 (0xfff0000000000000#64) = Bool.false ∧
    orderedGeZero64 (0x7ff8000000000001#64) = Bool.false ∧
    orderedGeZero64 (0xfff0000000000001#64) = Bool.false := by decide +kernel

#specNonCircular reference avoiding [source, step, exec, run]
#axiomsClean scatter32_readback
#axiomsClean scatter32_frame
#axiomsClean pre_gate_registers
#axiomsClean source_registers
#axiomsClean source_memory
#axiomsClean source_f64_frame
#axiomsClean reference_frame
#axiomsClean reference_idempotent
#axiomsClean comparison_boundaries
#guard_msgs (drop info) in
#auditModuleAxioms
end
