from pathlib import Path
import ast,hashlib,json
W=Path(__file__).resolve().parent
texts={'baseline':(W/'evidence/baseline-check_original_program_scope.py.txt').read_text(),
 'C':(W/'candidate/check_original_program_scope.py').read_text(),
 'A+C':(W/'candidate/combined-check_original_program_scope.py').read_text()}
dump=lambda n:ast.dump(n,include_attributes=False)
base=ast.parse(texts['baseline']);base_functions={n.name:n for n in base.body if isinstance(n,ast.FunctionDef)}
old=base_functions['validate_used_numerics'];old_loop_start=next(i for i,n in enumerate(old.body) if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='callback_links' for t in n.targets))
old_reference_loop=old.body[old_loop_start:old_loop_start+2]
checks=[]
for label in ('C','A+C'):
 tree=ast.parse(texts[label]);funcs={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
 helper=funcs['validate_callback_scope_references'];used=funcs['validate_used_numerics']
 assert [dump(n) for n in helper.body[2:-1]]==[dump(n) for n in old_reference_loop]
 changed={name for name in base_functions if dump(base_functions[name])!=dump(funcs[name])}
 assert changed==({'validate_used_numerics'} if label=='C' else {'validate_deferrals','validate_used_numerics'})
 assert set(funcs)-set(base_functions)=={'validate_callback_scope_references'}
 assert dump(used.args)==dump(old.args)
 nonfunctions=lambda tree:[dump(n) for n in tree.body if not isinstance(n,ast.FunctionDef)]
 assert nonfunctions(tree)==nonfunctions(base)
 calls=[n for n in ast.walk(used) if isinstance(n,ast.Call) and isinstance(n.func,ast.Name)]
 assert sum(n.func.id=='validated_exact_requirements' for n in calls)==1
 assert sum(n.func.id=='validate_exact_case_links' for n in calls)==1
 # Original end-of-function complete-used checks and return are literally unchanged.
 assert [dump(n) for n in used.body[-3:]]==[dump(n) for n in old.body[-3:]]
 call_order=[]
 for statement in used.body:
  if isinstance(statement,ast.Assign) and isinstance(statement.value,ast.Call) and isinstance(statement.value.func,ast.Name):
   call_order.append(statement.value.func.id)
 assert call_order.index('validate_deferrals')<call_order.index('validate_callback_scope_references')<call_order.index('validated_exact_requirements')
 assert not any(isinstance(n,ast.Return) for n in used.body[:-1])
 checks.append({'candidate':label,'only_expected_functions_changed':True,'new_helper_reuses_exact_original_loop_AST':True,'imports_and_top_level_unchanged':True,'public_signature_unchanged':True,'authority_call_occurrences':1,'exact_case_link_call_occurrences':1,'complete_used_checks_and_return_unchanged':True,'no_successful_early_return':True})
assert (W/'candidate/combined-A-C.patch').read_bytes()==(W/'candidate/combined-A-B.patch').read_bytes()
report={'status':'passed','checks':checks,'diff_aliases_identical':True}
(W/'evidence/structure-checks.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
