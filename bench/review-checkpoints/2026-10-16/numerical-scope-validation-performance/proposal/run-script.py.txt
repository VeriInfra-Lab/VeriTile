from pathlib import Path
import hashlib
import json
import subprocess
import sys
import time

sys.dont_write_bytecode = True
w = Path(__file__).resolve().parent
p = w / sys.argv[1]
e = w / 'evidence/script-attempts'
e.mkdir(exist_ok=True)
tag = f'{len(list(e.glob("*.json"))) + 1:04}-{p.stem}'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
snap = e / (tag + '.py.txt')
snap.write_bytes(p.read_bytes())
log = e / (tag + '.log')
start = time.monotonic()
with log.open('w') as f:
    run = subprocess.run([sys.executable, '-B', str(p)], cwd=w, stdout=f, stderr=subprocess.STDOUT)
record = {'script': str(p.relative_to(w)), 'source_sha256': sha(snap),
          'exit_code': run.returncode, 'log': str(log.relative_to(w)),
          'log_sha256': sha(log), 'seconds': time.monotonic() - start}
(e / (tag + '.json')).write_text(json.dumps(record, indent=2) + '\n')
print(log.read_text())
print(json.dumps(record))
sys.exit(run.returncode)
