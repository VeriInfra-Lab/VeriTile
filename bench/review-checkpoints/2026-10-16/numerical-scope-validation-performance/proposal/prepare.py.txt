import ast,difflib,hashlib,json

original=P.read_text()
start=original.index('    callback_links = {}\n', original.index('def validate_used_numerics('))
end=original.index('    used_bindings, used_links = set(), set()\n', start)
loop=original[start:end]
helper='''def validate_callback_scope_references(callback_bindings, bindings):
    """Reject malformed or undeferred references before deriving authority.

    This establishes only local reference consistency. Required callback names,
    source-derived authority, exact case ports and complete usage remain checked
    by validate_used_numerics before any successful return.
    """
    if not isinstance(callback_bindings, dict):
        raise ValueError('unclosed numerical callbacks lack exact registered assumptions')

needle='def validate_used_numerics('
assert candidate.count(needle)==1
