from pathlib import Path
import ast,difflib,hashlib,json
W=Path(__file__).resolve().parent
A=Path('/tmp/veritile-standardization-20261015/numerical-validator-performance-proposal-f2759fe')
source=(A/'prepare_candidate.py').read_text()
changes=next(ast.literal_eval(n.value) for n in ast.parse(source).body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='changes' for t in n.targets))
baseline=(W/'evidence/baseline-check_original_program_scope.py.txt').read_text()
b=(W/'candidate/check_original_program_scope.py').read_text()
a=baseline;combined=b
for before,after in changes:
 assert a.count(before)==1 and combined.count(before)==1,before
 a=a.replace(before,after);combined=combined.replace(before,after)
def functions(text,names):
 return {n.name:ast.dump(n,include_attributes=False) for n in ast.parse(text).body if isinstance(n,ast.FunctionDef) and n.name in names}
assert functions(a,{'validate_deferrals','validate_used_numerics'})==functions((A/'candidate/row_index_functions.py').read_text(),{'validate_deferrals','validate_used_numerics'})
ast.parse(combined)
(W/'candidate/combined-check_original_program_scope.py').write_text(combined)
(W/'candidate/combined-A-B.patch').write_text(''.join(difflib.unified_diff(baseline.splitlines(True),combined.splitlines(True),fromfile='a/bench/check_original_program_scope.py',tofile='b/bench/check_original_program_scope.py')))
(W/'evidence/reconstructed-A.py.txt').write_text(a)
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
paths=[A/'receipt.json',A/'prepare_candidate.py',A/'candidate/row_index_functions.py',A/'candidate/row-indexes.patch']
(W/'evidence/combined-input-pins.json').write_text(json.dumps({str(p):sha(p) for p in paths},indent=2)+'\n')
for text in (b,combined):
 old=next(n for n in ast.parse(baseline).body if isinstance(n,ast.FunctionDef) and n.name=='validate_used_numerics')
 new=next(n for n in ast.parse(text).body if isinstance(n,ast.FunctionDef) and n.name=='validate_used_numerics')
 assert ast.dump(old.args)==ast.dump(new.args)
print(json.dumps({'standalone_B_sha256':sha(W/'candidate/check_original_program_scope.py'),'combined_AB_sha256':sha(W/'candidate/combined-check_original_program_scope.py'),'A_extracted_functions_exactly_match_frozen':True,'signatures_unchanged':True,'applied':False}))
