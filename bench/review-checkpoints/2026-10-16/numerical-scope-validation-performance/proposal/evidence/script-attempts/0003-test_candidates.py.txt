def functions(source,names):
 return '\n\n'.join(ast.get_source_segment(source,n) for n in ast.parse(source).body if isinstance(n,ast.FunctionDef) and n.name in names)
def assignment(source,name):
 return next(ast.literal_eval(n.value) for n in ast.parse(source).body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id==name for t in n.targets))

class AuthorityReached(Exception):pass

def make_runner(source,f,sentinel=False):
 counts=Counter();trace=[];ns=dict(shared)
 ns.update(BASE='bench/tritonbench_g/',DTYPES=assignment(sources['baseline'],'DTYPES'),RULE_STATES=assignment(sources['baseline'],'RULE_STATES'),FIELDS=assignment(exact,'FIELDS'))

 exec(functions(source,{'validate_deferrals','validate_callback_scope_references','validate_used_numerics'}),ns)
 def derive(root):
  counts['derive_authority']+=1;trace.append('derive_authority')
  if sentinel:raise AuthorityReached('full authority derivation sentinel')
  if f.get('authority_error'):raise ValueError('source authority rejected fixture')
  return deepcopy(f['authority'])
 def collect(root):
  counts['collect_legacy']+=1;trace.append('collect_legacy');return deepcopy(f['legacy'])
 def validate(queue,authority,legacy):
  counts['validate_exact_queue']+=1;trace.append('validate_exact_queue')
  return shared['validate_exact_queue'](queue,authority,legacy)
 real_gate=ns['validated_exact_requirements']
 def exact_gate(*args):
  counts['validated_exact_requirements']+=1;trace.append('validated_exact_requirements');return real_gate(*args)
 real_links=ns['validate_exact_case_links']
 def links(*args):
  counts['validate_exact_case_links']+=1;return real_links(*args)
 ns.update(derive_authority=derive,validate_exact_queue=validate,validated_exact_requirements=exact_gate,validate_exact_case_links=links)
 module=ModuleType('deferred_numerical_experiments');module.collect_legacy=collect
 def run():
  prior=sys.modules.get('deferred_numerical_experiments');sys.modules['deferred_numerical_experiments']=module
  before=canonical({k:f[k] for k in ('task','row','public','proof_cases','queue') if k not in ('public','proof_cases')})
  try:
   value=ns['validate_used_numerics'](f['task'],f['row'],f['public'],f['proof_cases'],f['queue'],root=f.get('root'))
   result={'ok':True,'return_matches_input':value==(f['row'].get('deferred_numerical_assumptions',[]),f['row'].get('callback_assumptions',{}))}
  except (ValueError,TypeError,KeyError,AuthorityReached) as error:
   result={'ok':False,'type':type(error).__name__,'error':str(error)}
  finally:
   if prior is None:sys.modules.pop('deferred_numerical_experiments',None)
   else:sys.modules['deferred_numerical_experiments']=prior
  assert before==canonical({k:f[k] for k in ('task','row','public','proof_cases','queue') if k not in ('public','proof_cases')})
  return result
 return run,counts,trace
