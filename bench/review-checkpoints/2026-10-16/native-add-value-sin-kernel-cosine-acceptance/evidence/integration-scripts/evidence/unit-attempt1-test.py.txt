"""Focused mutation barriers for the future integration stages."""
from pathlib import Path
from unittest.mock import patch
import gzip
import tempfile
import unittest
import common as m

class IntegrationBarriers(unittest.TestCase):
 def test_actual_full_base_is_required(self):
  for value in [None,'','HEAD','97e11a9','f'*39]:
   with self.subTest(value=value),self.assertRaisesRegex(ValueError,'full actual'):m.explicit_base(Path('/tmp'),value)
 def test_wrong_and_dirty_base_rejected(self):
  value='a'*40
  with patch.object(m,'git',return_value=b'b'*40),self.assertRaisesRegex(ValueError,'HEAD differs'):m.explicit_base(Path('/tmp'),value)
  with patch.object(m,'git',side_effect=[value.encode(),b'commit',b' M bench/file.py']),self.assertRaisesRegex(ValueError,'clean committed'):m.explicit_base(Path('/tmp'),value,clean=True)
  with patch.object(m,'git',side_effect=[value.encode(),b'commit',b'']):m.explicit_base(Path('/tmp'),value,clean=True)
 def test_external_output_follows_symlinks(self):
  with tempfile.TemporaryDirectory() as temp:
   root=Path(temp);repo=root/'repo';repo.mkdir();alias=root/'alias';alias.symlink_to(repo,target_is_directory=True)
   for path in [repo,repo/'run',alias/'run']:
    with self.subTest(path=path),self.assertRaisesRegex(ValueError,'outside'):m.external_output(repo,path)
   self.assertEqual(m.external_output(repo,root/'run'),root/'run');self.assertFalse((root/'run').exists())
 def test_gzip_active_path_rejects_plain_and_dangling_duplicates(self):
  with tempfile.TemporaryDirectory() as temp:
   root=Path(temp);queue=root/m.QUEUE;queue.parent.mkdir(parents=True);queue.write_bytes(gzip.compress(b'{}',mtime=0));m.active_gzip(root)
   for rel in [m.QUEUE.removesuffix('.gz'),'bench/original_program_scope.json']:
    plain=root/rel;plain.write_bytes(b'{}')
    with self.assertRaisesRegex(ValueError,'Competing'):m.active_gzip(root)
    plain.unlink();plain.symlink_to(root/'absent')
    with self.assertRaisesRegex(ValueError,'Competing'):m.active_gzip(root)
    plain.unlink()
 def test_unsafe_artifact_paths_are_rejected(self):
  for path in ['/tmp/x','../x','a/../x','a\\b','./a','a//b','','.']:
   with self.subTest(path=path),self.assertRaisesRegex(ValueError,'Unsafe'):m.relative(path)
 def test_script_receipt_closure_is_copied_and_reverified(self):
  with tempfile.TemporaryDirectory() as temp:
   root=Path(temp);source=root/'source';source.mkdir();(source/'a.py').write_bytes(b'# source\n');(source/'nested').mkdir();(source/'nested/b.json').write_bytes(b'{}\n')
   receipt={'frozen':True,'artifacts':{name:m.digest(source/name) for name in ['a.py','nested/b.json']}};m.write(source/'candidate-receipt.json',receipt);expected=m.digest(source/'candidate-receipt.json');dest=root/'copy'
   with patch.object(m,'HERE',source):m.copy_scripts(dest,expected)
   self.assertEqual(m.verify_scripts(dest,expected),receipt)
   (dest/'nested/b.json').write_bytes(b'changed')
   with self.assertRaisesRegex(ValueError,'Changed/missing'):m.verify_scripts(dest,expected)
 def test_storage_receipt_chain_pins_sources_snapshot_plan_and_scripts(self):
  with tempfile.TemporaryDirectory() as temp:
   root=Path(temp);repo=root/'repo';script=root/'script';script.mkdir();m.write(script/'future-protected-pins.json',{})
   queue=repo/m.QUEUE;queue.parent.mkdir(parents=True);queue.write_bytes(gzip.compress(b'{}',mtime=0))
   policy=repo/m.PRIMARY_POLICY;policy.parent.mkdir(parents=True);policy.write_bytes(b'policy\n');policy_hash=m.digest(policy)
   cp=repo/m.STORAGE_CHECKPOINT;scriptcopy=cp/'evidence/integration-scripts';scriptcopy.mkdir(parents=True);(scriptcopy/'integration.py').write_bytes(b'# original\n')
   m.write(scriptcopy/'receipt.json',{'artifacts':{'integration.py':m.digest(scriptcopy/'integration.py')}})
   m.write(cp/'plan.json',{'scripts_receipt_sha256':m.digest(scriptcopy/'receipt.json')})
   manifest={'migration_plan_sha256':m.digest(cp/'plan.json'),'root_input_sha256':{m.QUEUE:m.digest(queue)},'documents':{},'legacy_status_counts':{'verified':1,'verified_with_deferred_numerics':7,'pending':175}}
   snapshot=cp/'snapshot/manifest.json.gz';m.write(snapshot,manifest)
   report={'status':'passed','queue_bytes_and_all_legacy_rows_preserved':True,'legacy_scope_result_unchanged':True,'coherent_snapshot':{'manifest':snapshot.relative_to(repo).as_posix(),'sha256':m.digest(snapshot)}}
   check=cp/'validation/check.json';m.write(check,report);expected=m.digest(check)
   with patch.object(m,'HERE',script),patch.object(m,'PRIMARY_POLICY_SHA256',policy_hash):
    self.assertEqual(m.verify_storage_base(repo,expected),report)
    for path in [queue,snapshot,cp/'plan.json',scriptcopy/'receipt.json',scriptcopy/'integration.py']:
     with self.subTest(path=path):
      before=path.read_bytes();path.write_bytes(before+b'changed')
      with self.assertRaises((ValueError,OSError,EOFError)):m.verify_storage_base(repo,expected)
      path.write_bytes(before)
 def test_stage_paths_do_not_write_into_frozen_script_package(self):
  import ast
  tree=ast.parse((m.HERE/'stage.py').read_text())
  # Staging's pathspec must be a fresh external temporary file; no write targets HERE.
  calls=[n for n in ast.walk(tree) if isinstance(n,ast.Call)]
  self.assertTrue(any(isinstance(n.func,ast.Attribute) and n.func.attr=='NamedTemporaryFile' for n in calls))
  self.assertFalse(any(isinstance(n.func,ast.Attribute) and n.func.attr in {'write_text','write_bytes'} and isinstance(n.func.value,ast.Name) and n.func.value.id=='pathspec' for n in calls))

if __name__=='__main__':unittest.main()
