import bench.tritonbench_g.vector_addition.Proofs.NativeArithmetic
import bench.tritonbench_g.cosine_compute.Proofs.NativeHostMath

namespace CosineNative
set_option maxHeartbeats 6000000
set_option maxRecDepth 24000
open VeriTile.Bench.TritonBenchG.VectorAddition
open DTypeScope TypedHost Finite
abbrev hostExtent := Native.hostExtent
abbrev grid := HostMath.grid
def gridSupported (n : Nat) : Prop := grid n≤2^31-1
abbrev bound := Native.bound
def offset (n pid lane : Nat) : BitVec 32 :=
  BitVec.ofNat 32 pid * BitVec.ofNat 32 (HostMath.block n) + BitVec.ofNat 32 lane
def mask (n pid lane : Nat) : Bool := (offset n pid lane).slt (bound n)
def pointer (data width n pid lane : Nat) : Nat :=
  (BitVec.ofNat 64 data + (offset n pid lane).signExtend 64 * BitVec.ofNat 64 width).toNat
def activeTile (n pid : Nat) : List Nat := (List.range (HostMath.block n)).filter (mask n pid)
abbrev Buffer := Native.Buffer
abbrev dtype (d : Storage) : DType := wideDType d
structure Layout where
  x : Buffer
  output : Buffer
  device : Nat
def Layout.Fresh (p : Layout) : Prop := p.x.Disjoint p.output

theorem all_masks_false_at_two31 (pid lane : Nat) : mask (2^31) pid lane=Bool.false := by
  have lower : -(2^31 : Int)≤(offset (2^31) pid lane).toInt := by
    simpa using BitVec.le_toInt (offset (2^31) pid lane)
  rw [mask,BitVec.slt_eq_decide,Native.bound_minimum]
  exact decide_eq_false (not_lt.mpr lower)
theorem active_tile_empty_at_two31 (pid : Nat) : activeTile (2^31) pid=[] := by
  apply List.filter_eq_nil_iff.mpr
  intro lane _
  rw [all_masks_false_at_two31]
  decide
theorem counterexample_grid : grid (2^31)=32768 ∧ gridSupported (2^31) ∧ hostExtent (2^31) := by
  unfold gridSupported hostExtent Native.hostExtent
  decide +kernel
theorem wrapped_negative_lane :
    (offset (2^31+1) 32768 0).toInt=-(2^31 : Int) ∧
      mask (2^31+1) 32768 0=Bool.true := by decide +kernel
theorem wrapped_zero_lane : offset (2^32+1) 32768 0=0 ∧
    mask (2^32+1) 32768 0=Bool.true := by decide +kernel

#axiomsClean all_masks_false_at_two31
#axiomsClean active_tile_empty_at_two31
#axiomsClean counterexample_grid
#axiomsClean wrapped_negative_lane
#axiomsClean wrapped_zero_lane
#guard_msgs (drop info) in
#auditModuleAxioms
end CosineNative
