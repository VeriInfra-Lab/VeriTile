import bench.tritonbench_g.sin_kernel.Proofs.NativeArithmetic
namespace SinKernelNative
open VeriTile.Bench.TritonBenchG.VectorAddition
open DTypeScope
inductive InputType where
  | admitted (storage : Storage)
  | boolean | int8 | uint8 | int16 | uint16 | int32 | uint32 | int64 | uint64
  | fp16 | bf16 | f8e4fn | f8e5 | fnuzE4 | fnuzE5
  | complex32 | complex64 | complex128 | e8m0 | packedFP4
  deriving DecidableEq
def InputType.name : InputType → String
  | .admitted .fp32 => "float32" | .admitted .fp64 => "float64"
  | .boolean => "bool" | .int8 => "int8" | .uint8 => "uint8"
  | .int16 => "int16" | .uint16 => "uint16" | .int32 => "int32" | .uint32 => "uint32"
  | .int64 => "int64" | .uint64 => "uint64" | .fp16 => "float16" | .bf16 => "bfloat16"
  | .f8e4fn => "float8_e4m3fn" | .f8e5 => "float8_e5m2"
  | .fnuzE4 => "float8_e4m3fnuz" | .fnuzE5 => "float8_e5m2fnuz"
  | .complex32 => "complex32" | .complex64 => "complex64" | .complex128 => "complex128"
  | .e8m0 => "float8_e8m0fnu" | .packedFP4 => "float4_e2m1fn_x2"
def inputTypes : List InputType :=
  [.boolean,.int8,.uint8,.int16,.uint16,.int32,.uint32,.int64,.uint64,.fp16,.bf16,
   .admitted .fp32,.admitted .fp64,.f8e4fn,.f8e5,.fnuzE4,.fnuzE5,.complex32,.complex64,.complex128,.e8m0,.packedFP4]
def parseType (name : String) : Option InputType := inputTypes.find? (fun k => k.name==name)
theorem parse_name (k : InputType) : parseType k.name=some k := by
  cases k with
  | admitted d => cases d <;> decide +kernel
  | _ => decide +kernel
inductive SetupReject where | binder | sineFrontend
  deriving DecidableEq, Repr
def admission : InputType → Except SetupReject Storage
  | .admitted d => .ok d
  | .complex32 | .complex64 | .complex128 | .e8m0 | .packedFP4 => .error .binder
  | _ => .error .sineFrontend
def outcomeStage : Except SetupReject Storage → Nat
  | .ok _ => 2 | .error .sineFrontend => 1 | .error .binder => 0
def stageCount (s : Nat) : Nat := (inputTypes.filter (fun t => outcomeStage (admission t)==s)).length
theorem admission_counts : inputTypes.length=22 ∧ stageCount 0=5 ∧ stageCount 1=15 ∧ stageCount 2=2 := by decide +kernel
#axiomsClean parse_name
#axiomsClean admission_counts
#guard_msgs (drop info) in
#auditModuleAxioms
end SinKernelNative
