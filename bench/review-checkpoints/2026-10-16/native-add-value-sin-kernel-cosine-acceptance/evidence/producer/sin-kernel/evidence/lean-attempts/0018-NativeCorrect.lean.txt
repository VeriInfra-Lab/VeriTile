/-- Actual metadata queries and access permissions of the trusted Torch and
device memory boundary. The output retains preserve-format strides. -/
structure Metadata where
  torchStorageBase : Tensor → Nat
  torchDataPointer : Tensor → Nat
  kernelArgumentPointer : Tensor → Nat
  buffer : Tensor → Buffer
  emptyLikeStrides : Tensor → List Int
  currentDevice : Nat
  schedule : (n : Nat) → Schedule n
def layout (metadata : Metadata) (x output : Tensor) : Layout :=
  ⟨metadata.buffer x,metadata.buffer output,metadata.currentDevice⟩

/-- Torch metadata and the actual successful extractPointer result are kept
distinct. Buffer.base is the allocation mapping in the launch address space;
it need not equal torchStorageBase. No owner-device equality or UVA identity
is inferred. Affine offsets within that mapped allocation are ordinary trusted
memory metadata; the kernel uses only kernelArgumentPointer. On branches that
never consume a successful extraction, these total metadata fields may carry
unused ordinary affine witnesses; their equalities do not assert native
extraction success. On successful packing they denote the actual extracted
pointer and mapped allocation. The setup outcome is separate. -/
structure TensorBinding (metadata : Metadata) (t : Tensor) (d : DType) : Prop where
  dtype : t.dtype=dtypeName d
  numel : t.numel=t.shape.prod
  range : hostExtent t.numel
  offset : 0≤t.storageOffset
  owner : (metadata.buffer t).device=t.deviceIndex
  pointerWidth : (metadata.buffer t).data<2^64
  extent : (metadata.buffer t).base+(metadata.buffer t).capacity≤2^64
  torchData : metadata.torchDataPointer t=if t.numel=0 then 0 else
    metadata.torchStorageBase t+t.storageOffset.toNat*bytes d
  extractedPointer : (metadata.buffer t).data=metadata.kernelArgumentPointer t
  mappedData : (metadata.buffer t).data=if t.numel=0 then 0 else
    (metadata.buffer t).base+t.storageOffset.toNat*bytes d

theorem uses_extracted_pointer (metadata : Metadata) (t : Tensor) (d : DType)
    (binding : TensorBinding metadata t d) : (metadata.buffer t).data=metadata.kernelArgumentPointer t :=
  binding.extractedPointer

/-- Ordinary allocation metadata and freshness, required unconditionally even
when every kernel mask is false. There is no index bound,
same-device condition, contiguous-layout assumption, or requested-span bound. -/
structure AllocationBinding (metadata : Metadata) (x output : Tensor) : Prop where
  returned : ReturnedMetadata x output
  outputStrides : output.strides=metadata.emptyLikeStrides x
  hostNumelX : x.numel=x.shape.prod
  hostRangeX : hostExtent x.numel
  outputOffset : output.storageOffset=0
  outputData : (metadata.buffer output).data=if output.numel=0 then 0 else (metadata.buffer output).base
  xBinding : ∀ d, x.dtype=dtypeName d → TensorBinding metadata x d
  outputBinding : ∀ d, output.dtype=dtypeName d → TensorBinding metadata output d
  fresh : (layout metadata x output).Fresh

inductive Event where
  | emptyLike (input : RegionName)
  | numel (input : RegionName) (n : Nat)
  | launch (n block grid : Nat)
  | returnTensor (output : RegionName)
  deriving DecidableEq, Repr
inductive SetupStage where | context | compilation | pointerPacking | launch
  deriving DecidableEq, Repr
abbrev Environment := SetupStage → Option String
abbrev Allocator (M : Type) := Tensor → M → Except String (Tensor×M)
inductive Failure where
  | allocation (detail : String)
  | setup (reason : SetupReject)
  | primitive (stage : SetupStage) (detail : String)
  | gridLimit (requested : Nat)
  deriving DecidableEq, Repr
inductive Outcome (M : Type) where
  | failed (trace : List Event) (why : Failure)
  | observed (trace : List Event) (issue : Issue)
  | returned (trace : List Event) (tensor : Tensor) (memory : M)

def checkedTrace (x _output : Tensor) : List Event :=
  [.numel x.region x.numel,.emptyLike x.region]
def attemptedTrace (x output : Tensor) : List Event :=
  checkedTrace x output ++ [.launch x.numel 1024 (grid x.numel)]

def successTrace (x output : Tensor) : List Event := attemptedTrace x output ++ [.returnTensor output.region]

/-- This environment represents only ordinary fallible setup primitives. It
has no access to input values and cannot provide a kernel result or execution.
All admitted lanes execute below after setup succeeds. -/
def executeAllocated {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (kind : InputType)
    (x output : Tensor) (allocated : M) : Outcome M :=
  match env .context with
    | some detail => .failed (attemptedTrace x output) (.primitive .context detail)
    | none => match admission kind with
      | .error reason => .failed (attemptedTrace x output) (.setup reason)
      | .ok d => match env .compilation with
        | some detail => .failed (attemptedTrace x output) (.primitive .compilation detail)
        | none =>
          if ¬gridSupported x.numel then .failed (attemptedTrace x output) (.gridLimit (grid x.numel))
          else match env .pointerPacking with
            | some detail => .failed (attemptedTrace x output) (.primitive .pointerPacking detail)
            | none => match env .launch with
              | some detail => .failed (attemptedTrace x output) (.primitive .launch detail)
              | none => match executeLaunch d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated with
                | .error issue => .observed (attemptedTrace x output) issue
                | .ok memory => .returned (successTrace x output) output memory

def executeHost {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (kind : InputType)
    (x : Tensor) (initial : M) : Outcome M :=
  match alloc x initial with
  | .error detail => .failed [.numel x.region x.numel,.emptyLike x.region] (.allocation detail)
  | .ok (output,allocated) => executeAllocated ops b env metadata kind x output allocated

/-- Source first-stage setup failure facts. These contain no access safety or
defined-value predicate and do not classify model observations as native errors. -/
def SetupFailure (env : Environment) (kind : InputType) (n : Nat) (why : Failure) : Prop :=
  (∃ detail, env .context=some detail ∧ why=.primitive .context detail) ∨
  (env .context=none ∧ ∃ reason, admission kind=.error reason ∧ why=.setup reason) ∨
  (env .context=none ∧ ∃ d, admission kind=.ok d ∧
    ((∃ detail, env .compilation=some detail ∧ why=.primitive .compilation detail) ∨
     (env .compilation=none ∧ ¬gridSupported n ∧ why=.gridLimit (grid n)) ∨
     (env .compilation=none ∧ gridSupported n ∧ ∃ detail, env .pointerPacking=some detail ∧ why=.primitive .pointerPacking detail) ∨
     (env .compilation=none ∧ env .pointerPacking=none ∧ gridSupported n ∧
       ∃ detail, env .launch=some detail ∧ why=.primitive .launch detail)))
def SetupReady (env : Environment) (kind : InputType) (n : Nat) (d : Storage) : Prop :=
  (∀ stage, env stage=none) ∧ admission kind=.ok d ∧ gridSupported n

structure ReturnedSpec {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (metadata : Metadata) (x output : Tensor) (allocated final : M) : Prop where
  allocation : AllocationBinding metadata x output
  writes : ∃ writes, launchPlan d ops b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated=.ok writes ∧
    final=applyWrites ops (dtype d) writes allocated ∧
    (∀ address, (∀ w∈writes, ¬w.Touches (dtype d) address) → ops.byte final address=ops.byte allocated address) ∧
    (∀ address, ops.byte final address=finalByte ops (dtype d) writes allocated address)
  exterior : OutsideOutput ops (layout metadata x output) final allocated

def AllocatedSpec {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (kind : InputType)
    (x output : Tensor) (allocated : M) : Outcome M → Prop
  | .failed trace why =>
      trace=attemptedTrace x output ∧ SetupFailure env kind x.numel why
  | .observed trace issue => trace=attemptedTrace x output ∧
      ∃ d, SetupReady env kind x.numel d ∧
        launchPlan d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated=.error issue
  | .returned trace tensor final => trace=successTrace x output ∧ tensor=output ∧
      ∃ d, SetupReady env kind x.numel d ∧ ReturnedSpec d (ops d) b metadata x output allocated final

theorem allocated_all_outcomes {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (kind : InputType) (x output : Tensor) (allocated : M)
    (binding : AllocationBinding metadata x output) :
    AllocatedSpec ops b env metadata kind x output allocated
      (executeAllocated ops b env metadata kind x output allocated) := by
  unfold executeAllocated
  cases he : env .context with
  | some detail => simp [AllocatedSpec,SetupFailure,he]
  | none =>
    cases ha : admission kind with
    | error reason => simp [AllocatedSpec,SetupFailure,he,ha]
    | ok d =>
      cases hcomp : env .compilation with
      | some detail => simp [AllocatedSpec,SetupFailure,he,ha,hcomp]
      | none =>
        simp only []
        by_cases hg : gridSupported x.numel
        · rw [if_neg (not_not.mpr hg)]
          cases hpack : env .pointerPacking with
          | some detail => simp [AllocatedSpec,SetupFailure,he,ha,hcomp,hg,hpack]
          | none =>
            cases hl : env .launch with
            | some detail => simp [AllocatedSpec,SetupFailure,he,ha,hcomp,hpack,hg,hl]
            | none =>
              have ready : SetupReady env kind x.numel d :=
                ⟨(by intro stage; cases stage <;> assumption),ha,hg⟩
              have corr := launch_correspondence d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated binding.fresh
              cases ht : launchPlan d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated with
              | error issue =>
                simp only [ht,Except.map] at corr
                rw [corr]
                exact ⟨rfl,d,ready,ht⟩
              | ok writes =>
                simp only [ht,Except.map] at corr
                rw [corr]
                refine ⟨rfl,rfl,d,ready,binding,⟨writes,ht,rfl,?_,?_⟩,?_⟩
                · exact fun address outside => writes_byte_frame (ops d) (dtype d) writes allocated address outside
                · exact fun address => writes_byte_result (ops d) (dtype d) writes allocated address
                · exact writes_outside (ops d) (layout metadata x output) (dtype d) writes allocated
                    (launch_write_bounds d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated writes ht)
        · simp [hg,AllocatedSpec,SetupFailure,he,ha,hcomp]

def HostSpec {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (kind : InputType)
    (x : Tensor) (initial : M) (outcome : Outcome M) : Prop :=
  (∀ d, WordMemory d (ops d)) ∧ x.dtype=kind.name ∧ (
  (∃ detail, alloc x initial=.error detail ∧ outcome=.failed [.numel x.region x.numel,.emptyLike x.region] (.allocation detail)) ∨
  (∃ output allocated, alloc x initial=.ok (output,allocated) ∧
    (∀ d, OutsideOutput (ops d) (layout metadata x output) allocated initial) ∧
    AllocatedSpec ops b env metadata kind x output allocated outcome))

/-- Full source master: original numel query, allocation, all22 signatures,
ordinary setup failures, exact finite requests/observations, and actual return.
The only allocation contract is ordinary metadata/freshness, at every extent. -/
specification sin_kernel_native_source_outcomes {M : Type} (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (raw : ∀ d, WordMemory d (ops d))
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (kind : InputType)
    (x : Tensor) (initial : M) (hx : x.dtype=kind.name)
    (contract : ∀ output allocated, alloc x initial=.ok (output,allocated) →
      AllocationBinding metadata x output ∧
      (∀ d, OutsideOutput (ops d) (layout metadata x output) allocated initial)) :
    HostSpec ops b alloc env metadata kind x initial (executeHost ops b alloc env metadata kind x initial) := by
  refine ⟨raw,hx,?_⟩
  cases ha : alloc x initial with
  | error detail => exact Or.inl ⟨detail,rfl,by simp [executeHost,ha]⟩
  | ok result =>
      rcases result with ⟨output,allocated⟩
      refine Or.inr ⟨output,allocated,rfl,(contract output allocated ha).2,?_⟩
      simpa only [executeHost,ha] using allocated_all_outcomes ops b env metadata kind x output allocated (contract output allocated ha).1

theorem allocated_return_of_launch {M : Type}
    (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (metadata : Metadata) (kind : InputType) (d : Storage)
    (x output : Tensor) (allocated : M)
    (hr : admission kind=.ok d)
    (hg : gridSupported x.numel)
    (hk : executeLaunch d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated=.ok allocated) :
    executeAllocated ops b (fun _ => none) metadata kind x output allocated=
      .returned (successTrace x output) output allocated := by
  simp only [executeAllocated,hr]
  rw [if_neg (not_not.mpr hg)]
  simp only [hk]

/-- At numel2^31 all masks are false. This source return preserves the exact
allocated tensor and its memory, with no input value/access/no-wrap premise. -/
theorem sin_kernel_two31_return {M : Type}
    (ops : (d : Storage) → Memory M (Word d)) (b : Numerics)
    (alloc : Allocator M) (metadata : Metadata) (kind : InputType) (d : Storage)
    (x output : Tensor) (initial allocated : M)
    (ha : alloc x initial=.ok (output,allocated))
    (hr : admission kind=.ok d) (hn : x.numel=2^31) :
    executeHost ops b alloc (fun _ => none) metadata kind x initial=
      .returned (successTrace x output) output allocated := by
  have hg : gridSupported x.numel := by simpa only [hn] using counterexample_grid.2.1
  have hk : executeLaunch d (ops d) b (layout metadata x output) x.numel (metadata.schedule x.numel) allocated=.ok allocated := by
    rw [hn]
    exact no_access_returned_memory d (ops d) b (layout metadata x output) (metadata.schedule (2^31)) allocated
  have ready := allocated_return_of_launch ops b metadata kind d x output allocated hr hg hk
  simpa only [executeHost,ha] using ready

/-- launchKernel parses the signed-int grid tuple before extracting any tensor
pointer. Even a configured pointer failure cannot supersede this first error. -/
theorem grid_parse_precedes_pointer_packing {M : Type}
    (ops : (d : Storage) → Memory M (Word d)) (b : Numerics) (env : Environment)
    (metadata : Metadata) (kind : InputType) (d : Storage)
    (x output : Tensor) (allocated : M)
    (he : env .context=none)
    (ha : admission kind=.ok d) (hcomp : env .compilation=none)
    (hg : ¬gridSupported x.numel) :
    executeAllocated ops b env metadata kind x output allocated=
      .failed (attemptedTrace x output) (.gridLimit (grid x.numel)) := by
  simp [executeAllocated,he,ha,hcomp,hg]
