def gridPlan {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (snapshot : M) : List Nat → List Nat → Except Issue (List (Write (Word d))×List Nat)
  | [],seen => .ok ([],seen)
  | pid::rest,seen => do
      let (head,owners) ← programPlan d ops b p n snapshot pid seen
      let (tail,finalOwners) ← gridPlan d ops b p n snapshot rest owners
      pure (head++tail,finalOwners)

def executeGrid {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) : List Nat → State M → Except Issue (State M)
  | [],state => .ok state
  | pid::rest,state => do
      let next ← executeProgram d ops b p n pid state
      executeGrid d ops b p n rest next

theorem grid_correspondence {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (snapshot : M) (fresh : p.Fresh) (pids : List Nat)
    (state : State M) (same : OutsideOutput ops p state.memory snapshot) :
    executeGrid d ops b p n pids state=(gridPlan d ops b p n snapshot pids state.owners).map
      (fun plan => ⟨applyWrites ops (dtype d) plan.1 state.memory,plan.2⟩) := by
  induction pids generalizing state with
  | nil => rfl
  | cons pid rest ih =>
      simp only [executeGrid,gridPlan,program_correspondence]
      rw [program_plan_frame d ops b p n state.memory snapshot same fresh]
      cases hh : programPlan d ops b p n snapshot pid state.owners with
      | error issue => rfl
      | ok head =>
          rcases head with ⟨head,owners⟩
          simp only [Except.map,bind_ok]
          have hb := program_write_bounds d ops b p n snapshot pid state.owners head owners hh
          have hs : OutsideOutput ops p (applyWrites ops (dtype d) head state.memory) snapshot :=
            outside_trans ops p _ _ _ (writes_outside ops p (dtype d) head state.memory hb) same
          rw [ih ⟨applyWrites ops (dtype d) head state.memory,owners⟩ hs]
          cases gridPlan d ops b p n snapshot rest owners with
          | error issue => rfl
          | ok tail =>
              rcases tail with ⟨tail,finalOwners⟩
              simp only [Except.map,bind_ok,pure_ok,writes_append]

theorem grid_write_bounds {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (snapshot : M) (pids : List Nat) (seen : List Nat)
    (writes : List (Write (Word d))) (owners : List Nat)
    (h : gridPlan d ops b p n snapshot pids seen=.ok (writes,owners)) :
    ∀ w∈writes, p.output.Inside w.address (bytes (dtype d)) := by
  induction pids generalizing seen writes owners with
  | nil => cases h; simp
  | cons pid rest ih =>
      simp only [gridPlan] at h
      cases hh : programPlan d ops b p n snapshot pid seen with
      | error issue => simp [hh] at h
      | ok head =>
          rcases head with ⟨head,nextOwners⟩
          cases ht : gridPlan d ops b p n snapshot rest nextOwners with
          | error issue => simp [hh,ht] at h
          | ok tail =>
              rcases tail with ⟨tail,finalOwners⟩
              simp [hh,ht] at h
              rcases h with ⟨rfl,rfl⟩
              intro w hw
              rcases List.mem_append.1 hw with hw|hw
              · exact program_write_bounds d ops b p n snapshot pid seen head nextOwners hh w hw
              · exact ih _ _ _ ht w hw

structure Schedule (n : Nat) where
  pids : List Nat
  complete : pids.Perm (List.range (grid n))

def ascendingSchedule (n : Nat) : Schedule n := ⟨List.range (grid n),List.Perm.refl _⟩

def launchPlan {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (schedule : Schedule n) (snapshot : M) : Except Issue (List (Write (Word d))) :=
  (gridPlan d ops b p n snapshot schedule.pids []).map Prod.fst
def executeLaunch {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (schedule : Schedule n) (memory : M) : Except Issue M :=
  (executeGrid d ops b p n schedule.pids ⟨memory,[]⟩).map State.memory

theorem launch_correspondence {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (schedule : Schedule n) (memory : M) (fresh : p.Fresh) :
    executeLaunch d ops b p n schedule memory=(launchPlan d ops b p n schedule memory).map
      (fun writes => applyWrites ops (dtype d) writes memory) := by
  unfold executeLaunch launchPlan
  rw [grid_correspondence d ops b p n memory fresh _ _ (outside_refl ops p memory)]
  cases gridPlan d ops b p n memory schedule.pids [] <;> rfl

theorem launch_write_bounds {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (schedule : Schedule n) (memory : M) (writes : List (Write (Word d)))
    (h : launchPlan d ops b p n schedule memory=.ok writes) :
    ∀ w∈writes, p.output.Inside w.address (bytes (dtype d)) := by
  unfold launchPlan at h
  cases hg : gridPlan d ops b p n memory schedule.pids [] with
  | error issue => simp [hg,Except.map] at h
  | ok plan =>
      simp only [hg,Except.map,Except.ok.injEq] at h
      subst writes
      exact grid_write_bounds d ops b p n memory _ _ _ _ hg

/-- No safety/definedness predicate restricts the master launch. Every finite
request either yields a recorded observation or an independent write formula. -/
theorem launch_all_outcomes {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (n : Nat) (schedule : Schedule n) (memory : M) (fresh : p.Fresh) :
    (∃ issue, launchPlan d ops b p n schedule memory=.error issue ∧ executeLaunch d ops b p n schedule memory=.error issue) ∨
    (∃ writes, launchPlan d ops b p n schedule memory=.ok writes ∧
      executeLaunch d ops b p n schedule memory=.ok (applyWrites ops (dtype d) writes memory) ∧
      OutsideOutput ops p (applyWrites ops (dtype d) writes memory) memory ∧
      (∀ address, (∀ w∈writes, ¬w.Touches (dtype d) address) →
        ops.byte (applyWrites ops (dtype d) writes memory) address=ops.byte memory address)) := by
  have hc := launch_correspondence d ops b p n schedule memory fresh
  cases ht : launchPlan d ops b p n schedule memory with
  | error issue => exact Or.inl ⟨issue,rfl,by simpa [ht,Except.map] using hc⟩
  | ok writes =>
      exact Or.inr ⟨writes,rfl,by simpa [ht,Except.map] using hc,
        writes_outside ops p (dtype d) writes memory (launch_write_bounds d ops b p n schedule memory writes ht),
        fun address outside => writes_byte_frame ops (dtype d) writes memory address outside⟩

theorem no_access_grid {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (memory : M) (pids : List Nat) (seen : List Nat) :
    gridPlan d ops b p (2^31) memory pids seen=.ok ([],seen) := by
  induction pids generalizing seen with
  | nil => rfl
  | cons pid rest ih => simp only [gridPlan,no_access_program,bind_ok,ih,pure_ok,List.nil_append]
theorem no_access_returned_memory {M : Type} (d : Storage) (ops : Memory M (Word d)) (b : Numerics)
    (p : Layout) (schedule : Schedule (2^31)) (memory : M) : executeLaunch d ops b p (2^31) schedule memory=.ok memory := by
  have programs : ∀ pid (state : State M), executeProgram d ops b p (2^31) pid state=.ok state := by
    intro pid state
    rw [program_correspondence,no_access_program]
    rfl
  have grids : ∀ pids (state : State M), executeGrid d ops b p (2^31) pids state=.ok state := by
    intro pids
    induction pids with
    | nil => intro state; rfl
    | cons pid rest ih => intro state; simp only [executeGrid,programs,bind_ok,ih]
  rw [executeLaunch,grids]
  rfl
