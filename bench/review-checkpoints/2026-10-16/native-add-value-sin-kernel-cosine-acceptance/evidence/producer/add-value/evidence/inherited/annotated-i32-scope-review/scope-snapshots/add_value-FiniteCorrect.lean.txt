structure FullSuccess {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (d : Storage) (context : Context) (x output : Tensor) (metadata : TensorMetadata)
    (initial allocated final : M) : Prop where
  host : puzzle1 b rt ops undef views x initial=⟨successTrace x output,.ok output final⟩
  kernel : sourceGrid b d (ops d) views (undef d) x.region output.region output.numel allocated=.ok final
  allocation : AllocationContract ops x output views metadata initial allocated
  values : GridResult b d (ops d) (views x.region) (views output.region) output.numel allocated final
  tail : ∀ i, output.numel ≤ i → readView (ops d) final (views output.region) i=readView (ops d) allocated (views output.region) i
  safety : 0<output.numel → LaunchSafety (views x.region) (views output.region) output.numel
  permissions : 0<output.numel → MemoryApplicable (ops d) (views x.region) (views output.region) output.numel
  currentDevice : 0<output.numel → (ops d).launchDevice=context.currentDevice
  schedules : ∀ jobs, (∀ i, i∈jobs ↔ i<output.numel) → ∃ final,
    executeLive (ops d) (fun a ↦ (source d).eval b a argument) (views x.region) (views output.region) jobs allocated=.ok final ∧
    ∀ i, i<output.numel → some (readView (ops d) final (views output.region) i)=result b d (readView (ops d) allocated (views x.region) i) argument

theorem host_success {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (d : Storage) (context : Context) (x output : Tensor) (initial allocated final : M)
    (dtype : x.dtype=(InputType.admitted d).name)
    (ha : rt.emptyLike x initial=.ok output allocated) (hc : cudaGuard x output=Bool.true)
    (hp : rt.prepare x output output.numel=.ok context)
    (hn : rt.nativeReady context d x output output.numel=.ok ())
    (hk : sourceGrid b d (ops d) views (undef d) x.region output.region output.numel allocated=.ok final) :
    puzzle1 b rt ops undef views x initial=⟨successTrace x output,.ok output final⟩ := by
  simp [puzzle1,ha,hc,launch,hp,dtype,parse_name,dispatch,hn,hk]

theorem add_value_finite_success {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (d : Storage) (context : Context) (x output : Tensor) (metadata : TensorMetadata)
    (initial allocated : M) (dtype : x.dtype=(InputType.admitted d).name)
    (ha : rt.emptyLike x initial=.ok output allocated) (hc : cudaGuard x output=Bool.true)
    (hp : rt.prepare x output output.numel=.ok context)
    (hn : rt.nativeReady context d x output output.numel=.ok ())
    (allocation : AllocationContract ops x output views metadata initial allocated)
    (applicable : 0<output.numel → SafeAPI x output views ∧
      MemoryApplicable (ops d) (views x.region) (views output.region) output.numel ∧
      (ops d).launchDevice=context.currentDevice ∧
      ValuesDefined b d (ops d) (views x.region) output.numel allocated) :
    ∃ final, FullSuccess b rt ops undef views d context x output metadata initial allocated final := by
  by_cases hz : output.numel=0
  · have executed : sourceGrid b d (ops d) views (undef d) x.region output.region output.numel allocated=.ok allocated := by
      simp [sourceGrid,hz,runPrograms]
    refine ⟨allocated,host_success b rt ops undef views d context x output initial allocated allocated dtype ha hc hp hn executed,
      executed,allocation,?_,?_,?_,?_,?_,?_⟩
    · refine ⟨?_,?_,?_⟩
      · intro i hi; omega
      · intro i hi; rfl
      · intro address outside; rfl
    · intro i hi; rfl
    · intro impossible; omega
    · intro impossible; omega
    · intro impossible; omega
    · intro jobs complete
      cases jobs with
      | nil => exact ⟨allocated,rfl,by intro i hi; omega⟩
      | cons i rest => have := (complete i).1 (by simp); omega
  · obtain ⟨safe,permissions,device,defined⟩ := applicable (by omega)
    obtain ⟨final,executed,correct,tail⟩ := grid_defined_success b d (ops d) (views x.region) (views output.region)
      output.numel allocated safe.inputValid safe.outputValid safe.separate defined
    have sourceExecuted := (sourceGrid_eq_executeGrid b d (ops d) views (undef d) x.region output.region output.numel allocated).trans executed
    refine ⟨final,host_success b rt ops undef views d context x output initial allocated final dtype ha hc hp hn sourceExecuted,
      sourceExecuted,allocation,correct,tail,(fun _ ↦ launch_safety x output views safe),(fun _ ↦ permissions),(fun _ ↦ device),?_⟩
    intro jobs complete
    exact complete_schedule b d (ops d) (views x.region) (views output.region) output.numel allocated
      safe.inputValid safe.outputValid safe.separate defined jobs complete

/-- Native errors remain fallible ordinary primitive outcomes. The successful
branch has exact original source execution, actual metadata even for zero-grid,
and independent typed results. Undefined conversion diagnostics belong only to
the mathematical interpreter; no native exception or rollback is asserted. -/
def SourceOutcome {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (kind : InputType) (x : Tensor) (metadata : TensorMetadata) (initial : M) : Prop :=
  match rt.emptyLike x initial with
  | .error reason failed => puzzle1 b rt ops undef views x initial=⟨[.emptyLike x.region],.error reason failed⟩
  | .ok output allocated => if cudaGuard x output then
      match rt.prepare x output output.numel with
      | .error reason => puzzle1 b rt ops undef views x initial=⟨launchTrace x output,.error (.native reason) allocated⟩
      | .ok context => match kind with
        | .fnuzE4 | .fnuzE5 => puzzle1 b rt ops undef views x initial=⟨launchTrace x output,.error .architectureRejected allocated⟩
        | .complex32 | .complex64 | .complex128 | .e8m0 | .packedFP4 =>
            puzzle1 b rt ops undef views x initial=⟨launchTrace x output,.error .signatureRejected allocated⟩
        | .admitted d => match rt.nativeReady context d x output output.numel with
          | .error reason => puzzle1 b rt ops undef views x initial=⟨launchTrace x output,.error (.native reason) allocated⟩
          | .ok () => ∃ final, FullSuccess b rt ops undef views d context x output metadata initial allocated final
    else puzzle1 b rt ops undef views x initial=
      ⟨[.emptyLike x.region,.cudaCheck x.isCUDA output.isCUDA],.error .cudaAssertion allocated⟩

specification add_value_finite_source_outcomes {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (d : Storage) → MemoryPrimitives M (Word (.storage d)))
    (undef : (d : Storage) → Word (.storage d)) (views : RegionName → View)
    (kind : InputType) (x : Tensor) (metadata : TensorMetadata) (initial : M)
    (dtype : x.dtype=kind.name)
    (allocation : ∀ output allocated, rt.emptyLike x initial=.ok output allocated →
      AllocationContract ops x output views metadata initial allocated)
    (applicable : ∀ output allocated context d, rt.emptyLike x initial=.ok output allocated →
      cudaGuard x output=Bool.true → rt.prepare x output output.numel=.ok context → kind=.admitted d →
      rt.nativeReady context d x output output.numel=.ok () → 0<output.numel →
      SafeAPI x output views ∧ MemoryApplicable (ops d) (views x.region) (views output.region) output.numel ∧
      (ops d).launchDevice=context.currentDevice ∧ ValuesDefined b d (ops d) (views x.region) output.numel allocated) :
    SourceOutcome b rt ops undef views kind x metadata initial := by
  cases ha : rt.emptyLike x initial with
  | error reason failed => simp [SourceOutcome,puzzle1,ha]
  | ok output allocated =>
    cases hc : cudaGuard x output with
    | «false» => simp [SourceOutcome,puzzle1,ha,hc]
    | «true» =>
      cases hp : rt.prepare x output output.numel with
      | error reason => simp [SourceOutcome,puzzle1,ha,hc,launch,hp]
      | ok context =>
        cases kind with
        | admitted d =>
          cases hn : rt.nativeReady context d x output output.numel with
          | error reason => simp [SourceOutcome,puzzle1,ha,hc,launch,hp,dtype,parse_name,dispatch,hn]
          | ok done =>
            cases done
            simp only [SourceOutcome,ha,hc,↓reduceIte,hp,hn]
            exact add_value_finite_success b rt ops undef views d context x output metadata initial allocated dtype ha hc hp hn
              (allocation output allocated ha) (applicable output allocated context d ha hc hp rfl hn)
        | _ => simp [SourceOutcome,puzzle1,ha,hc,launch,hp,dtype,parse_name,dispatch]

#guard_msgs (drop info) in

end
