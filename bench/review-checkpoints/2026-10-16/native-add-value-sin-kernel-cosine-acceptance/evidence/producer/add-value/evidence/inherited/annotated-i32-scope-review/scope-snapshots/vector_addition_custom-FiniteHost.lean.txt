inductive Failure where
  | native (message : String)
  | rankZero
  | signature (argument : DTypeAdmission.Argument) (dtype : DType)
  | addition
  | unsupportedCanonical
  | interpreter
  deriving DecidableEq

inductive Result (M : Type) where
  | ok (output : Tensor) (memory : M)
  | error (reason : Failure) (memory : M)

inductive Event where
  | emptyLike (input : RegionName)
  | sizeZero (output : RegionName)
  | block (value : Nat)
  | grid (count : Nat)
  | launch (size block : Nat)
  | returnTensor (output : RegionName)
  deriving DecidableEq

structure Run (M : Type) where
  trace : List Event
  result : Result M

structure LaunchContext where
  currentDevice : Nat

/-- Ordinary trusted Torch allocation and fallible native launch primitives.
`prepare` performs current-device/stream lookup and argument binding, before
frontend compilation. `nativeReady` performs backend/load/argument/launch setup
for the admitted frontend route. Neither returns a kernel result or memory. -/
structure Runtime (M : Type) where
  emptyLike : Tensor → M → Result M
  prepare : Tensor → Tensor → Tensor → Nat → Except Failure LaunchContext
  nativeReady : LaunchContext → Route → Tensor → Tensor → Tensor → Nat → Except Failure Unit

def parameters (x y output : Tensor) (n : Nat) : ValueProofs.Parameters :=
  ⟨x.region,y.region,output.region,n,16⟩

def launchTrace (x output : Tensor) (n : Nat) : List Event :=
  [.emptyLike x.region,.sizeZero output.region,.block 16,.grid ((n+15)/16),.launch n 16]

def successTrace (x output : Tensor) (n : Nat) : List Event :=
  launchTrace x output n ++ [.returnTensor output.region]

def launch {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (x y output : Tensor) (n : Nat) (allocated : M) : Result M :=
  match rt.prepare x y output n with
  | .error error => .error error allocated
  | .ok context => match parseDType x.dtype,parseDType y.dtype with
    | some xd,some yd => match DTypeAdmission.admission xd yd with
      | .signatureRejected arg bad => .error (.signature arg bad) allocated
      | .additionRejected => .error .addition allocated
      | .passesDTypeGates _ _ => match route xd yd with
        | none => .error .interpreter allocated
        | some r => match rt.nativeReady context r x y output n with
          | .error error => .error error allocated
          | .ok () => match runGrid (ops r) (scalarSource b r) views (undef r)
              x.region y.region output.region n allocated with
            | none => .error .interpreter allocated
            | some final => .ok output final
    | _,_ => .error .unsupportedCanonical allocated

/-- Original host order: allocation, size(0), B16, cdiv, launch, return.
There is intentionally no CUDA assertion in this source. A rank-zero tensor
fails only after empty_like has already produced the fresh allocation. -/
def customAdd {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (x y : Tensor) (initial : M) : Run M :=
  match rt.emptyLike x initial with
  | .error error failed => ⟨[.emptyLike x.region],.error error failed⟩
  | .ok output allocated => match output.shape with
    | [] => ⟨[.emptyLike x.region,.sizeZero output.region],.error .rankZero allocated⟩
    | n::_ => match launch b rt ops undef views x y output n allocated with
      | .error error failed => ⟨launchTrace x output n,.error error failed⟩
      | .ok returned final => ⟨successTrace x output n,.ok returned final⟩

structure SafeAPI (x y output : Tensor) (xD yD : DType)
    (views : RegionName → View) (metadata : TensorMetadata) (n : Nat) : Prop where
  indexBound : n<2^31
  numelX : x.numel=x.shape.prod
  numelY : y.numel=y.shape.prod
  numelOutput : output.numel=output.shape.prod
  xView : (views x.region).Matches x xD
  yView : (views y.region).Matches y yD
  outputView : (views output.region).Matches output xD
  devices : (∀ d, x.deviceIndex=some d → (views x.region).device=d) ∧
    (∀ d, y.deviceIndex=some d → (views y.region).device=d) ∧
    (∀ d, output.deviceIndex=some d → (views output.region).device=d)
  xBinding : (views x.region).Binds metadata x
  yBinding : (views y.region).Binds metadata y
  outputBinding : (views output.region).Binds metadata output
  outputStrides : output.strides=metadata.emptyLikeStrides x
  xAligned : (views x.region).Aligned
  yAligned : (views y.region).Aligned
  outputAligned : (views output.region).Aligned
  xValid : (views x.region).Valid n (rounded n)
  yValid : (views y.region).Valid n (rounded n)
  outputValid : (views output.region).Valid n (rounded n)
  outputSeparateX : StorageDisjoint (views output.region) (views x.region)
  outputSeparateY : StorageDisjoint (views output.region) (views y.region)

/-- Access permissions are relative to the current launch address space.
They may include ordinary peer/mapped memory. The source does not compare
CUDA membership or owning-device tags, so no such assertion is introduced. -/
structure MemoryApplicable {M α : Type} (ops : MemoryPrimitives M α)
    (x y output : View) (n : Nat) : Prop where
  readX : ∀ i byte, i<n → byte<bytes x.dtype → ops.readableByte (x.address i+byte)
  readY : ∀ i byte, i<n → byte<bytes y.dtype → ops.readableByte (y.address i+byte)
  readOutput : ∀ i byte, i<n → byte<bytes output.dtype → ops.readableByte (output.address i+byte)
  writeOutput : ∀ i byte, i<n → byte<bytes output.dtype → ops.writableByte (output.address i+byte)

structure LaunchSafety (x y output : View) (n : Nat) : Prop where
  alignment : ∀ v∈[x,y,output], ∀ i, v.address i % bytes v.dtype=0
  indices : ∀ pid lane, pid<(n+15)/16 → lane<16 →
    (start32 pid).toInt=(pid*16 : Nat) ∧
    (offset32 pid lane).toInt=(pid*16+lane : Nat) ∧
    mask32 n pid lane=decide (pid*16+lane<n)
  pointers : ∀ v∈[x,y,output], ∀ pid lane, pid<(n+15)/16 → lane<16 →
    (pointer64 v pid lane).toNat=v.address (pid*16+lane)
  accesses : ∀ v∈[x,y,output], ∀ i byte, i<n → byte<bytes v.dtype →
    v.storageBase≤v.address i+byte ∧ v.address i+byte<v.storageBase+v.storageBytes ∧ v.address i+byte<2^64
  ownership : ∀ i, i<n → i/16<(n+15)/16 ∧ i%16<16 ∧ (i/16)*16+i%16=i
  outputDistinct : ∀ i j, i≠j → SpanDisjoint (output.address i) (output.finish i) (output.address j) (output.finish j)

theorem launch_safety (x y output : View) (n : Nat) (hn : n<2^31)
    (hx : x.Valid n (rounded n)) (hy : y.Valid n (rounded n)) (ho : output.Valid n (rounded n))
    (ax : x.Aligned) (ay : y.Aligned) (ao : output.Aligned) :
    LaunchSafety x y output n := by
  have valid : ∀ v∈[x,y,output], v.Valid n (rounded n) := by
    intro v hv
    simp only [List.mem_cons,List.not_mem_nil,or_false] at hv
    rcases hv with rfl|rfl|rfl
    · exact hx
    · exact hy
    · exact ho
  have align : ∀ v∈[x,y,output], v.Aligned := by
    intro v hv
    simp only [List.mem_cons,List.not_mem_nil,or_false] at hv
    rcases hv with rfl|rfl|rfl
    · exact ax
    · exact ay
    · exact ao
  exact ⟨fun v hv i ↦ address_aligned v (align v hv) i,
    fun pid lane hp hl ↦ source_index_arithmetic n pid lane hn hp hl,
    fun v hv pid lane hp hl ↦ source_pointer_exact v n pid lane hn (valid v hv) hp hl,
    fun v hv i byte hi hb ↦ active_byte_access v n i byte (valid v hv) hi hb,
    fun i hi ↦ active_owner n i hi,fun i j hij ↦ distinct_cells output i j hij⟩

structure AllocationContract {M : Type} (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (x output : Tensor) (views : RegionName → View) (queries : TensorMetadata)
    (initial allocated : M) : Prop where
  metadata : ReturnedMetadata x output
  /-- Ordinary allocator metadata is required even when the grid is empty.
  These equalities impose no pointer validity or kernel access premise. -/
  outputBinding : (views output.region).Binds queries output
  outputStrides : output.strides=queries.emptyLikeStrides x
  frame : ∀ r address, (address<(views output.region).storageBase ∨ (views output.region).storageBase+(views output.region).storageBytes≤address) →
    (ops r).observeByte allocated address=(ops r).observeByte initial address

structure FullSuccess {M σ : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (r : Route) (context : LaunchContext) (x y output : Tensor) (n : Nat)
    (queries : TensorMetadata) (snapshot : World σ) (initial allocated final : M) : Prop where
  host : customAdd b rt ops undef views x y initial=⟨successTrace x output n,.ok output final⟩
  kernel : runGrid (ops r) (scalarSource b r) views (undef r) x.region y.region output.region n allocated=some final
  metadata : ReturnedMetadata x output
  outputBinding : (views output.region).Binds queries output
  outputStrides : output.strides=queries.emptyLikeStrides x
  correct : KernelCorrect b r (ops r) views (parameters x y output n) snapshot allocated final
  safety : 0<n → LaunchSafety (views x.region) (views y.region) (views output.region) n
  permissions : 0<n → MemoryApplicable (ops r) (views x.region) (views y.region) (views output.region) n
  currentDevice : 0<n → (ops r).launchDevice=context.currentDevice
  allocationFrame : ∀ rr address, (address<(views output.region).storageBase ∨ (views output.region).storageBase+(views output.region).storageBytes≤address) →
    (ops rr).observeByte allocated address=(ops rr).observeByte initial address

theorem launch_selected {M : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (xD yD : DType) (r : Route) (context : LaunchContext)
    (x y output : Tensor) (n : Nat) (allocated final : M)
    (hx : x.dtype=dtypeName xD) (hy : y.dtype=dtypeName yD)
    (hp : rt.prepare x y output n=.ok context) (selected : route xD yD=some r)
    (hn : rt.nativeReady context r x y output n=.ok ())
    (hk : runGrid (ops r) (scalarSource b r) views (undef r) x.region y.region output.region n allocated=some final) :
    launch b rt ops undef views x y output n allocated=.ok output final := by
  have h := route_matches_admission xD yD
  rw [selected] at h
  have stage : (DTypeAdmission.admission xD yD).stage=.admitted := of_decide_eq_true h.symm
  cases ha : DTypeAdmission.admission xD yD with
  | signatureRejected arg bad => simp [ha,DTypeAdmission.Outcome.stage] at stage
  | additionRejected => simp [ha,DTypeAdmission.Outcome.stage] at stage
  | passesDTypeGates computation store =>
    simp [launch,hp,hx,hy,parse_dtype,ha,selected,hn,hk]

#axiomsClean launch_safety
#axiomsClean launch_selected
#guard_msgs (drop info) in
#auditModuleAxioms
end
