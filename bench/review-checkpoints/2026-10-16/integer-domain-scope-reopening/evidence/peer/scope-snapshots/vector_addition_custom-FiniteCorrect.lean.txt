theorem vector_addition_custom_finite_success {M σ : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (xD yD : DType) (r : Route) (context : LaunchContext)
    (x y output : Tensor) (n : Nat) (shapeTail : List Nat)
    (snapshot : World σ) (initial allocated : M) (metadata : TensorMetadata)
    (hx : x.dtype=dtypeName xD) (hy : y.dtype=dtypeName yD)
    (ha : rt.emptyLike x initial=.ok output allocated)
    (shape : output.shape=n::shapeTail)
    (hp : rt.prepare x y output n=.ok context) (selected : route xD yD=some r)
    (hn : rt.nativeReady context r x y output n=.ok ())
    (allocation : AllocationContract ops x output views metadata initial allocated)
    (safe : SafeAPI x y output xD yD views metadata n)
    (permissions : MemoryApplicable (ops r) (views x.region) (views y.region) (views output.region) n)
    (device : (ops r).launchDevice=context.currentDevice)
    (domain : ValueDomain b r (parameters x y output n) snapshot)
    (inputs : InputsRelation r (ops r) (parameters x y output n) snapshot allocated (views x.region) (views y.region)) :
    ∃ final, FullSuccess b rt ops undef views r context x y output n metadata snapshot initial allocated final := by
  obtain ⟨final,executed,correct⟩ := source_grid_correct b r (ops r) views (undef r)
    (parameters x y output n) snapshot allocated safe.xValid safe.yValid safe.outputValid
      safe.outputSeparateX safe.outputSeparateY domain inputs
  refine ⟨final,?_,executed,allocation.metadata,allocation.outputBinding,allocation.outputStrides,correct,?_,(fun _ ↦ permissions),(fun _ ↦ device),allocation.frame⟩
  · have launched := launch_selected b rt ops undef views xD yD r context x y output n allocated final
      hx hy hp selected hn executed
    simp [customAdd,ha,shape,launched]
  · intro _
    exact launch_safety (views x.region) (views y.region) (views output.region) n safe.indexBound
      safe.xValid safe.yValid safe.outputValid safe.xAligned safe.yAligned safe.outputAligned

theorem vector_addition_custom_finite_empty {M σ : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (xD yD : DType) (r : Route) (context : LaunchContext)
    (x y output : Tensor) (tail : List Nat) (metadata : TensorMetadata) (snapshot : World σ) (initial allocated : M)
    (hx : x.dtype=dtypeName xD) (hy : y.dtype=dtypeName yD)
    (ha : rt.emptyLike x initial=.ok output allocated) (shape : output.shape=0::tail)
    (hp : rt.prepare x y output 0=.ok context) (selected : route xD yD=some r)
    (hn : rt.nativeReady context r x y output 0=.ok ())
    (allocation : AllocationContract ops x output views metadata initial allocated) :
    FullSuccess b rt ops undef views r context x y output 0 metadata snapshot initial allocated allocated := by
  have empty := source_grid_empty b r (ops r) views (undef r) (parameters x y output 0) snapshot allocated rfl
  refine ⟨?_,empty.1,allocation.metadata,allocation.outputBinding,allocation.outputStrides,empty.2,?_,?_,?_,allocation.frame⟩
  · have launched := launch_selected b rt ops undef views xD yD r context x y output 0 allocated allocated
      hx hy hp selected hn empty.1
    simp [customAdd,ha,shape,launched]
  all_goals intro impossible; omega

/-- Complete source outcomes under the concrete safe/defined applicability
of admitted launches. Native preparation and setup can fail, including CPU
pointers and current-device/argument incompatibilities; no CUDA assertion or
invented rollback is added. Undefined conversions have no claimed value. -/
def SourceOutcome {M σ : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (xD yD : DType) (x y : Tensor) (initial : M) (snapshot : Tensor → M → World σ)
    (metadata : TensorMetadata) : Prop :=
  match rt.emptyLike x initial with
  | .error reason failed => customAdd b rt ops undef views x y initial=⟨[.emptyLike x.region],.error reason failed⟩
  | .ok output allocated => match output.shape with
    | [] => customAdd b rt ops undef views x y initial=
        ⟨[.emptyLike x.region,.sizeZero output.region],.error .rankZero allocated⟩
    | n::_ => match rt.prepare x y output n with
      | .error reason => customAdd b rt ops undef views x y initial=⟨launchTrace x output n,.error reason allocated⟩
      | .ok context => match DTypeAdmission.admission xD yD with
        | .signatureRejected argument bad => customAdd b rt ops undef views x y initial=
            ⟨launchTrace x output n,.error (.signature argument bad) allocated⟩
        | .additionRejected => customAdd b rt ops undef views x y initial=
            ⟨launchTrace x output n,.error .addition allocated⟩
        | .passesDTypeGates _ _ => ∃ r, route xD yD=some r ∧
            match rt.nativeReady context r x y output n with
            | .error reason => customAdd b rt ops undef views x y initial=
                ⟨launchTrace x output n,.error reason allocated⟩
            | .ok () => ∃ final, FullSuccess b rt ops undef views r context x y output n
                metadata (snapshot output allocated) initial allocated final

specification vector_addition_custom_finite_source_outcomes {M σ : Type} (b : Numerics) (rt : Runtime M)
    (ops : (r : Route) → MemoryPrimitives M (Carrier r))
    (undef : (r : Route) → Carrier r) (views : RegionName → View)
    (xD yD : DType) (x y : Tensor) (initial : M) (snapshot : Tensor → M → World σ)
    (metadata : TensorMetadata) (hx : x.dtype=dtypeName xD) (hy : y.dtype=dtypeName yD)
    (allocation : ∀ output allocated, rt.emptyLike x initial=.ok output allocated →
      AllocationContract ops x output views metadata initial allocated)
    (applicable : ∀ output allocated n tail context r,
      rt.emptyLike x initial=.ok output allocated → output.shape=n::tail →
      rt.prepare x y output n=.ok context → route xD yD=some r →
      rt.nativeReady context r x y output n=.ok () → 0<n →
      SafeAPI x y output xD yD views metadata n ∧
      MemoryApplicable (ops r) (views x.region) (views y.region) (views output.region) n ∧
      (ops r).launchDevice=context.currentDevice ∧
      ValueDomain b r (parameters x y output n) (snapshot output allocated) ∧
      InputsRelation r (ops r) (parameters x y output n) (snapshot output allocated) allocated
        (views x.region) (views y.region)) :
    SourceOutcome b rt ops undef views xD yD x y initial snapshot metadata := by
  cases ha : rt.emptyLike x initial with
  | error reason failed => simp [SourceOutcome,customAdd,ha]
  | ok output allocated =>
    cases hs : output.shape with
    | nil => simp [SourceOutcome,customAdd,ha,hs]
    | cons n tail =>
      cases hp : rt.prepare x y output n with
      | error reason => simp [SourceOutcome,customAdd,launch,ha,hs,hp]
      | ok context =>
        cases hd : DTypeAdmission.admission xD yD with
        | signatureRejected argument bad =>
          simp [SourceOutcome,customAdd,launch,ha,hs,hp,hx,hy,parse_dtype,hd]
        | additionRejected =>
          simp [SourceOutcome,customAdd,launch,ha,hs,hp,hx,hy,parse_dtype,hd]
        | passesDTypeGates computation stored =>
          have existsRoute := route_matches_admission xD yD
          simp only [hd,DTypeAdmission.Outcome.stage,decide_true] at existsRoute
          cases hr : route xD yD with
          | none => simp [hr] at existsRoute
          | some r =>
            simp only [SourceOutcome,ha,hs,hp,hd]
            refine ⟨r,hr,?_⟩
            cases hn : rt.nativeReady context r x y output n with
            | error reason =>
              simp [hn,customAdd,launch,ha,hs,hp,hx,hy,parse_dtype,hd,hr]
            | ok done =>
              cases done
              simp only [hn]
              by_cases hz : n=0
              · subst n
                exact ⟨allocated,vector_addition_custom_finite_empty b rt ops undef views xD yD r context x y output tail metadata
                  (snapshot output allocated) initial allocated hx hy ha hs hp hr hn (allocation output allocated ha)⟩
              · obtain ⟨safe,permissions,device,domain,inputs⟩ := applicable output allocated n tail context r ha hs hp hr hn (by omega)
                exact vector_addition_custom_finite_success b rt ops undef views xD yD r context x y output n tail
                  (snapshot output allocated) initial allocated metadata hx hy ha hs hp hr hn
                  (allocation output allocated ha) safe permissions device domain inputs

#axiomsClean vector_addition_custom_finite_source_outcomes
#guard_msgs (drop info) in
#auditModuleAxioms
end
