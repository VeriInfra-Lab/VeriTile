import bench.tritonbench_g.vector_addition.Proofs.NativeArithmetic

namespace VeriTile.Bench.TritonBenchG.VectorAddition.Native
open DTypeScope TypedHost
set_option maxHeartbeats 10000000
set_option maxRecDepth 24000

/-- All22 dense scalar dtype names exposed by the pinned Torch installation.
The five extra names fail the original JIT's canonicalization before TTIR. -/
inductive DenseDType where
  | canonical (dtype : DType)
  | complex32 | complex64 | complex128 | e8m0 | fp4packed
  deriving DecidableEq, Repr
def denseName : DenseDType → String
  | .canonical d => dtypeName d
  | .complex32 => "complex32" | .complex64 => "complex64" | .complex128 => "complex128"
  | .e8m0 => "float8_e8m0fnu" | .fp4packed => "float4_e2m1fn_x2"
def denseTypes : List DenseDType := profiles.map DenseDType.canonical ++
  [.complex32,.complex64,.complex128,.e8m0,.fp4packed]
def densePairs : List (DenseDType×DenseDType) := denseTypes.flatMap fun x => denseTypes.map (x,·)
def binderSupported : DenseDType → Bool | .canonical _ => Bool.true | _ => Bool.false

inductive SetupReject where
  | binder (argument : DTypeAdmission.Argument) (dtype : DenseDType)
  | signature (argument : DTypeAdmission.Argument) (dtype : DType)
  | addition
  | backend (x y : DType)
  deriving DecidableEq, Repr

/-- Actual sm90 backend failures retained separately from source typed routes:
same-fp8 addition and the four fp64↔fp8 conversions fail lowering in Triton3.4. -/
def backendRejected (x y : DType) : Bool :=
  decide ((x=.fp8e4nv ∧ y=.fp8e4nv) ∨ (x=.fp8e5 ∧ y=.fp8e5) ∨
    (x=.fp64 ∧ (y=.fp8e4nv ∨ y=.fp8e5)) ∨ (y=.fp64 ∧ (x=.fp8e4nv ∨ x=.fp8e5)))

def admission (x y : DenseDType) : Except SetupReject Route :=
  if !binderSupported x then .error (.binder .x x)
  else if !binderSupported y then .error (.binder .y y)
  else match x,y with
    | .canonical xd,.canonical yd =>
        match DTypeAdmission.admission xd yd with
        | .signatureRejected arg d => .error (.signature arg d)
        | .additionRejected => .error .addition
        | .passesDTypeGates _ _ =>
            if backendRejected xd yd then .error (.backend xd yd)
            else match route xd yd with
              | some r => .ok r
              | none => .error .addition
    | _,_ => .error .addition

def outcomeStage : Except SetupReject Route → Nat
  | .ok _ => 4
  | .error (.binder _ _) => 0
  | .error (.signature _ _) => 1
  | .error .addition => 2
  | .error (.backend _ _) => 3
def stageCount (stage : Nat) : Nat :=
  (densePairs.filter fun pair => outcomeStage (admission pair.1 pair.2)==stage).length
theorem dense_inventory : denseTypes.length=22 ∧ densePairs.length=484 ∧ denseTypes.Nodup := by decide +kernel
theorem native_admission_counts : stageCount 0=195 ∧ stageCount 1=64 ∧ stageCount 2=36 ∧
    stageCount 3=6 ∧ stageCount 4=183 := by decide +kernel
theorem admitted_typed_route (x y : DenseDType) (r : Route) (h : admission x y=.ok r) :
    ∃ xd yd, x=.canonical xd ∧ y=.canonical yd ∧ route xd yd=some r ∧ backendRejected xd yd=Bool.false := by
  cases x <;> cases y
  all_goals try {simp [admission,binderSupported] at h}
  case canonical.canonical xd yd =>
    simp only [admission,binderSupported,Bool.not_true,Bool.false_eq_true,↓reduceIte] at h
    cases ha : DTypeAdmission.admission xd yd with
    | signatureRejected arg d => simp [ha] at h
    | additionRejected => simp [ha] at h
    | passesDTypeGates compute store =>
        simp only [ha] at h
        split at h
        · cases h
        · rename_i hb
          cases hr : route xd yd with
          | none => simp [hr] at h
          | some result =>
              simp [hr] at h
              subst result
              exact ⟨xd,yd,rfl,rfl,hr,Bool.eq_false_iff.mpr hb⟩

theorem fp32_admitted : admission (.canonical .fp32) (.canonical .fp32)=.ok (.floating .fp32 .fp32) := by decide +kernel
theorem binder_precedes_fnuz_signature :
    admission (.canonical .fp8e4b8) .complex64=.error (.binder .y .complex64) := by decide +kernel

#axiomsClean dense_inventory
#axiomsClean native_admission_counts
#axiomsClean admitted_typed_route
#axiomsClean fp32_admitted
#axiomsClean binder_precedes_fnuz_signature
#guard_msgs (drop info) in
#auditModuleAxioms
end VeriTile.Bench.TritonBenchG.VectorAddition.Native
