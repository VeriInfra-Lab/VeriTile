"""Seal the producer evidence after validation; never confers acceptance."""
import hashlib,json,re,shutil
from pathlib import Path
W=Path(__file__).resolve().parent
R=Path('/mnt/VerifiableAILab/VerifiableAILab/zenan.li/VeriTile')
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
load=lambda p:json.loads(Path(p).read_text())
def dump(p,d):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(d,indent=2)+'\n')
receipt=W/'evidence/receipt.json'
assert not receipt.exists(), 'Refusing to mutate frozen evidence'
deps=load(W/'evidence/dependencies-before.json')
for rel,v in deps['project'].items():
 assert sha(R/rel)==v['sha256'],('source changed',rel)
 assert sha(v['object'])==v['object_sha256'],('object changed',rel)
modules={str(p.relative_to(W/'after')):sha(p) for p in sorted((W/'after').rglob('*.lean'))}
assert len(modules)==7
attempts=[(p,load(p)) for p in sorted((W/'evidence/lean-attempts').glob('*.json'))]
final=[];clean=noncircular=0
for rel,h in modules.items():
 source=W/'after'/rel;t=source.read_text()
 assert not re.search(r'(?m)^\s*(axiom\s|sorry\b|admit\b)',t),rel
 matching=[(p,v) for p,v in attempts if v['source']=='after/'+rel and v['source_sha256']==h and v['exit_code']==0 and v['source_unchanged']]
 assert matching,rel
 p,v=matching[-1];log=(W/v['log']).read_text();assert sha(W/v['log'])==v['log_sha256']
 c=t.count('#axiomsClean');n=t.count('#specNonCircular')
 assert log.count('axiom footprint ⊆ standard base ✓')==c,(rel,c)
 assert log.count('non-circular ✓')==n,(rel,n)
 clean+=c;noncircular+=n
 obj=(W/'build-overlay'/rel).with_suffix('.olean');assert obj.exists() and not obj.is_symlink(),rel
 dst=(W/'evidence/final-objects'/rel).with_suffix('.olean');dst.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(obj,dst)
 final.append({'module':rel,'source_sha256':h,'attempt':str(p.relative_to(W)),'log':v['log'],'log_sha256':v['log_sha256'],'object':str(dst.relative_to(W)),'object_sha256':sha(dst),'clean_axiom_checks':c,'noncircular_checks':n})
dump(W/'evidence/current-final-rebuild.json',final)
dump(W/'evidence/module-map.json',modules)
dump(W/'evidence/dependencies-after.json',{'all_unchanged':True,'capture_sha256':sha(W/'evidence/dependencies-before.json'),'project_sources':len(deps['project']),'project_objects':len(deps['project']),'modules':modules})
delta=load(W/'evidence/numerical-authority-delta.json')
for p in delta['provenance'].values():
 assert p['lean_source_sha256']==modules,'stale numerical module map'
 for field in ['raw_ttir','frontend_receipt','producer_scalar_inventory']:
  ref=p[field];assert sha(W/ref['path'])==ref['sha256']
 for field in ['execution_source','host_source']:
  ref=p[field];assert modules[ref['module']]==ref['sha256']
for n in ['authority-check','proposal-validation']:
 assert load(W/f'evidence/{n}.json')['status']=='passed'
verification={'status':'passed','final_modules':len(modules),'clean_axiom_checks':clean,'noncircular_checks':noncircular,'successful_attempts':sum(v['exit_code']==0 for p,v in attempts),'failed_attempts_preserved':sum(v['exit_code']!=0 for p,v in attempts),'existing_project_sources_unchanged':len(deps['project']),'existing_project_objects_unchanged':len(deps['project']),'external_import_modules_recorded':len(deps['external_imports']),'scope_cases':4498,'scalar_bindings':296,'scalar_families':37,'compiled_C_pointer_checks':1425408,'original_dtype_pairs':484,'full_backend_compiles':183,'frontend_typed_routes':189,'backend_rejects':6,'abi_profiles':14,'GPU_launches':0,'large_allocations':0,'independent_review':'pending','acceptance_increment':0,'trust_boundary':['ordinary empty_like allocation and metadata contract','successful installed CUDA pointer extraction and mapped backing metadata','ordinary typed byte loads and width-specific pure packing/stores','explicit scalar IEEE arithmetic/conversion ports; no result callback'],'limits':['structural observations do not claim native exceptions, races, rollback or postfailure memory','no arbitrary tagged-carrier readback law','original bounded Nat-prefix correctness remains conditional; actual-i32 theorem has full nonnegative signed64 extent domain','synthetic adapter checks do not constitute independent approval']}
dump(W/'evidence/final-verification.json',verification)
# Exclude only build links, duplicate compiler caches and transient cache dirs.
excluded={'build-overlay','cache','tmp','__pycache__'}
artifacts={str(p.relative_to(W)):sha(p) for p in sorted(W.rglob('*')) if p.is_file() and p!=receipt and not excluded.intersection(p.relative_to(W).parts) and 'after' not in p.relative_to(W).parts}
result={'frozen':True,'task':'vector_addition','revision':'actual-i32-source-revision1','producer':'matrix_transpose_acceptance','independent_review':'pending','acceptance_increment':0,'modules':modules,'checkpoint_module_source_paths':{p:p+'.txt' for p in modules},'artifacts':artifacts,'artifact_count':len(artifacts),'verification':verification,'excluded_paths':['after/** (separately bound by modules)','build-overlay/**','**/cache/**','**/tmp/**','**/__pycache__/**']}
dump(receipt,result)
assert all(sha(W/p)==h for p,h in artifacts.items())
print(json.dumps({'receipt':str(receipt),'sha256':sha(receipt),'modules':len(modules),'artifacts':len(artifacts),'verification':verification},indent=2))
