/-- Actual metadata queries and access permissions of the trusted Torch and
device memory boundary. The output retains preserve-format strides. -/
structure Metadata where
  buffer : Tensor → Buffer
  emptyLikeStrides : Tensor → List Int
  currentDevice : Nat
def layout (metadata : Metadata) (x y output : Tensor) : Layout :=
  ⟨metadata.buffer x,metadata.buffer y,metadata.buffer output,metadata.currentDevice⟩

def TensorBinding (metadata : Metadata) (t : Tensor) (d : DType) : Prop :=
  t.dtype=dtypeName d ∧ t.numel=t.shape.prod ∧ hostExtent t.numel ∧
  0≤t.storageOffset ∧ (metadata.buffer t).device=t.deviceIndex ∧
  (metadata.buffer t).data<2^64 ∧ (metadata.buffer t).base+(metadata.buffer t).capacity≤2^64 ∧
  (if t.numel=0 then (metadata.buffer t).data=0 else
    (metadata.buffer t).data=(metadata.buffer t).base+t.storageOffset.toNat*bytes d)

/-- Ordinary allocation metadata and freshness, required unconditionally even
when every kernel mask is false. There is no index bound, shape equality for y,
same-device condition, contiguous-layout assumption, or requested-span bound. -/
structure AllocationBinding (metadata : Metadata) (x y output : Tensor) : Prop where
  returned : ReturnedMetadata x output
  outputStrides : output.strides=metadata.emptyLikeStrides x
  hostNumelX : x.numel=x.shape.prod
  hostNumelY : y.numel=y.shape.prod
  hostRangeX : hostExtent x.numel
  hostRangeY : hostExtent y.numel
  outputOffset : output.storageOffset=0
  outputData : (metadata.buffer output).data=if output.numel=0 then 0 else (metadata.buffer output).base
  xBinding : ∀ d, x.dtype=dtypeName d → TensorBinding metadata x d
  yBinding : ∀ d, y.dtype=dtypeName d → TensorBinding metadata y d
  outputBinding : ∀ d, output.dtype=dtypeName d → TensorBinding metadata output d
  fresh : (layout metadata x y output).Fresh

inductive SetupStage where | context | compilation | pointerPacking | launch
  deriving DecidableEq, Repr
abbrev Environment := SetupStage → Option String
abbrev Allocator (M : Type) := Tensor → M → Except String (Tensor×M)
inductive Failure where
  | allocation (detail : String)
  | cudaAssertion
  | setup (reason : SetupReject)
  | primitive (stage : SetupStage) (detail : String)
  | gridLimit (requested : Nat)
  deriving DecidableEq, Repr
inductive Outcome (M : Type) where
  | failed (trace : List Event) (why : Failure)
  | observed (trace : List Event) (issue : Issue)
  | returned (trace : List Event) (tensor : Tensor) (memory : M)

def checkedTrace (x y output : Tensor) : List Event :=
  [.emptyLike x.region,.cudaCheck x.isCUDA y.isCUDA output.isCUDA]
def attemptedTrace (x y output : Tensor) : List Event :=
  checkedTrace x y output ++ [.numel output.region output.numel,.launch output.numel 1024 (grid output.numel)]

/-- This environment represents only ordinary fallible setup primitives. It
has no access to input values and cannot provide a kernel result or execution.
All admitted lanes execute below after setup succeeds. -/
def executeAllocated {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (xd yd : DenseDType)
    (x y output : Tensor) (allocated : M) : Outcome M :=
  if cudaGuard x y output=Bool.false then .failed (checkedTrace x y output) .cudaAssertion
  else match env .context with
    | some detail => .failed (attemptedTrace x y output) (.primitive .context detail)
    | none => match admission xd yd with
      | .error reason => .failed (attemptedTrace x y output) (.setup reason)
      | .ok r => match env .compilation,env .pointerPacking with
        | some detail,_ => .failed (attemptedTrace x y output) (.primitive .compilation detail)
        | none,some detail => .failed (attemptedTrace x y output) (.primitive .pointerPacking detail)
        | none,none =>
          if ¬gridSupported output.numel then .failed (attemptedTrace x y output) (.gridLimit (grid output.numel))
          else match env .launch with
            | some detail => .failed (attemptedTrace x y output) (.primitive .launch detail)
            | none => match executeLaunch r (ops r) b (layout metadata x y output) output.numel allocated with
              | .error issue => .observed (attemptedTrace x y output) issue
              | .ok memory => .returned (successTrace x y output) output memory

def executeHost {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (xd yd : DenseDType)
    (x y : Tensor) (initial : M) : Outcome M :=
  match alloc x initial with
  | .error detail => .failed [.emptyLike x.region] (.allocation detail)
  | .ok (output,allocated) => executeAllocated ops b env metadata xd yd x y output allocated

/-- Source first-stage setup failure facts. These contain no access safety or
defined-value predicate and do not classify model observations as native errors. -/
def SetupFailure (env : Environment) (xd yd : DenseDType) (n : Nat) (why : Failure) : Prop :=
  (∃ detail, env .context=some detail ∧ why=.primitive .context detail) ∨
  (env .context=none ∧ ∃ reason, admission xd yd=.error reason ∧ why=.setup reason) ∨
  (env .context=none ∧ ∃ r, admission xd yd=.ok r ∧
    ((∃ detail, env .compilation=some detail ∧ why=.primitive .compilation detail) ∨
     (env .compilation=none ∧ ∃ detail, env .pointerPacking=some detail ∧ why=.primitive .pointerPacking detail) ∨
     (env .compilation=none ∧ env .pointerPacking=none ∧ ¬gridSupported n ∧ why=.gridLimit (grid n)) ∨
     (env .compilation=none ∧ env .pointerPacking=none ∧ gridSupported n ∧
       ∃ detail, env .launch=some detail ∧ why=.primitive .launch detail)))
def SetupReady (env : Environment) (xd yd : DenseDType) (n : Nat) (r : Route) : Prop :=
  (∀ stage, env stage=none) ∧ admission xd yd=.ok r ∧ gridSupported n

structure ReturnedSpec {M : Type} (r : Route) (ops : Memory M (Carrier r)) (b : Numerics)
    (metadata : Metadata) (x y output : Tensor) (allocated final : M) : Prop where
  allocation : AllocationBinding metadata x y output
  writes : ∃ writes, launchPlan r ops b (layout metadata x y output) output.numel allocated=.ok writes ∧
    final=applyWrites ops r.signature.1 writes allocated ∧
    (∀ address, (∀ w∈writes, ¬w.Touches r.signature.1 address) → ops.byte final address=ops.byte allocated address) ∧
    (∀ address, (∀ w∈writes, w.address=address ∨ SpanDisjoint w.address (w.address+bytes r.signature.1)
        address (address+bytes r.signature.1)) → ops.load final r.signature.1 address=finalCell ops r.signature.1 writes allocated address)
  exterior : OutsideOutput ops (layout metadata x y output) final allocated

def AllocatedSpec {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (xd yd : DenseDType)
    (x y output : Tensor) (allocated : M) : Outcome M → Prop
  | .failed trace why =>
      (cudaGuard x y output=Bool.false ∧ trace=checkedTrace x y output ∧ why=.cudaAssertion) ∨
      (cudaGuard x y output=Bool.true ∧ trace=attemptedTrace x y output ∧ SetupFailure env xd yd output.numel why)
  | .observed trace issue => cudaGuard x y output=Bool.true ∧ trace=attemptedTrace x y output ∧
      ∃ r, SetupReady env xd yd output.numel r ∧
        launchPlan r (ops r) b (layout metadata x y output) output.numel allocated=.error issue
  | .returned trace tensor final => cudaGuard x y output=Bool.true ∧ trace=successTrace x y output ∧ tensor=output ∧
      ∃ r, SetupReady env xd yd output.numel r ∧ ReturnedSpec r (ops r) b metadata x y output allocated final

theorem allocated_all_outcomes {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (env : Environment) (metadata : Metadata) (xd yd : DenseDType) (x y output : Tensor) (allocated : M)
    (binding : AllocationBinding metadata x y output) :
    AllocatedSpec ops b env metadata xd yd x y output allocated
      (executeAllocated ops b env metadata xd yd x y output allocated) := by
  unfold executeAllocated
  by_cases hc : cudaGuard x y output=Bool.false
  · simp [hc,AllocatedSpec]
  · have cuda : cudaGuard x y output=Bool.true := by cases hb : cudaGuard x y output <;> simp_all
    rw [if_neg hc]
    cases he : env .context with
    | some detail => simp [AllocatedSpec,cuda,SetupFailure,he]
    | none =>
      cases ha : admission xd yd with
      | error reason => simp [AllocatedSpec,cuda,SetupFailure,he,ha]
      | ok r =>
        cases hcomp : env .compilation with
        | some detail => simp [AllocatedSpec,cuda,SetupFailure,he,ha,hcomp]
        | none =>
          cases hpack : env .pointerPacking with
          | some detail => simp [AllocatedSpec,cuda,SetupFailure,he,ha,hcomp,hpack]
          | none =>
            simp only []
            by_cases hg : gridSupported output.numel
            · rw [if_neg (not_not.mpr hg)]
              cases hl : env .launch with
              | some detail => simp [AllocatedSpec,cuda,SetupFailure,he,ha,hcomp,hpack,hg,hl]
              | none =>
                have ready : SetupReady env xd yd output.numel r :=
                  ⟨(by intro stage; cases stage <;> assumption),ha,hg⟩
                have corr := launch_correspondence r (ops r) b (layout metadata x y output) output.numel allocated binding.fresh
                cases ht : launchPlan r (ops r) b (layout metadata x y output) output.numel allocated with
                | error issue =>
                  simp only [ht,Except.map] at corr
                  rw [corr]
                  exact ⟨cuda,rfl,r,ready,ht⟩
                | ok writes =>
                  simp only [ht,Except.map] at corr
                  rw [corr]
                  refine ⟨cuda,rfl,rfl,r,ready,binding,⟨writes,ht,rfl,?_,?_⟩,?_⟩
                  · exact fun address outside => writes_byte_frame (ops r) r.signature.1 writes allocated address outside
                  · exact fun address separated => writes_cell_result (ops r) r.signature.1 writes allocated address separated
                  · exact writes_outside (ops r) (layout metadata x y output) r.signature.1 writes allocated
                      (launch_write_bounds r (ops r) b (layout metadata x y output) output.numel allocated writes ht)
            · simp [hg,AllocatedSpec,cuda,SetupFailure,he,ha,hcomp,hpack]

def HostSpec {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (xd yd : DenseDType)
    (x y : Tensor) (initial : M) (outcome : Outcome M) : Prop :=
  (∃ detail, alloc x initial=.error detail ∧ outcome=.failed [.emptyLike x.region] (.allocation detail)) ∨
  (∃ output allocated, alloc x initial=.ok (output,allocated) ∧
    AllocatedSpec ops b env metadata xd yd x y output allocated outcome)

/-- Full source master: allocation, original CUDA assertion, all22 signatures,
ordinary setup failures, exact finite requests/observations, and actual return.
The only allocation contract is ordinary metadata/freshness, at every extent. -/
theorem vector_addition_native_source_outcomes {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (env : Environment) (metadata : Metadata) (xd yd : DenseDType)
    (x y : Tensor) (initial : M) (hx : x.dtype=denseName xd) (hy : y.dtype=denseName yd)
    (contract : ∀ output allocated, alloc x initial=.ok (output,allocated) → AllocationBinding metadata x y output) :
    HostSpec ops b alloc env metadata xd yd x y initial (executeHost ops b alloc env metadata xd yd x y initial) := by
  cases ha : alloc x initial with
  | error detail => exact Or.inl ⟨detail,ha,by simp [executeHost,ha]⟩
  | ok result =>
      rcases result with ⟨output,allocated⟩
      refine Or.inr ⟨output,allocated,ha,?_⟩
      simpa only [executeHost,ha] using allocated_all_outcomes ops b env metadata xd yd x y output allocated (contract output allocated ha)

/-- The newly recovered path returns the exact allocated tensor and unchanged
allocated memory. No permission, input encoding, scalar value, or index bound
premise is needed because no lane loads or stores at this narrowed bound. -/
theorem vector_addition_two31_return {M : Type} (ops : (r : Route) → Memory M (Carrier r)) (b : Numerics)
    (alloc : Allocator M) (metadata : Metadata) (xd yd : DenseDType) (r : Route)
    (x y output : Tensor) (initial allocated : M)
    (ha : alloc x initial=.ok (output,allocated)) (hc : cudaGuard x y output=Bool.true)
    (hr : admission xd yd=.ok r) (hn : output.numel=2^31) :
    executeHost ops b alloc (fun _ => none) metadata xd yd x y initial=
      .returned (successTrace x y output) output allocated := by
  simp only [executeHost,ha,executeAllocated,hc,Bool.true_eq_false,↓reduceIte,hr,hn,
    counterexample_grid.2.1,not_true_eq_false,no_access_returned_memory]

#specNonCircular HostSpec avoiding [executeHost,executeAllocated,executeLaunch,executeGrid,executeProgram,sourceWrites]
#specNonCircular ReturnedSpec avoiding [executeHost,executeAllocated,executeLaunch,executeGrid,executeProgram,sourceWrites]
#axiomsClean allocated_all_outcomes
#axiomsClean vector_addition_native_source_outcomes
#axiomsClean vector_addition_two31_return
#guard_msgs (drop info) in
#auditModuleAxioms
end
