open Classical

/-- Reference source requests, all read from the original host snapshot.
Ownership is accumulated within this launch, and reset between launches. -/
def nativeGridPlan (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (snapshot : ByteMemory) :
    List Nat → List Nat → Except NativeIssue (List NativeWrite × List Nat)
  | [],seen => .ok ([],seen)
  | pid::rest,seen => do
      let (head,owners) ← nativeProgramPlan b d a alloc p snapshot pid seen
      let (tail,finalOwners) ← nativeGridPlan b d a alloc p snapshot rest owners
      pure (head++tail,finalOwners)

/-- Each next program reads the actually mutated byte memory. -/
def executeNativeGrid (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) : List Nat → NativeState → Except NativeIssue NativeState
  | [],state => .ok state
  | pid::rest,state => do
      let next ← executeNativeProgram b d a alloc p pid state
      executeNativeGrid b d a alloc p rest next

theorem native_grid_correspondence (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot : ByteMemory)
    (hi : p.input.Disjoint p.output) (hv : p.inverse.Disjoint p.output)
    (pids : List Nat) (state : NativeState) (same : OutsideOutput p state.memory snapshot) :
    executeNativeGrid b d a alloc p pids state=
      (nativeGridPlan b d a alloc p snapshot pids state.owners).map
        (fun plan => ⟨applyNativeWrites plan.1 state.memory,plan.2⟩) := by
  induction pids generalizing state with
  | nil => rfl
  | cons pid rest ih =>
    simp only [executeNativeGrid,nativeGridPlan,native_program_correspondence]
    rw [native_program_plan_frame b d a alloc p state.memory snapshot same hi hv]
    cases hh : nativeProgramPlan b d a alloc p snapshot pid state.owners with
    | error issue => rfl
    | ok head =>
      rcases head with ⟨head,owners⟩
      simp only [Except.map,native_bind_ok]
      have hb := native_program_write_bounds b d a alloc p snapshot pid state.owners head owners hh
      have hs : OutsideOutput p (applyNativeWrites head state.memory) snapshot :=
        outside_output_trans p _ _ _ (writes_outside_output p head state.memory hb) same
      rw [ih ⟨applyNativeWrites head state.memory,owners⟩ hs]
      cases nativeGridPlan b d a alloc p snapshot rest owners with
      | error issue => rfl
      | ok tail =>
        rcases tail with ⟨tail,finalOwners⟩
        simp only [Except.map,native_bind_ok,native_pure,writes_append]

theorem native_grid_write_bounds (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot : ByteMemory)
    (pids : List Nat) (seen : List Nat) (writes : List NativeWrite) (owners : List Nat)
    (h : nativeGridPlan b d a alloc p snapshot pids seen=.ok (writes,owners)) :
    ∀ w∈writes, p.output.Contains w.outputAddress := by
  induction pids generalizing seen writes owners with
  | nil => cases h; simp
  | cons pid rest ih =>
    simp only [nativeGridPlan] at h
    cases hh : nativeProgramPlan b d a alloc p snapshot pid seen with
    | error issue => simp [hh] at h
    | ok head =>
      rcases head with ⟨head,nextOwners⟩
      cases ht : nativeGridPlan b d a alloc p snapshot rest nextOwners with
      | error issue => simp [hh,ht] at h
      | ok tail =>
        rcases tail with ⟨tail,finalOwners⟩
        simp [hh,ht] at h
        rcases h with ⟨rfl,rfl⟩
        intro w hw
        rcases List.mem_append.1 hw with hw|hw
        · exact native_program_write_bounds b d a alloc p snapshot pid seen head nextOwners hh w hw
        · exact ih _ _ _ ht w hw

/-- Both original config entries are retained. Their block/group/warp/stage
metadata coincide, as in the Python source. -/
def nativePids (a : Input) (_cfg : Config) : List Nat := List.range (gridM a.M*gridN a.N)

def nativeLaunchPlan (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (snapshot : ByteMemory) (cfg : Config) : Except NativeIssue (List NativeWrite) :=
  (nativeGridPlan b d a alloc p snapshot (nativePids a cfg) []).map Prod.fst

def executeNativeLaunch (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (cfg : Config) (memory : ByteMemory) : Except NativeIssue ByteMemory :=
  (executeNativeGrid b d a alloc p (nativePids a cfg) ⟨memory,[]⟩).map NativeState.memory

theorem native_launch_correspondence (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot memory : ByteMemory)
    (hi : p.input.Disjoint p.output) (hv : p.inverse.Disjoint p.output)
    (cfg : Config) (same : OutsideOutput p memory snapshot) :
    executeNativeLaunch b d a alloc p cfg memory=
      (nativeLaunchPlan b d a alloc p snapshot cfg).map (fun writes => applyNativeWrites writes memory) := by
  unfold executeNativeLaunch nativeLaunchPlan
  rw [native_grid_correspondence b d a alloc p snapshot hi hv _ _ same]
  cases nativeGridPlan b d a alloc p snapshot (nativePids a cfg) [] <;> rfl

theorem native_launch_write_bounds (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot : ByteMemory)
    (cfg : Config) (writes : List NativeWrite)
    (h : nativeLaunchPlan b d a alloc p snapshot cfg=.ok writes) :
    ∀ w∈writes, p.output.Contains w.outputAddress := by
  unfold nativeLaunchPlan at h
  cases hg : nativeGridPlan b d a alloc p snapshot (nativePids a cfg) [] with
  | error issue => cases h
  | ok plan =>
    cases h
    exact native_grid_write_bounds b d a alloc p snapshot _ _ _ _ hg

/-- All finite autotune trials followed by the selected final config. -/
def nativeTrialsPlan (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (snapshot : ByteMemory) : List Config → Except NativeIssue (List NativeWrite)
  | [] => .ok []
  | cfg::rest => do
      let head ← nativeLaunchPlan b d a alloc p snapshot cfg
      let tail ← nativeTrialsPlan b d a alloc p snapshot rest
      pure (head++tail)

def executeNativeTrials (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) : List Config → ByteMemory → Except NativeIssue ByteMemory
  | [],memory => .ok memory
  | cfg::rest,memory => do
      let next ← executeNativeLaunch b d a alloc p cfg memory
      executeNativeTrials b d a alloc p rest next

theorem native_trials_correspondence (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot : ByteMemory)
    (hi : p.input.Disjoint p.output) (hv : p.inverse.Disjoint p.output)
    (configs : List Config) (memory : ByteMemory) (same : OutsideOutput p memory snapshot) :
    executeNativeTrials b d a alloc p configs memory=
      (nativeTrialsPlan b d a alloc p snapshot configs).map (fun writes => applyNativeWrites writes memory) := by
  induction configs generalizing memory with
  | nil => rfl
  | cons cfg rest ih =>
    simp only [executeNativeTrials,nativeTrialsPlan]
    rw [native_launch_correspondence b d a alloc p snapshot memory hi hv cfg same]
    cases hh : nativeLaunchPlan b d a alloc p snapshot cfg with
    | error issue => rfl
    | ok head =>
      simp only [Except.map,native_bind_ok]
      have hb := native_launch_write_bounds b d a alloc p snapshot cfg head hh
      have hs : OutsideOutput p (applyNativeWrites head memory) snapshot :=
        outside_output_trans p _ _ _ (writes_outside_output p head memory hb) same
      rw [ih _ hs]
      cases nativeTrialsPlan b d a alloc p snapshot rest with
      | error issue => rfl
      | ok tail => simp only [Except.map,native_bind_ok,native_pure,writes_append]

theorem native_trials_write_bounds (b : Numerics) (d : Storage) (a : Input)
    (alloc : Allocations) (p : PhysicalLayout) (snapshot : ByteMemory)
    (configs : List Config) (writes : List NativeWrite)
    (h : nativeTrialsPlan b d a alloc p snapshot configs=.ok writes) :
    ∀ w∈writes, p.output.Contains w.outputAddress := by
  induction configs generalizing writes with
  | nil => cases h; simp
  | cons cfg rest ih =>
    simp only [nativeTrialsPlan] at h
    cases hh : nativeLaunchPlan b d a alloc p snapshot cfg with
    | error issue => simp [hh] at h
    | ok head =>
      cases ht : nativeTrialsPlan b d a alloc p snapshot rest with
      | error issue => simp [hh,ht] at h
      | ok tail =>
        simp [hh,ht] at h
        subst writes
        intro w hw
        rcases List.mem_append.1 hw with hw|hw
        · exact native_launch_write_bounds b d a alloc p snapshot cfg head hh w hw
        · exact ih _ ht w hw

#specNonCircular nativeTrialsPlan avoiding [executeNativeTrials,executeNativeLaunch,executeNativeGrid,executeNativeProgram,applyNativeWrites,source,Expr.eval]
#axiomsClean native_grid_correspondence
#axiomsClean native_trials_correspondence
#axiomsClean native_trials_write_bounds
#guard_msgs (drop info) in
#auditModuleAxioms
end
end QuantTransposeFinite
