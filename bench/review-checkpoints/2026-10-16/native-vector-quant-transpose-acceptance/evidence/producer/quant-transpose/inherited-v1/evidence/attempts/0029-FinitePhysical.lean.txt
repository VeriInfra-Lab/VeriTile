abbrev ByteMemory := Nat → Option (BitVec 8)

def bytes (d : Storage) : Nat := d.width/8

theorem bytes_positive (d : Storage) : 0<bytes d := by cases d <;> decide

structure ByteRegion where
  base : Nat
  capacity : Nat
  ownerDevice : Nat
  readable : Nat → Prop
  writable : Nat → Prop

def ByteRegion.Valid (r : ByteRegion) : Prop := r.base+r.capacity≤2^64

def ByteRegion.Contains (r : ByteRegion) (p : Nat) : Prop := r.base≤p ∧ p<r.base+r.capacity

def ByteRegion.Disjoint (r s : ByteRegion) : Prop :=
  r.base+r.capacity≤s.base ∨ s.base+s.capacity≤r.base

structure PhysicalLayout where
  input : ByteRegion
  maximum : ByteRegion
  inverse : ByteRegion
  output : ByteRegion
  /-- Actual x.data_ptr(), encoded as an unsigned native pointer word. -/
  inputPointer : BitVec 64
  initialBytes : ByteMemory

def byteOf {d : Storage} (w : Word d) (k : Nat) : BitVec 8 :=
  BitVec.ofNat 8 (w.toNat/2^(8*k))

/-- External memory relation, independent of every execution function. It
binds typed source cells to bytes at the actual view data pointer. -/
def InputBytes (d : Storage) (a : Input) (alloc : Allocations) (m : Memory) (x : String) (p : PhysicalLayout) : Prop :=
  ∀ i, (i∈a.logicalOffsets ∨ ∃ j, j<a.numel ∧ i=inputOffset (parameters a alloc x) j) → ∀ w, m x i=some (.mk d w) →
    ∀ k, k<bytes d → p.initialBytes (p.inputPointer.toNat+i*bytes d+k)=some (byteOf w k)

/-- Allocation/metadata and access mechanics are trusted primitive inputs;
none is an oracle for a successful launch or final output. All actual base,
capacity, device, permissions and disjointness facts are explicit here. -/
structure PhysicalBinding (d : Storage) (a : Input) (alloc : Allocations)
    (m : Memory) (x : String) (p : PhysicalLayout) : Prop where
  inputBase : p.input.base=a.storageBase
  maximumBase : p.maximum.base=alloc.maximumBase
  inverseBase : p.inverse.base=alloc.scaleBase
  outputBase : p.output.base=alloc.outputBase
  inputValid : p.input.Valid
  maximumValid : p.maximum.Valid
  inverseValid : p.inverse.Valid
  outputValid : p.output.Valid
  inputCapacity : p.input.capacity=a.storageLength*bytes d
  inputPointer : p.inputPointer.toNat=p.input.base+a.storageOffset*bytes d
  maximumCapacity : bytes d≤p.maximum.capacity
  inverseCapacity : bytes (compute d)≤p.inverse.capacity
  outputCapacity : a.numel≤p.output.capacity
  inputDevice : p.input.ownerDevice=a.deviceIndex
  maximumDevice : p.maximum.ownerDevice=a.deviceIndex
  inverseDevice : p.inverse.ownerDevice=a.deviceIndex
  outputDevice : p.output.ownerDevice=alloc.currentDevice
  inputHostRead : p.input.readable a.deviceIndex
  inputKernelRead : p.input.readable alloc.currentDevice
  maximumHostWrite : p.maximum.writable a.deviceIndex
  inverseHostWrite : p.inverse.writable a.deviceIndex
  inverseKernelRead : p.inverse.readable alloc.currentDevice
  outputKernelWrite : p.output.writable alloc.currentDevice
  inputAlignment : p.inputPointer.toNat%bytes d=0
  maximumAlignment : p.maximum.base%bytes d=0
  inverseAlignment : p.inverse.base%bytes (compute d)=0
  inputMaximum : p.input.Disjoint p.maximum
  inputInverse : p.input.Disjoint p.inverse
  inputOutput : p.input.Disjoint p.output
  maximumInverse : p.maximum.Disjoint p.inverse
  maximumOutput : p.maximum.Disjoint p.output
  inverseOutput : p.inverse.Disjoint p.output
  inputWords : InputBytes d a alloc m x p

/-- Physical access theorem for active source lanes. Both byte widths and
base-plus-offset arithmetic occur before converting to a native pointer. -/
specification quant_transpose_physical_accesses
    (d : Storage) (a : Input) (alloc : Allocations) (m : Memory) (x : String)
    (logical : List (Word d)) (xs : Nat → Word d) (p : PhysicalLayout)
    (D : Domain d a alloc m x logical xs) (P : PhysicalBinding d a alloc m x p)
    (i k : Nat) (hi : i<a.numel) (hk : k<bytes d) :
    p.input.Contains (p.inputPointer.toNat+inputOffset (parameters a alloc x) i*bytes d+k) ∧
    p.inputPointer.toNat+inputOffset (parameters a alloc x) i*bytes d+k<2^64 ∧
    p.output.Contains (p.output.base+i) ∧ p.output.base+i<2^64 ∧
    p.input.readable alloc.currentDevice ∧ p.inverse.readable alloc.currentDevice ∧
    p.output.writable alloc.currentDevice := by
  have hidx : a.storageOffset+inputOffset (parameters a alloc x) i+1≤a.storageLength := by
    have h:=D.stridedBounds i (by simpa [parameters,Parameters.n,←D.matrixCount] using hi)
    omega
  have hmul := Nat.mul_le_mul_right (bytes d) hidx
  have hiv := P.inputValid
  have hov := P.outputValid
  have hoc := P.outputCapacity
  rw [ByteRegion.Valid] at hiv hov
  rw [P.inputCapacity] at hiv
  rw [P.inputPointer]
  refine ⟨?_,?_,?_,?_,P.inputKernelRead,P.inverseKernelRead,P.outputKernelWrite⟩
  · constructor
    · omega
    · rw [P.inputCapacity]
      simp only [Nat.add_mul,Nat.one_mul] at hmul
      omega
  · simp only [Nat.add_mul,Nat.one_mul] at hmul
    omega
  · constructor <;> omega
  · omega

/-- Little-endian representation of a typed cell; absent/wrongly typed cells
remain absent, rather than being totalized to a numerical zero. -/
def cellByte (d : Storage) (c : Option Cell) (k : Nat) : Option (BitVec 8) :=
  (readCell d c).map (fun w => byteOf w k)

def inRange (base length address : Nat) : Prop := base≤address ∧ address<base+length
instance (base length address : Nat) : Decidable (inRange base length address) :=
  inferInstanceAs (Decidable (base≤address ∧ address<base+length))

/-- External representation of only the source-owned write footprint. This
is a representation map on the proved logical execution, not a second kernel
or a callback that selects final values. Every other physical byte is framed. -/
def externalBytes (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (ret : Returned) : ByteMemory := fun address =>
  if inRange p.output.base a.numel address then
    cellByte .i8 (ret.memory alloc.output (address-p.output.base)) 0
  else if inRange p.maximum.base (bytes d) address then
    cellByte d (ret.memory alloc.maximum 0) (address-p.maximum.base)
  else if inRange p.inverse.base (bytes (compute d)) address then
    cellByte (compute d) (ret.memory alloc.scale 0) (address-p.inverse.base)
  else p.initialBytes address

/-- Independent physical result: exact source-owned output bytes and raw
frame, plus all metadata including current-device allocation and unsqueeze alias. -/
def PhysicalResult (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations)
    (p : PhysicalLayout) (xs : Nat → Word d) (maximum : Word d)
    (ret : Returned) (memory : ByteMemory) : Prop :=
  ret.tensors=(outputTensor a alloc,maximumTensor d a alloc) ∧
  (∀ i, i<a.numel → memory (p.output.base+i)=result b d (xs i) (b.reciprocal d maximum)) ∧
  (∀ k, k<bytes d → memory (p.maximum.base+k)=some (byteOf maximum k)) ∧
  (∀ k, k<bytes (compute d) → memory (p.inverse.base+k)=some (byteOf (b.reciprocal d maximum) k)) ∧
  (∀ address, p.input.Contains address → memory address=p.initialBytes address) ∧
  (∀ address, ¬inRange p.output.base a.numel address →
    ¬inRange p.maximum.base (bytes d) address →
    ¬inRange p.inverse.base (bytes (compute d)) address → memory address=p.initialBytes address)

theorem cell_output_byte (v : Option (BitVec 8)) :
    cellByte .i8 (v.map (fun z => Cell.mk .i8 z)) 0=v := by
  cases v with
  | none => rfl
  | some z => simp [cellByte,readCell,byteOf,Storage.width]; rfl

theorem disjoint_range (r s : ByteRegion) (length address : Nat)
    (sep : r.Disjoint s) (cap : length≤s.capacity) (inside : r.Contains address) :
    ¬inRange s.base length address := by
  unfold ByteRegion.Disjoint at sep
  unfold ByteRegion.Contains at inside
  unfold inRange
  rcases sep with h|h <;> omega

theorem disjoint_symm (r s : ByteRegion) (h : r.Disjoint s) : s.Disjoint r := h.symm

specification quant_transpose_external_result
    (b : Numerics) (d : Storage) (a : Input) (alloc : Allocations) (m : Memory) (x : String)
    (p : PhysicalLayout) (xs : Nat → Word d) (maximum : Word d) (ret : Returned)
    (count : a.numel=a.M*a.N) (P : PhysicalBinding d a alloc m x p) (R : Result b d a alloc m xs maximum ret) :
    PhysicalResult b d a alloc p xs maximum ret (externalBytes d a alloc p ret) := by
  refine ⟨R.1,?_,?_,?_,?_,?_⟩
  · intro i hi
    simp only [externalBytes,inRange,Nat.le_add_right,Nat.add_lt_add_iff_left,hi,and_self,if_pos,
      Nat.add_sub_cancel_left,R.2.1 i (by simpa [parameters,Parameters.n,←count] using hi),outCell]
    exact cell_output_byte _
  · intro k hk
    have inside : p.maximum.Contains (p.maximum.base+k) := by
      unfold ByteRegion.Contains
      have h:=P.maximumCapacity
      constructor <;> omega
    have ho := disjoint_range p.maximum p.output _ _ P.maximumOutput P.outputCapacity inside
    have hm : inRange p.maximum.base (bytes d) (p.maximum.base+k) := by
      unfold inRange; constructor <;> omega
    simp [externalBytes,ho,hm,R.2.2.1,cellByte,read_same]
  · intro k hk
    have inside : p.inverse.Contains (p.inverse.base+k) := by
      unfold ByteRegion.Contains
      have h:=P.inverseCapacity
      constructor <;> omega
    have ho := disjoint_range p.inverse p.output _ _ P.inverseOutput P.outputCapacity inside
    have hm := disjoint_range p.inverse p.maximum _ _
      (disjoint_symm _ _ P.maximumInverse) P.maximumCapacity inside
    have hv : inRange p.inverse.base (bytes (compute d)) (p.inverse.base+k) := by
      unfold inRange; constructor <;> omega
    simp [externalBytes,ho,hm,hv,R.2.2.2.1,cellByte,read_same]
  · intro address inside
    have ho := disjoint_range p.input p.output _ _ P.inputOutput P.outputCapacity inside
    have hm := disjoint_range p.input p.maximum _ _ P.inputMaximum P.maximumCapacity inside
    have hv := disjoint_range p.input p.inverse _ _ P.inputInverse P.inverseCapacity inside
    simp [externalBytes,ho,hm,hv]
  · intro address ho hm hv
    simp [externalBytes,ho,hm,hv]

specification quant_transpose_scalar_physical_accesses
    (d : Storage) (a : Input) (alloc : Allocations) (m : Memory) (x : String)
    (p : PhysicalLayout) (P : PhysicalBinding d a alloc m x p) (k : Nat)
    (hk : k<bytes (compute d)) :
    p.inverse.Contains (p.inverse.base+k) ∧ p.inverse.base+k<2^64 ∧
    p.inverse.readable alloc.currentDevice := by
  have hc:=P.inverseCapacity
  have hv:=P.inverseValid
  unfold ByteRegion.Valid at hv
  exact ⟨⟨by omega,by omega⟩,by omega,P.inverseKernelRead⟩
