import bench.tritonbench_g.vector_addition.Proofs.Binary64Words
import Mathlib.Data.List.Range

/-! Typed scalar source for every admitted CUDA signature in the inspected
Torch default-float32 runtime. Native arithmetic is an explicit primitive
boundary; undefined llrint has no invented result. -/
namespace QuantizeGlobalFinite
open VeriTile.Triton
set_option maxRecDepth 16000
set_option maxHeartbeats 4000000

inductive Storage where
  | boolean | u8 | i8 | i16 | i32 | i64 | fp32 | fp64
  deriving DecidableEq, Repr

def Storage.width : Storage → Nat
  | .boolean | .u8 | .i8 => 8 | .i16 => 16 | .i32 | .fp32 => 32 | .i64 | .fp64 => 64

abbrev Word (d : Storage) := BitVec d.width

def compute : Storage → Storage | .fp64 => .fp64 | _ => .fp32

structure Numerics where
  /-- Each host primitive consumes exactly its source-produced typed arguments. -/
  abs : (d : Storage) → Word d → Word d
  maximum : (d : Storage) → Word d → List (Word d) → Word d
  reciprocal : (d : Storage) → Word d → Word (compute d)
  unsignedTo32 : Word .u8 → Word .fp32
  signed8To32 : Word .i8 → Word .fp32
  signed16To32 : Word .i16 → Word .fp32
  signed32To32 : Word .i32 → Word .fp32
  signed64To32 : Word .i64 → Word .fp32
  mul32 : Word .fp32 → Word .fp32 → Word .fp32
  mul64 : Word .fp64 → Word .fp64 → Word .fp64

def promoted (b : Numerics) : (d : Storage) → Word d → Word (compute d)
  | .boolean, x => if x=0 then 0 else 0x3f800000
  | .u8, x => b.unsignedTo32 x
  | .i8, x => b.signed8To32 x
  | .i16, x => b.signed16To32 x
  | .i32, x => b.signed32To32 x
  | .i64, x => b.signed64To32 x
  | .fp32, x => x
  | .fp64, x => x

def mul (b : Numerics) : (d : Storage) → Word (compute d) → Word (compute d) → Word (compute d)
  | .fp64 => b.mul64 | _ => b.mul32

def scale127 : (d : Storage) → Word (compute d)
  | .fp64 => 0x405fc00000000000 | _ => 0x42fe0000

def projection : (d : Storage) → Word (compute d) → Option Rat
  | .fp64, x => VeriTile.Bench.TritonBenchG.VectorAddition.Binary64Words.toRat? x
  | _, x => (FP.Value.mk x : FP.Value .fp32).toRat?

/-- Nearest integer, with ties to even, on the exact finite rational decoded
from the floating word. Range is checked after rounding, before narrowing. -/
def nearestEven (q : Rat) : Int :=
  let n := q.num.natAbs / q.den
  let r := q.num.natAbs % q.den
  let magnitude := if 2*r>q.den ∨ (2*r=q.den ∧ n%2=1) then n+1 else n
  if q.num<0 then -(magnitude : Int) else magnitude

def llrint (d : Storage) (x : Word (compute d)) : Option (BitVec 64) := do
  let q ← projection d x
  let z := nearestEven q
  if -(2^63 : Int)≤z ∧ z<2^63 then some (BitVec.ofInt 64 z) else none

/-- The intended conversion domain is explicit, independent of any execution
success assumption. NaN, infinity and rounded out-of-range values are excluded. -/
def LlrintDefined (d : Storage) (x : Word (compute d)) : Prop :=
  ∃ q, projection d x=some q ∧ -(2^63 : Int)≤nearestEven q ∧ nearestEven q<2^63

theorem llrint_defined_iff (d : Storage) (x : Word (compute d)) :
    (llrint d x).isSome=true ↔ LlrintDefined d x := by
  cases hp : projection d x with
  | none => simp [llrint,LlrintDefined,hp]
  | some q => by_cases h : -(2^63 : Int)≤nearestEven q ∧ nearestEven q<2^63 <;>
      simp [llrint,LlrintDefined,hp,h]

inductive RegType (d : Storage) where
  | input | arithmetic | integer64 | output8

def RegType.width {d : Storage} : RegType d → Nat
  | .input => d.width | .arithmetic => (compute d).width | .integer64 => 64 | .output8 => 8

inductive Expr (d : Storage) : RegType d → Type where
  | x : Expr d .input
  | scale : Expr d .arithmetic
  | literal127 : Expr d .arithmetic
  | promote : Expr d .input → Expr d .arithmetic
  | multiply : Expr d .arithmetic → Expr d .arithmetic → Expr d .arithmetic
  | round64 : Expr d .arithmetic → Expr d .integer64
  | narrow8 : Expr d .integer64 → Expr d .output8

def Expr.eval {d : Storage} (b : Numerics) (x : Word d) (inverse : Word (compute d)) :
    Expr d t → Option (BitVec t.width)
  | .x => some x
  | .scale => some inverse
  | .literal127 => some (scale127 d)
  | .promote e => (e.eval b x inverse).map (promoted b d)
  | .multiply a c => do pure (mul b d (← a.eval b x inverse) (← c.eval b x inverse))
  | .round64 e => (e.eval b x inverse).bind (llrint d)
  | .narrow8 e => (e.eval b x inverse).map (fun z => BitVec.ofInt 8 z.toInt)

/-- Exact original parenthesization and destination conversion; no float
output is substituted for the source's llrint64 then int8 store. -/
def source (d : Storage) : Expr d .output8 :=
  .narrow8 (.round64 (.multiply .literal127 (.multiply (.promote .x) .scale)))

def scaled (b : Numerics) (d : Storage) (x : Word d) (inverse : Word (compute d)) :=
  mul b d (scale127 d) (mul b d (promoted b d x) inverse)

def result (b : Numerics) (d : Storage) (x : Word d) (inverse : Word (compute d)) : Option (BitVec 8) :=
  (llrint d (scaled b d x inverse)).map (fun z => BitVec.ofInt 8 z.toInt)

theorem source_scalar (b : Numerics) (d : Storage) (x : Word d) (inverse : Word (compute d)) :
    (source d).eval b x inverse=result b d x inverse := by rfl

theorem defined_result (b : Numerics) (d : Storage) (x : Word d) (inverse : Word (compute d)) :
    (result b d x inverse).isSome=true ↔ LlrintDefined d (scaled b d x inverse) := by
  simpa [result] using llrint_defined_iff d (scaled b d x inverse)

theorem special_conversions :
    llrint .fp32 0x7f800000=none ∧ llrint .fp32 0x7fc00000=none ∧
    llrint .fp64 0x7ff0000000000000=none ∧ llrint .fp64 0x7ff8000000000000=none ∧
    llrint .fp32 0x5f000000=none ∧ llrint .fp64 0x43e0000000000000=none := by
  decide +kernel

theorem conversion_boundaries :
    llrint .fp32 0x3f000000=some 0 ∧ llrint .fp32 0x3fc00000=some 2 ∧
    llrint .fp32 0xbfc00000=some (BitVec.ofInt 64 (-2)) ∧
    llrint .fp64 0xc3e0000000000000=some (BitVec.ofInt 64 (-(2^63))) := by
  decide +kernel

#specNonCircular result avoiding [source,Expr.eval]
#specNonCircular LlrintDefined avoiding [source,Expr.eval,llrint]
#axiomsClean source_scalar
#axiomsClean defined_result
#axiomsClean special_conversions
#axiomsClean conversion_boundaries
#guard_msgs (drop info) in
#auditModuleAxioms
end QuantizeGlobalFinite
