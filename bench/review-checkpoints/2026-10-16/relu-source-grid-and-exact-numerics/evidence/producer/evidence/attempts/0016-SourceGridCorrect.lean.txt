import bench.tritonbench_g.relu_triton_kernel.Proofs.CheckedSchedule

/-! Whole-grid composition of checked source blocks in every complete order.
All pids execute their load and result statements before the original store
gate; only pid zero stores. Native instruction interleavings are not modeled. -/
namespace VeriTile.Bench.TritonBenchG.ReluTritonKernel.SourceGridCorrect
open VeriTile.Triton FP.Structural IntegerWords FiniteAddress TypedValue SourceStages
open AccessMemory CheckedBlock CheckedSchedule
noncomputable section

specification relu_checked_source_grid_correctness
    (M : Algebra (BitVec 32)) (cmp16 cmp32 : BitVec 32 → BitVec 32 → Bool)
    (e : Encoding) (x y : RegionName) (inputBase outputBase N : Nat)
    (order : List Nat) (complete : order.Perm (List.range (GatedGrid.gridCount N blockSize)))
    (initial : State (BitVec 32)) (access : Memory)
    (view : ByteMemory.LogicalView e access.bytes inputBase outputBase N x y initial.mem)
    (distinct : x ≠ y) (separate : ByteMemory.Separate e inputBase outputBase N)
    (inputReadable : Readable access inputBase (N * elementBytes e))
    (outputWritable : Writable access outputBase (min N blockSize * 4)) :
    ∃ traces logicalFinal final,
      CheckedSchedule.runSchedule M cmp16 cmp32 e x y inputBase outputBase N order initial access =
        .ok (traces, logicalFinal, final) ∧
      SourceSchedule M cmp16 cmp32 e x y N order initial traces logicalFinal ∧
      (traces, logicalFinal) = scheduleSpec M e x y N order initial ∧
      traces.length = GatedGrid.gridCount N blockSize ∧
      logicalFinal.mem = wordSpec M e (ByteMemory.loadInput e access.bytes inputBase) y N blockSize initial.mem ∧
      final.bytes = ByteMemory.byteGrid M e inputBase outputBase N access.bytes ∧
      final.bytes = ByteMemory.byteSpec M e inputBase outputBase N access.bytes ∧
      ByteMemory.LogicalView e final.bytes inputBase outputBase N x y logicalFinal.mem ∧
      SameAccess access final ∧
      AccessMemory.runSchedule M e inputBase outputBase N order access = .ok final ∧
      (∀ address, address < outputBase ∨ outputBase + min N blockSize * 4 ≤ address →
        final.bytes address = access.bytes address ∧ partialContents final address = partialContents access address) := by
  let traces := (scheduleSpec M e x y N order initial).1
  let logicalFinal := (scheduleSpec M e x y N order initial).2
  let final := withBytes access (ByteMemory.byteGrid M e inputBase outputBase N access.bytes)
  have memory := complete_schedule_memory M cmp16 cmp32 e x y inputBase N order complete initial access view.1
  refine ⟨traces, logicalFinal, final,
    CheckedSchedule.complete_schedule_success M cmp16 cmp32 e x y inputBase outputBase N order complete
      initial access view.1 distinct separate inputReadable outputWritable,
    schedule_spec_trace M cmp16 cmp32 e x y N order initial, Prod.eta _, ?_, memory, rfl,
    ByteMemory.byte_grid_correct M e inputBase outputBase N access.bytes, ?_,
    with_bytes_access access _,
    AccessMemory.complete_schedule_success M e inputBase outputBase N order complete access inputReadable outputWritable,
    ?_⟩
  · exact (schedule_spec_length M e x y N order initial).trans (complete.length_eq.trans (List.length_range _))
  · change ByteMemory.LogicalView e (ByteMemory.byteGrid M e inputBase outputBase N access.bytes)
      inputBase outputBase N x y (scheduleSpec M e x y N order initial).2.mem
    rw [memory]
    exact ByteMemory.word_spec_view M e inputBase outputBase N x y access.bytes initial.mem distinct separate view
  · intro address outside
    exact ⟨ByteMemory.byte_grid_frame M e inputBase outputBase N address access.bytes outside,
      AccessMemory.grid_partial_frame M e inputBase outputBase N address access outside⟩

#axiomsClean relu_checked_source_grid_correctness
#guard_msgs (drop info) in
#auditModuleAxioms
end
end VeriTile.Bench.TritonBenchG.ReluTritonKernel.SourceGridCorrect
