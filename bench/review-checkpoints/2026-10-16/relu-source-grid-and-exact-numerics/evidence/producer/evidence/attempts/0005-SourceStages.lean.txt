import bench.tritonbench_g.relu_triton_kernel.Proofs.AccessGrid

/-! Exact register states for each of the seven statements of the frozen
typed original body. All nine integer/bool profiles execute the comparison,
conversion and selection before the pid-zero store gate. Inactive loads retain
the initial undef value; the source has no `other=0` argument. -/
namespace VeriTile.Bench.TritonBenchG.ReluTritonKernel.SourceStages
open VeriTile.Triton FP.Structural IntegerWords FiniteAddress TypedValue
noncomputable section
set_option maxHeartbeats 200000
set_option maxRecDepth 24000

def statement (e : Encoding) (x y : RegionName) (N B : Nat) (index : Fin 7) : ComputeStmt :=
  (TypedValue.source e.profile x y N B).surfaceBody[index.val]'(by
    simp only [TypedValue.source, materialized, ComputeKernel.surfaceBody, List.length_cons,
      List.length_nil]
    exact index.isLt)

structure Trace (α : Type) where
  pid : State α
  start : State α
  offsets : State α
  mask : State α
  loaded : State α
  result : State α
  final : State α

def Valid {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) (trace : Trace α) : Prop :=
  step M (statement e x y N B 0) initial = some trace.pid ∧
  step M (statement e x y N B 1) trace.pid = some trace.start ∧
  step M (statement e x y N B 2) trace.start = some trace.offsets ∧
  step M (statement e x y N B 3) trace.offsets = some trace.mask ∧
  step M (statement e x y N B 4) trace.mask = some trace.loaded ∧
  step M (statement e x y N B 5) trace.loaded = some trace.result ∧
  step M (statement e x y N B 6) trace.result = some trace.final

def pidState {α : Type} (initial : State α) : State α :=
  initial.setReg "pid" .nat [] (fun _ => initial.pids 0)

def startState {α : Type} (B : Nat) (initial : State α) : State α :=
  (pidState initial).setReg "block_start" .nat [] (fun _ => initial.pids 0 * B)

def offsetValues {α : Type} (B : Nat) (initial : State α) : Values α .nat [B] :=
  fun i => initial.pids 0 * B + i.1.val

def offsetsState {α : Type} (B : Nat) (initial : State α) : State α :=
  (startState B initial).setReg "offsets" .nat [B] (offsetValues B initial)

def maskValues {α : Type} (N B : Nat) (initial : State α) : Values α .bool [B] :=
  fun i => decide (offsetValues B initial i < N)

def maskState {α : Type} (N B : Nat) (initial : State α) : State α :=
  (offsetsState B initial).setReg "mask" .bool [B] (maskValues N B initial)

def loadedValues {α : Type} [Inhabited α] (e : Encoding) (x : RegionName)
    (N B : Nat) (initial : State α) : Values α e.kind.inputType [B] :=
  fun i => if offsetValues B initial i < N then
    (initial.mem x (offsetValues B initial i)).read e.kind.inputType
  else initial.undef e.kind.inputType x (offsetValues B initial i)

def loadedState {α : Type} [Inhabited α] (e : Encoding) (x : RegionName)
    (N B : Nat) (initial : State α) : State α :=
  (maskState N B initial).setReg "x" e.kind.inputType [B] (loadedValues e x N B initial)

/-- Scalar reference, including the value of masked inactive lanes. No
interpreter, source AST, byte load or checked execution occurs here. -/
def scalar {α : Type} (M : Algebra α) (e : Encoding) : Value α e.kind.inputType → α :=
  match e with
  | .signed _ => fun (value : Int) => if 0 ≤ value then M.fromInt none value else M.literal none .fp32 0
  | .unsigned _ => fun (value : Nat) => M.fromNat none value
  | .boolean => fun (value : Bool) => M.fromNat none (if value then 1 else 0)

def resultValues {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x : RegionName) (N B : Nat) (initial : State α) : Values α .fp32 [B] :=
  fun i => scalar M e (loadedValues e x N B initial i)

def resultState {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x : RegionName) (N B : Nat) (initial : State α) : State α :=
  (loadedState e x N B initial).setReg "result" .fp32 [B] (resultValues M e x N B initial)

def finalState {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) : State α :=
  if initial.pids 0 = 0 then
    (TileShape.allIndices [B]).foldl (fun current i =>
      if offsetValues B initial i < N then
        current.write y (offsetValues B initial i) (.mk .fp32 (resultValues M e x N B initial i))
      else current) (resultState M e x N B initial)
  else resultState M e x N B initial

def expected {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) : Trace α :=
  ⟨pidState initial, startState B initial, offsetsState B initial, maskState N B initial,
    loadedState e x N B initial, resultState M e x N B initial, finalState M e x y N B initial⟩

theorem prefix_steps {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) :
    step M (statement e x y N B 0) initial = some (pidState initial) ∧
    step M (statement e x y N B 1) (pidState initial) = some (startState B initial) ∧
    step M (statement e x y N B 2) (startState B initial) = some (offsetsState B initial) ∧
    step M (statement e x y N B 3) (offsetsState B initial) = some (maskState N B initial) := by
  refine ⟨rfl, rfl, rfl, ?_⟩
  change some ((offsetsState B initial).setReg "mask" .bool [B]
    (fun i => decide (initial.pids 0 * B + i.1.val < N))) = _
  rfl

theorem load_step {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) :
    step M (statement e x y N B 4) (maskState N B initial) =
      some (loadedState e x N B initial) := by
  unfold statement TypedValue.source
  rw [profile_kind]
  change some ((maskState N B initial).setReg "x" e.kind.inputType [B]
    (fun i => if decide (offsetValues B initial i < N) then
      (initial.mem x (offsetValues B initial i)).read e.kind.inputType
      else initial.undef e.kind.inputType x (offsetValues B initial i))) = _
  simp only [decide_eq_true_eq, loadedState, loadedValues]

theorem result_step {α : Type} [Inhabited α] (M : Algebra α)
    (cmp16 cmp32 : α → α → Bool) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) :
    step (backend M cmp16 cmp32) (statement e x y N B 5) (loadedState e x N B initial) =
      some (resultState M e x N B initial) := by
  unfold statement TypedValue.source
  rw [profile_kind]
  cases e <;> simp [materialized, ComputeKernel.surfaceBody,
    step, evalExpr, evalOp_unfold, resultOp, boolNat, resultState, resultValues,
    loadedState, Encoding.kind, Kind.inputType, Kind.resultType, backend, scalar,
    numericLe, natLe, FP.Structural.bop, ofFloat, toFloat]

theorem gate_step {α : Type} [Inhabited α] (M : Algebra α)
    (cmp16 cmp32 : α → α → Bool) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) :
    step (backend M cmp16 cmp32) (statement e x y N B 6) (resultState M e x N B initial) =
      some (finalState M e x y N B initial) := by
  cases e <;> by_cases first : initial.pids 0 = 0
  all_goals simp [first, statement, TypedValue.source, materialized, ComputeKernel.surfaceBody,
    step, evalExpr, evalOp_unfold, run, storedOp, resultState, loadedState,
    maskState, offsetsState, startState, pidState, maskValues, offsetValues,
    Encoding.kind, Kind.resultType, numericEq, FP.Structural.bop, store, finalState, Region.cast]

theorem source_trace {α : Type} [Inhabited α] (M : Algebra α)
    (cmp16 cmp32 : α → α → Bool) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) :
    Valid (backend M cmp16 cmp32) e x y N B initial (expected M e x y N B initial) := by
  have firstFour := prefix_steps (backend M cmp16 cmp32) e x y N B initial
  exact ⟨firstFour.1, firstFour.2.1, firstFour.2.2.1, firstFour.2.2.2,
    load_step _ e x y N B initial, result_step M cmp16 cmp32 e x y N B initial,
    gate_step M cmp16 cmp32 e x y N B initial⟩

theorem valid_exec {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x y : RegionName) (N B : Nat) (initial : State α) (trace : Trace α)
    (valid : Valid M e x y N B initial trace) :
    FP.Structural.exec M (TypedValue.source e.profile x y N B) initial = some trace.final := by
  rcases valid with ⟨p, s, o, m, l, r, f⟩
  change run M [statement e x y N B 0, statement e x y N B 1,
    statement e x y N B 2, statement e x y N B 3, statement e x y N B 4,
    statement e x y N B 5, statement e x y N B 6] initial = _
  simp only [run, p, s, o, m, l, r, f, bind, Option.bind]

theorem result_register {α : Type} [Inhabited α] (M : Algebra α) (e : Encoding)
    (x : RegionName) (N B : Nat) (initial : State α) :
    (resultState M e x N B initial).regs .fp32 [B] "result" =
      some (resultValues M e x N B initial) := State.setReg_same _ _ _ _ _

theorem inactive_load {α : Type} [Inhabited α] (e : Encoding) (x : RegionName)
    (N B : Nat) (initial : State α) (i : TileIndex [B]) (inactive : N ≤ offsetValues B initial i) :
    loadedValues e x N B initial i = initial.undef e.kind.inputType x (offsetValues B initial i) := by
  exact if_neg (Nat.not_lt.mpr inactive)

#specNonCircular scalar avoiding [TypedValue.source, materialized, FP.Structural.exec, step]
#specNonCircular expected avoiding [TypedValue.source, materialized, FP.Structural.exec, step]
#axiomsClean prefix_steps
#axiomsClean load_step
#axiomsClean result_step
#axiomsClean gate_step
#axiomsClean source_trace
#axiomsClean valid_exec
#axiomsClean result_register
#axiomsClean inactive_load
#guard_msgs (drop info) in
#auditModuleAxioms
end
