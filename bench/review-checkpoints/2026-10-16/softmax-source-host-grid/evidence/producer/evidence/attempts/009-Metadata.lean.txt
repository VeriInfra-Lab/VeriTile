/-- Exact assertion and Python modulo normalization for a canonical integer dim. -/
def normalizeDim (shape : List Nat) (dim : Int) : Option (Fin shape.length) :=
  if h : -(shape.length : Int) ≤ dim ∧ dim < (shape.length : Int) then
    have hp : (0:Int) < shape.length := by omega
    have hn := Int.emod_nonneg dim (by omega : (shape.length : Int) ≠ 0)
    have hl := Int.emod_lt_of_pos dim hp
    some ⟨(dim % (shape.length : Int)).toNat, by omega⟩
  else none

structure Dimensions where
  M : Nat
  N : Nat
  K : Nat
  deriving DecidableEq, Repr

def dimensions (shape : List Nat) (axis : Fin shape.length) : Dimensions :=
  let M := (shape.take axis.val).prod
  let N := shape[axis.val]
  ⟨M,N,shape.prod/M/N⟩

theorem normalize_none (shape : List Nat) (dim : Int) :
    normalizeDim shape dim = none ↔
      ¬ (-(shape.length : Int) ≤ dim ∧ dim < (shape.length : Int)) := by
  unfold normalizeDim
  split_ifs <;> simp_all

theorem shape_factorization (shape : List Nat) (axis : Fin shape.length) :
    shape.prod=(shape.take axis.val).prod*shape[axis.val]*(shape.drop (axis.val+1)).prod := by
  have h := List.prod_take_mul_prod_drop shape axis.val
  rw [List.drop_eq_getElem_cons axis.isLt,List.prod_cons] at h
  simpa [Nat.mul_assoc] using h.symm

theorem dimensions_suffix (shape : List Nat) (axis : Fin shape.length)
    (hM : 0<(shape.take axis.val).prod) (hN : 0<shape[axis.val]) :
    (dimensions shape axis).K=(shape.drop (axis.val+1)).prod := by
  simp only [dimensions]
  rw [shape_factorization shape axis]
  rw [Nat.mul_assoc,Nat.mul_div_right _ hM,Nat.mul_div_right _ hN]

theorem dimensions_numel (shape : List Nat) (axis : Fin shape.length)
    (hM : 0<(shape.take axis.val).prod) (hN : 0<shape[axis.val]) :
    shape.prod=(dimensions shape axis).M*(dimensions shape axis).N*(dimensions shape axis).K := by
  rw [dimensions_suffix shape axis hM hN]
  exact shape_factorization shape axis

/-- This is the actual Python scalar primitive `(num_blocks / NUM_SMS) > 1`.
The correctness proof is uniform in its result: its only effect is choosing a
legal tile. It is not a kernel execution or output-value callback. -/
abbrev WavesAboveOne := Nat → Nat → Bool

def tileKLoop (waves : WavesAboveOne) (M K sms tile : Nat) (ht : 0<tile) : Nat :=
  if h : tile ≤ min K 8192 ∧ waves (M*RealAddress.rounds K tile) sms = Bool.true ∧
      tile*2 ≤ min K 8192 then
    tileKLoop waves M K sms (tile*2) (by omega)
  else tile
termination_by 8193-tile
decreasing_by
  have hm := Nat.min_le_right K 8192
  omega

def heurTileK (waves : WavesAboveOne) (M K sms : Nat) : Nat :=
  tileKLoop waves M K sms 1 (by decide)

theorem tileKLoop_positive (waves : WavesAboveOne) (M K sms tile : Nat) (ht : 0<tile) :
    0<tileKLoop waves M K sms tile ht := by
  fun_induction tileKLoop <;> simp_all

theorem tileKLoop_bound (waves : WavesAboveOne) (M K sms tile : Nat) (ht : 0<tile)
    (hb : tile≤8192) : tileKLoop waves M K sms tile ht≤8192 := by
  fun_induction tileKLoop
  · apply_assumption
    exact h.2.2.trans (Nat.min_le_right K 8192)
  · exact hb

theorem tileKLoop_power (waves : WavesAboveOne) (M K sms tile : Nat) (ht : 0<tile)
    (hp : tile.isPowerOfTwo) : (tileKLoop waves M K sms tile ht).isPowerOfTwo := by
  fun_induction tileKLoop
  · apply_assumption
    exact Nat.isPowerOfTwo_mul_two_of_isPowerOfTwo hp
  · exact hp

def heurTileNInner (N : Nat) : Nat :=
  if N≤32768 then N.nextPowerOfTwo else 4096

def numWarps (tileSize : Nat) : Nat :=
  if tileSize<2048 then 4 else if tileSize<4096 then 8 else 16

def oneTile (N B : Nat) : Bool := decide (N≤B)

/-- Exactly the six source-declared autotune configurations. -/
def autotuneTile (choice : Fin 6) : Nat := 32*2^choice.val

structure LaunchConfig where
  B : Nat
  C : Nat
  one : Bool
  warps : Nat

def forwardConfig (waves : WavesAboveOne) (dims : Dimensions) (sms : Nat) : LaunchConfig :=
  if 1<dims.K then
    let C := heurTileK waves dims.M dims.K sms
    let B := RealAddress.rounds 8192 C
    ⟨B,C,oneTile dims.N B,numWarps (B*C)⟩
  else
    let B := heurTileNInner dims.N
    ⟨B,1,oneTile dims.N B,numWarps B⟩

def backwardConfig (dims : Dimensions) (choice : Fin 6) : LaunchConfig :=
  if 1<dims.K then
    let C := autotuneTile choice
    let B := max 1 (1024/C)
    ⟨B,C,oneTile dims.N B,4⟩
  else
    let B := autotuneTile choice
    let C := max 1 (1024/B)
    ⟨B,C,oneTile dims.N B,4⟩

theorem forwardConfig_positive (waves : WavesAboveOne) (dims : Dimensions) (sms : Nat) :
    0<(forwardConfig waves dims sms).B ∧ 0<(forwardConfig waves dims sms).C := by
  unfold forwardConfig
  split_ifs
  · have hC : 0<heurTileK waves dims.M dims.K sms := tileKLoop_positive _ _ _ _ _ _
    exact ⟨RealAddress.rounds_positive (by decide) hC,hC⟩
  · refine ⟨?_,by decide⟩
    unfold heurTileNInner
    split_ifs
    · exact Nat.pos_of_isPowerOfTwo (Nat.isPowerOfTwo_nextPowerOfTwo _)
    · decide

theorem backwardConfig_positive (dims : Dimensions) (choice : Fin 6) :
    0<(backwardConfig dims choice).B ∧ 0<(backwardConfig dims choice).C := by
  have hp : 0<autotuneTile choice := Nat.mul_pos (by decide) (Nat.two_pow_pos _)
  unfold backwardConfig
  split_ifs <;> simp only
  · exact ⟨Nat.lt_of_lt_of_le (by decide) (Nat.le_max_left _ _),hp⟩
  · exact ⟨hp,Nat.lt_of_lt_of_le (by decide) (Nat.le_max_left _ _)⟩

theorem forwardConfig_one (waves : WavesAboveOne) (dims : Dimensions) (sms : Nat) :
    (forwardConfig waves dims sms).one = oneTile dims.N (forwardConfig waves dims sms).B := by
  unfold forwardConfig
  split_ifs <;> rfl

theorem backwardConfig_one (dims : Dimensions) (choice : Fin 6) :
    (backwardConfig dims choice).one = oneTile dims.N (backwardConfig dims choice).B := by
  unfold backwardConfig
  split_ifs <;> rfl
