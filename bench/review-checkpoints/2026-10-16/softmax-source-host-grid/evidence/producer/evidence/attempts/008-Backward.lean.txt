import SoftmaxHost.Core

namespace VeriTile.Bench.TritonBenchG.SoftmaxFlaggems.HostGrid
open VeriTile Triton FullSource
set_option maxHeartbeats 2000000

/-- The complete initial-memory formula retains every source padding sample.
No zero-fill or finite-padding hypothesis is introduced. -/
noncomputable def backwardInnerValue (base : BlockState) (P G : RegionName)
    (M N B : Nat) (one : Bool) (a : MemCellAddr) : ℝ :=
  RealBackwardInner.expected base P G M N (RealBackwardInner.paddedLength one N B)
    (a.2/N) (a.2%N)

noncomputable def backwardNonInnerValue (base : BlockState) (P G : RegionName)
    (N K B : Nat) (one : Bool) (a : MemCellAddr) : ℝ :=
  RealBackwardNonInner.expected base P G (a.2/K/N) N K
    (RealBackwardNonInner.paddedLength one N B) (a.2/K%N) (a.2%K)

theorem backward_inner_grid (storage : Storage) (base : BlockState)
    (P G Y : RegionName) (M N B C : Nat) (one : Bool)
    (h : RealBackwardInner.ExecutionDomain one P G Y N B)
    (hPY : P≠Y) (hGY : G≠Y) (hC : 0<C)
    (ps : List Program) (schedule : Schedule (RealAddress.rounds M C) 1 ps) :
    ∃ final, run (softmax_backward_kernel_inner storage P G Y M N C B one).eraseDType.toAlgKernel
        (RealAddress.rounds M C) 1 ps base=some final ∧
      BlockState.WriteWithin (fullSlice Y (M*N)) base final ∧
      (∀ r n, r<M → n<N → final.readMem Y (r*N+n)=
        RealBackwardInner.expected base P G M N (RealBackwardInner.paddedLength one N B) r n) ∧
      final.undef=base.undef := by
  let F : Program → WriteFootprint := fun p ↦ RealBackwardInner.outputFootprint Y M N (p.1*C) C
  let val := backwardInnerValue base P G M N B one
  have step : ∀ p s, Preserved Y base s →
      ∃ t, exec (softmax_backward_kernel_inner storage P G Y M N C B one).eraseDType.toAlgKernel
          (setup (RealAddress.rounds M C) 1 p s)=some t ∧
        BlockState.WriteWithin (F p) s t ∧
        (∀ a, F p a → t.readMem a.1 a.2=val a) ∧ t.undef=s.undef := by
    intro p s hs
    obtain ⟨t,he,hframe,hvalue,_,_,hundef⟩ :=
      RealBackwardInner.original_all_io storage (setup (RealAddress.rounds M C) 1 p s)
        P G Y M N B C one h
    refine ⟨t,he,?_,?_,hundef⟩
    · simpa only [F,RealBackwardInner.outputFootprint,setup,ite_true] using hframe
    · rintro a ⟨hy,r,n,hr,hM,hn,ha⟩
      have got := hvalue r n hr (by simpa [setup] using hM) hn
      simp only [setup,ite_true] at got
      have transport : RealBackwardInner.expected (setup (RealAddress.rounds M C) 1 p s)
          P G M N (RealBackwardInner.paddedLength one N B) (p.1*C+r) n =
          RealBackwardInner.expected base P G M N
            (RealBackwardInner.paddedLength one N B) (p.1*C+r) n := by
        simp only [RealBackwardInner.expected,RealMath.backwardPadded,RealBackwardInner.sample,
          BlockState.readMem,setup,hs.1 P hPY,hs.1 G hGY,hs.2]
      have stable := got.trans transport
      simpa only [val,backwardInnerValue,ha,hy,RealBackwardInner.offset,
        (decode_inner hn).1,(decode_inner hn).2] using stable
  obtain ⟨final,he,hf,hv,hu⟩ := run_io _ (RealAddress.rounds M C) 1 Y F val base
    (fun _ _ ha ↦ ha.1) step ps base (preserved_refl Y base)
  refine ⟨final,he,frame_mono ?_ hf,?_,hu⟩
  · rintro a ⟨p,_,hy,r,n,_,hr,hn,hoff⟩
    exact ⟨hy,hoff ▸ inner_bound hr hn⟩
  · intro r n hr hn
    have hp : (r/C,0) ∈ ps := (schedule (r/C,0)).2 (mem_grid.mpr ⟨owner_lt hC hr,by decide⟩)
    have lane := Nat.mod_lt r hC
    have off := owner_offset r C
    have got := hv (Y,r*N+n) ⟨(r/C,0),hp,rfl,r%C,n,lane,by simpa [off] using hr,hn,by simp [RealBackwardInner.offset,off]⟩
    simpa only [val,backwardInnerValue,(decode_inner hn).1,(decode_inner hn).2] using got

theorem backward_noninner_grid (storage : Storage) (base : BlockState)
    (P G Y : RegionName) (M N K B C : Nat) (one : Bool)
    (h : RealBackwardNonInner.ExecutionDomain one P G Y N B)
    (hPY : P≠Y) (hGY : G≠Y) (hC : 0<C)
    (ps : List Program) (schedule : Schedule M (RealAddress.rounds K C) ps) :
    ∃ final, run (softmax_backward_kernel_non_inner storage P G Y M N K B C one).eraseDType.toAlgKernel
        M (RealAddress.rounds K C) ps base=some final ∧
      BlockState.WriteWithin (fullSlice Y (M*N*K)) base final ∧
      (∀ r n k, r<M → n<N → k<K → final.readMem Y (r*N*K+n*K+k)=
        RealBackwardNonInner.expected base P G r N K
          (RealBackwardNonInner.paddedLength one N B) n k) ∧
      final.undef=base.undef := by
  let F : Program → WriteFootprint := fun p ↦ RealBackwardNonInner.outputFootprint Y p.1 N K (p.2*C) C
  let val := backwardNonInnerValue base P G N K B one
  have step : ∀ p s, Preserved Y base s →
      ∃ t, exec (softmax_backward_kernel_non_inner storage P G Y M N K B C one).eraseDType.toAlgKernel
          (setup M (RealAddress.rounds K C) p s)=some t ∧
        BlockState.WriteWithin (F p) s t ∧
        (∀ a, F p a → t.readMem a.1 a.2=val a) ∧ t.undef=s.undef := by
    intro p s hs
    obtain ⟨t,he,hframe,hvalue,_,_,hundef⟩ :=
      RealBackwardNonInner.original_all_io storage
        (setup M (RealAddress.rounds K C) p s) P G Y M N K B C one h
    refine ⟨t,he,?_,?_,hundef⟩
    · simpa only [F,RealForwardNonInner.outputFootprint,setup,ite_true,
        Nat.reduceEqDiff,ite_false] using hframe
    · rintro a ⟨hy,n,k,hn,hk,hK,ha⟩
      have got := hvalue n k hn hk (by simpa [setup] using hK)
      simp only [setup,ite_true,Nat.reduceEqDiff,ite_false] at got
      have transport : RealBackwardNonInner.expected (setup M (RealAddress.rounds K C) p s)
          P G p.1 N K (RealBackwardNonInner.paddedLength one N B) n (p.2*C+k) =
          RealBackwardNonInner.expected base P G p.1 N K
            (RealBackwardNonInner.paddedLength one N B) n (p.2*C+k) := by
        simp only [RealBackwardNonInner.expected,RealMath.backwardPadded,RealBackwardNonInner.sample,
          BlockState.readMem,setup,hs.1 P hPY,hs.1 G hGY,hs.2]
      have stable := got.trans transport
      simpa only [val,backwardNonInnerValue,ha,hy,RealBackwardNonInner.offset,
        (decode_noninner hn hK).1,(decode_noninner hn hK).2.1,(decode_noninner hn hK).2.2] using stable
  obtain ⟨final,he,hf,hv,hu⟩ := run_io _ M (RealAddress.rounds K C) Y F val base
    (fun _ _ ha ↦ ha.1) step ps base (preserved_refl Y base)
  refine ⟨final,he,frame_mono ?_ hf,?_,hu⟩
  · rintro a ⟨p,hp,hy,n,k,hn,_,hk,hoff⟩
    have hp' := (mem_grid.mp ((schedule p).1 hp)).1
    exact ⟨hy,hoff ▸ RealAddress.non_inner_bound hp' hn hk⟩
  · intro r n k hr hn hk
    have hp : (r,k/C) ∈ ps := (schedule (r,k/C)).2 (mem_grid.mpr ⟨hr,owner_lt hC hk⟩)
    have lane := Nat.mod_lt k hC
    have off := owner_offset k C
    have got := hv (Y,r*N*K+n*K+k) ⟨(r,k/C),hp,rfl,n,k%C,hn,lane,by simpa [off] using hk,by simp [off]⟩
    simpa only [val,backwardNonInnerValue,(decode_noninner hn hk).1,
      (decode_noninner hn hk).2.1,(decode_noninner hn hk).2.2] using got

#axiomsClean backward_inner_grid
#axiomsClean backward_noninner_grid
#specNonCircular backwardInnerValue avoiding [run,exec]
#specNonCircular backwardNonInnerValue avoiding [run,exec]
#guard_msgs (drop info) in
#auditModuleAxioms
end VeriTile.Bench.TritonBenchG.SoftmaxFlaggems.HostGrid
